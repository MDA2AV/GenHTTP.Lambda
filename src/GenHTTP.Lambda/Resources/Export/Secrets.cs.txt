namespace Platform;

/// <summary>
/// The secrets the platform kept for the app, read from environment variables
/// of the same name: Secret.Read("STRIPE_KEY") is $STRIPE_KEY.
/// </summary>
/// <remarks>
/// The values stayed on the platform - nobody can read them back from there -
/// so set the variables before the app starts.
/// </remarks>
public sealed class Secrets
{
    /// <summary>The value, or an exception naming the variable that is not set.</summary>
    public string Read(string name)
        => Environment.GetEnvironmentVariable(name)
        ?? throw new KeyNotFoundException($"There is no secret called '{name}': set the environment variable {name}.");

    /// <summary>Whether the environment variable is set.</summary>
    public bool Exists(string name) => Environment.GetEnvironmentVariable(name) != null;
}
