namespace Platform;

/// <summary>
/// What the platform provided: two folders, and the secrets.
/// </summary>
public static partial class LambdaEnvironment
{
    /// <summary>What the app writes, kept in "workspace" in the directory it runs in.</summary>
    public static readonly Folder Workspace = new("workspace");

    /// <summary>What the app shipped with, resources/, copied beside the program when it is built.</summary>
    public static readonly ResourceFolder Resources = new(Path.Combine(AppContext.BaseDirectory, "resources"));

    /// <summary>What the resources were called before.</summary>
    public static ResourceFolder Assets => Resources;

    /// <summary>The secrets it reads, which are environment variables here.</summary>
    public static readonly Secrets Secret = new();
}

/// <summary>
/// Lets every file say Workspace, Resources and Secret, as it could on the platform.
/// </summary>
/// <remarks>
/// Not Assets, the old name of the resources: GenHTTP.Modules.Files has a type
/// called Assets, and a member of the same name imported beside it would make
/// both ambiguous.
/// </remarks>
public static partial class LambdaScope
{
    public static Folder Workspace => LambdaEnvironment.Workspace;

    public static ResourceFolder Resources => LambdaEnvironment.Resources;

    public static Secrets Secret => LambdaEnvironment.Secret;
}
