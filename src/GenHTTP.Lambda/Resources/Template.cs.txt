var books = new ConcurrentDictionary<int, Book>
{
    [1] = new Book(1, "The Hobbit", "J. R. R. Tolkien"),
    [2] = new Book(2, "Dune", "Frank Herbert")
};

var sequence = books.Count;

var service = Inline.Create()
                    .Get(() => books.Values.OrderBy(b => b.Id))
                    .Get(":id", (int id) => Find(id))
                    .Post((Book book) =>
                    {
                        var created = book with { Id = Interlocked.Increment(ref sequence) };

                        books[created.Id] = created;

                        return new Result<Book>(created).Status(ResponseStatus.Created);
                    })
                    .Delete(":id", (int id) =>
                    {
                        if (!books.TryRemove(id, out _))
                        {
                            throw new ProviderException(ResponseStatus.NotFound, $"There is no book with ID {id}.");
                        }
                    });

Book Find(int id) => books.TryGetValue(id, out var book)
                   ? book
                   : throw new ProviderException(ResponseStatus.NotFound, $"There is no book with ID {id}.");

var index = Resource.FromString("""
                                <!doctype html>
                                <title>My Lambda</title>
                                <h1>It works</h1>
                                <p>Try <a href="books/">books/</a> or the <a href="scalar/">API browser</a>.</p>
                                """)
                    .Type(ContentType.TextHtml);

return Layout.Create()
             .Index(Content.From(index))
             .Add("books", service)
             .AddOpenApi()
             .AddScalar();

record Book(int Id, string Title, string Author);
