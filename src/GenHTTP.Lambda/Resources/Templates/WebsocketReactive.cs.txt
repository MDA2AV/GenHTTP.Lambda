// The same websocket, written as a class. Connect to it with
//
//   wss://<this-host>/lambda/{{KEY}}/
//
// The platform calls you when something happens, which is usually what you
// want: no loop to write, and no frame types to recognise.
//
// Connections are not thread safe and the methods run on the message pump -
// keep them short, and start a Task for anything that takes a while.

return Websocket.Reactive().Handler(new Chat());

public class Chat : IReactiveHandler
{
    private static readonly List<IReactiveConnection> Clients = [];

    public ValueTask OnConnected(IReactiveConnection connection)
    {
        lock (Clients)
        {
            Clients.Add(connection);
        }

        return connection.WritePayloadAsync("welcome");
    }

    public async ValueTask OnMessage(IReactiveConnection connection, IWebsocketFrame message)
    {
        var text = await message.ReadPayloadAsync<string>();

        IReactiveConnection[] targets;

        lock (Clients)
        {
            targets = Clients.ToArray();
        }

        foreach (var client in targets)
        {
            await client.WritePayloadAsync($"someone said: {text}");
        }
    }

    public ValueTask OnClose(IReactiveConnection connection, IWebsocketFrame message)
    {
        lock (Clients)
        {
            Clients.Remove(connection);
        }

        return ValueTask.CompletedTask;
    }
}
