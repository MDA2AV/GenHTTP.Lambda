// The shop itself. One file, no build step, no framework.
//
// It keeps a token in local storage and sends it with everything: that token
// is the basket, and there is nothing else to sign in with. Prices are only
// ever formatted here, never worked out - the pence come from the server and
// so does the total, because a shop where the two halves add up separately is
// a shop where they eventually disagree.

<script type="module">
const base = location.pathname.replace(/\/$/, '');
const $ = (id) => document.getElementById(id);
const escape = (s) => (s ?? '').replace(/[&<>"]/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]));
const money = (pence) => `£${(pence / 100).toFixed(2)}`;

// the token is the basket, and the only thing this browser is known by
const token = (() => {
  let held = localStorage.getItem('shelf-token');
  if (!held) { held = crypto.randomUUID().replace(/-/g, ''); localStorage.setItem('shelf-token', held); }
  return held;
})();

const TINTS = ['slate', 'rose', 'amber', 'green', 'sky', 'violet'];
let tint = TINTS[0];
let items = [];
let basket = { lines: [], count: 0, pence: 0, total: '£0.00' };

async function api(method, path, body) {
  const response = await fetch(`${base}/api${path}`, {
    method,
    headers: body === undefined ? {} : { 'Content-Type': 'application/json' },
    body: body === undefined ? undefined : JSON.stringify(body),
  });
  return response.json();
}

function quantityOf(id) {
  return basket.lines.find((l) => l.itemId === id)?.quantity ?? 0;
}

function drawShelf() {
  $('grid').innerHTML = items.map((item) => {
    const have = quantityOf(item.id);
    const gone = item.stock === 0;

    return `
      <div class="card">
        <div class="top" style="background:var(--${item.tint})"></div>
        <div class="body">
          <h3>${escape(item.name)}</h3>
          <p>${escape(item.blurb)}</p>
          <div class="row between">
            <span class="price">${money(item.pence)}</span>
            <span class="${gone ? 'out' : 'stock'}">${gone ? 'sold out' : `${item.stock} left`}</span>
          </div>
          <div class="row" style="margin-top:.7rem">
            ${have === 0
              ? `<button class="btn" style="flex:1" data-add="${item.id}" ${gone ? 'disabled' : ''}>Add to basket</button>`
              : `<span class="step">
                   <button data-set="${item.id}" data-q="${have - 1}">−</button>
                   <span>${have}</span>
                   <button data-set="${item.id}" data-q="${have + 1}" ${have >= item.stock ? 'disabled' : ''}>+</button>
                 </span>`}
          </div>
        </div>
      </div>`;
  }).join('');
}

function drawBasket() {
  $('count').textContent = basket.count;
  $('total').textContent = basket.total;

  $('lines').innerHTML = basket.lines.length === 0
    ? '<p class="empty">Nothing in it yet.</p>'
    : basket.lines.map((line) => `
        <div class="line">
          <span class="name">${escape(line.name)}</span>
          <span class="step">
            <button data-set="${line.itemId}" data-q="${line.quantity - 1}">−</button>
            <span>${line.quantity}</span>
            <button data-set="${line.itemId}" data-q="${line.quantity + 1}" ${line.quantity >= line.stock ? 'disabled' : ''}>+</button>
          </span>
          <span class="price">${money(line.pence * line.quantity)}</span>
        </div>`).join('');
}

function complain(where, text) {
  const box = $(where);
  box.textContent = text ?? '';
  box.hidden = !text;
}

async function refresh() {
  [items, basket] = await Promise.all([api('GET', '/items'), api('GET', `/basket/${token}`)]);
  drawShelf();
  drawBasket();
}

async function set(id, quantity) {
  const answer = await api('PUT', `/basket/${token}`, { itemId: id, quantity });
  basket = answer.basket;
  complain('basketProblem', answer.ok ? null : answer.problem);
  drawShelf();
  drawBasket();
  if (!answer.ok) setTimeout(() => complain('basketProblem', null), 3500);
}

document.addEventListener('click', (event) => {
  const add = event.target.closest('[data-add]');
  if (add) return set(add.dataset.add, 1);

  const step = event.target.closest('[data-set]');
  if (step) return set(step.dataset.set, Number(step.dataset.q));
});

function showBasket(open) {
  $('basket').hidden = !open;
  $('shade').hidden = !open;
}

$('open').onclick = () => showBasket(true);
$('close').onclick = () => showBasket(false);
$('shade').onclick = () => showBasket(false);

$('checkout').onclick = async () => {
  const answer = await api('POST', `/orders/${token}`, { buyer: $('buyer').value });

  if (!answer.ok) {
    complain('basketProblem', answer.problem);
    return;
  }

  complain('basketProblem', null);
  showBasket(false);

  $('doneRef').textContent = answer.order.reference;
  $('doneTotal').textContent = money(answer.order.pence);
  $('doneDialog').showModal();

  await refresh();
};

$('tints').innerHTML = TINTS.map((t, i) =>
  `<button type="button" data-tint="${t}" aria-pressed="${i === 0}" style="background:var(--${t})"></button>`).join('');

$('tints').onclick = (event) => {
  const chosen = event.target.closest('[data-tint]');
  if (!chosen) return;
  tint = chosen.dataset.tint;
  for (const button of $('tints').children) button.setAttribute('aria-pressed', button === chosen);
};

$('sell').onclick = () => { complain('sellProblem', null); $('sellDialog').showModal(); };
$('sellCancel').onclick = () => $('sellDialog').close();
$('doneGo').onclick = () => $('doneDialog').close();
$('ordersGo').onclick = () => $('ordersDialog').close();

$('sellForm').onsubmit = async (event) => {
  event.preventDefault();

  const answer = await api('POST', '/items', {
    name: $('itemName').value,
    blurb: $('itemBlurb').value,
    // pounds on the screen, pence on the wire: the server counts in the
    // smallest unit there is and never sees a fraction
    pence: Math.round(Number($('itemPrice').value) * 100),
    stock: Number($('itemStock').value),
    tint,
  });

  if (!answer.ok) { complain('sellProblem', answer.problem); return; }

  $('sellDialog').close();
  $('itemName').value = '';
  $('itemBlurb').value = '';
  await refresh();
};

$('history').onclick = async () => {
  const orders = await api('GET', `/orders/${token}`);

  $('ordersList').innerHTML = orders.length === 0
    ? '<p class="empty">Nothing yet.</p>'
    : orders.map((o) => `
        <div class="line receipt">
          <span class="name"><span class="ref">${escape(o.reference)}</span><br>
            <small>${o.lines.map((l) => `${l.quantity}× ${escape(l.name)}`).join(', ')}</small></span>
          <span class="price">${money(o.pence)}</span>
        </div>`).join('');

  $('ordersDialog').showModal();
};

refresh();
</script>
</html>
""";
}
