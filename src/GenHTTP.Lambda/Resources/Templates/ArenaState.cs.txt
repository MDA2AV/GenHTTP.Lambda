// What the server keeps about a player while they are in the arena.
//
// Mutable, because it is stepped twenty times a second and rebuilding every
// player to move them a few units would be a lot of rubbish for the collector.
// What leaves the process is the immutable shape in Protocol.cs.

/// <summary>
/// One pellet on the floor.
/// </summary>
/// <remarks>
/// Mutable for the same reason a player is: pellets are dragged about by
/// whatever is big enough to drag them, and replacing seven hundred records
/// twenty times a second to move some of them a few units would be a lot of
/// rubbish for no gain.
///
/// The slot is a small number rather than a name, recycled when a pellet is
/// eaten, so that the browser can tell which pellet is which across two
/// frames without paying for an identifier per pellet per tick.
/// </remarks>
sealed class Pellet
{
    public int Slot;

    public double X;
    public double Y;

    /// <summary>Which of the palette's colours it is.</summary>
    public int C;

    /// <summary>
    /// How much it is worth, as a grade rather than an amount.
    /// </summary>
    /// <remarks>
    /// Ordinary ones are grade zero. The bigger grades are what a large blob
    /// sheds when something hurts it.
    /// </remarks>
    public int Grade;
}

/// <summary>
/// A harpoon in the air.
/// </summary>
/// <remarks>
/// It travels rather than simply hitting, which is what makes it a thing to
/// aim rather than a button to press: it takes the better part of a second
/// to cross its range, and anything it is thrown at can move.
/// </remarks>
sealed class Spear
{
    public string Owner;

    public double X;
    public double Y;

    public double DX;
    public double DY;

    /// <summary>Ticks left before it falls short.</summary>
    public int Life;
}

/// <summary>
/// Something lying on the floor that grants an ability to whoever takes it.
/// </summary>
/// <remarks>
/// Anchored: gravity does not move it and a large blob cannot take it. It is
/// the small player's tool, and a giant that could hoover them up would
/// simply be collecting the answers to itself.
/// </remarks>
sealed class Prize
{
    public int Slot;

    public double X;
    public double Y;

    /// <summary>Which ability it grants.</summary>
    public int Perk;

    /// <summary>Ticks left before it goes somewhere else.</summary>
    public int Life;
}

sealed class Player
{
    public string Id;
    public string Name;
    public string Colour;

    public double X;
    public double Y;
    public double Radius;

    /// <summary>Where the browser last said it would like to be.</summary>
    public double WantX;
    public double WantY;

    /// <summary>
    /// Whether they are holding the space bar: rooted in place, pulling
    /// everything nearby much harder, and burning themselves to do it.
    /// </summary>
    public bool Hold;

    /// <summary>How long they have been holding, which the pull ramps over.</summary>
    public double Wound;

    /// <summary>How long until they may hold again.</summary>
    public double Rest;

    /// <summary>
    /// How hard everything bigger is dragging them this tick, in units a
    /// second, worked out before anybody has moved so that the order players
    /// happen to be in cannot change the outcome.
    /// </summary>
    public double PullX;
    public double PullY;

    /// <summary>
    /// Whether any of that pull came from somebody holding on, which is what
    /// decides how much of their own speed they can be robbed of.
    /// </summary>
    public bool Clutched;

    /// <summary>
    /// What they are carrying, if anything: see Perk in Protocol.cs. One at
    /// a time, and picking up a second while holding one does nothing.
    /// </summary>
    public int Perk;

    /// <summary>How long is left of a lit fuse, or nought if none is lit.</summary>
    public double Lit;

    /// <summary>
    /// Set by the socket when they press, and acted on by the clock.
    /// </summary>
    /// <remarks>
    /// The socket does not spend the thing itself. Everything that changes
    /// the world happens on the one thread that steps it, or a harpoon
    /// thrown while the clock is halfway through walking the list of them
    /// lands in a list being read by somebody else.
    /// </remarks>
    public bool Spending;

    /// <summary>
    /// How much area has been taken off them by abilities lately, and when
    /// that tally was last started, so that no amount of coordination can
    /// delete somebody in one go.
    /// </summary>
    public double Bruised;
    public double Bruising;

    /// <summary>
    /// Where a harpoon is dragging them, and for how long. A hit hauls the
    /// thrower in towards what they hit - which is the point of it: the
    /// harpoon is also the claim ticket for what falls out.
    /// </summary>
    public double HauledX;
    public double HauledY;
    public double Hauling;

    /// <summary>Set when somebody has eaten them; the socket is told and closed.</summary>
    public bool Gone;
    public string EatenBy;

    /// <summary>When this blob may teleport again, on the arena clock.</summary>
    public double WarpReady;

    /// <summary>Where a teleport already under way is taking this blob.</summary>
    public double WarpToX;

    /// <summary>The other half of <see cref="WarpToX"/>.</summary>
    public double WarpToY;

    /// <summary>
    /// The arena clock reading at which a teleport already under way
    /// finishes, or nought when there is not one.
    /// </summary>
    public double WarpLands;

    /// <summary>
    /// Who this blob has eaten since the last frame went out, which is worth
    /// saying on their screen and nowhere else.
    /// </summary>
    public List<string> Ate;

    /// <summary>
    /// The line this blob is running, if it is running one.
    /// </summary>
    /// <remarks>
    /// Kept as the points the browser drew rather than as a direction,
    /// because the point of it is the shape: a curve round the back of
    /// somebody is a different thing from the straight line between its ends,
    /// and reducing it to a heading would throw away exactly the part the
    /// player chose.
    /// </remarks>
    public double[] StreakX;
    public double[] StreakY;

    /// <summary>Which leg of the line is being run, and how far along it.</summary>
    public int StreakLeg;
    public double StreakGone;

    /// <summary>Whether nobody is driving this one.</summary>
    public bool Bot;

    /// <summary>What a bot is heading for, and when it will think again.</summary>
    public double AimX;
    public double AimY;
    public double Rethink;
}
