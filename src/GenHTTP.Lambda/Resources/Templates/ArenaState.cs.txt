// What the server keeps about a player while they are in the arena.
//
// Mutable, because it is stepped twenty times a second and rebuilding every
// player to move them a few units would be a lot of rubbish for the collector.
// What leaves the process is the immutable shape in Protocol.cs.

/// <summary>
/// One pellet on the floor.
/// </summary>
/// <remarks>
/// Mutable for the same reason a player is: pellets are dragged about by
/// whatever is big enough to drag them, and replacing seven hundred records
/// twenty times a second to move some of them a few units would be a lot of
/// rubbish for no gain.
///
/// The slot is a small number rather than a name, recycled when a pellet is
/// eaten, so that the browser can tell which pellet is which across two
/// frames without paying for an identifier per pellet per tick.
/// </remarks>
sealed class Pellet
{
    public int Slot;

    public double X;
    public double Y;

    /// <summary>Which of the palette's colours it is.</summary>
    public int C;

    /// <summary>
    /// How much it is worth, as a grade rather than an amount.
    /// </summary>
    /// <remarks>
    /// Ordinary ones are grade zero. The bigger grades are what a large blob
    /// sheds when something hurts it.
    /// </remarks>
    public int Grade;
}

sealed class Player
{
    public string Id;
    public string Name;
    public string Colour;

    public double X;
    public double Y;
    public double Radius;

    /// <summary>Where the browser last said it would like to be.</summary>
    public double WantX;
    public double WantY;

    /// <summary>
    /// Whether they are holding the space bar: rooted in place, pulling
    /// everything nearby much harder, and burning themselves to do it.
    /// </summary>
    public bool Hold;

    /// <summary>How long they have been holding, which the pull ramps over.</summary>
    public double Wound;

    /// <summary>How long until they may hold again.</summary>
    public double Rest;

    /// <summary>
    /// How hard everything bigger is dragging them this tick, in units a
    /// second, worked out before anybody has moved so that the order players
    /// happen to be in cannot change the outcome.
    /// </summary>
    public double PullX;
    public double PullY;

    /// <summary>
    /// Whether any of that pull came from somebody holding on, which is what
    /// decides how much of their own speed they can be robbed of.
    /// </summary>
    public bool Clutched;

    /// <summary>Set when somebody has eaten them; the socket is told and closed.</summary>
    public bool Gone;
    public string EatenBy;
}
