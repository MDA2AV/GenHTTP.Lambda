// What the server keeps about somebody while they are in the arena.
//
// Mutable, because it is stepped twenty times a second. What leaves the
// process is the immutable shape in Protocol.cs.

sealed class Fighter
{
    public string Id;
    public string Name;

    /// <summary>Which end they arrive at. Nought or one.</summary>
    public int Side;

    public double X;
    public double Y;
    public double Z;

    public double VX;
    public double VY;
    public double VZ;

    /// <summary>Where they are looking, in degrees, as the browser last said.</summary>
    public double Yaw;
    public double Pitch;

    /// <summary>
    /// A heading the browser is being asked to adopt, or nothing.
    /// </summary>
    /// <remarks>
    /// Arriving somewhere and being turned to face the way out only works if
    /// the browser is told, because the browser owns the aim: it sends one
    /// twenty times a second and the next one would put it straight back. It
    /// is set the moment somebody arrives and cleared the moment it has been
    /// sent, so it cannot be lost to an input that happened to land in
    /// between.
    /// </remarks>
    public double? Turn;

    /// <summary>What is held down: see Buttons in Protocol.cs.</summary>
    public int Held;

    /// <summary>
    /// How far into crouching they are, nought to one.
    /// </summary>
    /// <remarks>
    /// A fraction rather than a flag, because going down takes a tenth of a
    /// second and being halfway matters: it is the part of a peek where the
    /// head is still up.
    /// </remarks>
    public double Folded;

    public bool Grounded;

    public double Health;

    public int Ammo;

    /// <summary>When the magazine will be full again, or nought.</summary>
    public double Reloading;

    /// <summary>When they may shoot again, and how many they have held down.</summary>
    public double Ready;
    public int Burst;

    /// <summary>When the barrel was last used, so it knows when to settle.</summary>
    public double Fired;

    /// <summary>When they come back, or nought if they are already here.</summary>
    public double Returns;

    /// <summary>Until when they cannot be hurt.</summary>
    public double Sheltered;

    public int Kills;
    public int Deaths;

    /// <summary>
    /// What the browser last acknowledged, echoed back so it knows which of
    /// its own guesses have been answered and which it is still replaying.
    /// </summary>
    public int Seen;

    /// <summary>Set when they were hurt this tick, and from where.</summary>
    public double Hurt;
    public double HurtFrom;

    public bool Alive => Returns <= 0;

    public double Height => World.Height(Folded > 0.5);

    public double Eye => Y + World.EyeTall - (World.EyeTall - World.EyeSquat) * Folded;
}

/// <summary>
/// Where everybody was at one moment, kept so that a shot can be judged
/// against what the shooter was actually looking at.
/// </summary>
/// <remarks>
/// A browser draws a tenth of a second in the past, so the thing it aimed at
/// was a tenth of a second old before the shot even left. Judging that
/// against where everybody is now means missing people who were plainly in
/// the sights, which is the oldest complaint in the genre. Eight of these is
/// four hundred milliseconds, and no shot is honoured further back than that.
/// </remarks>
sealed class Moment
{
    public double At;

    public (string Id, double X, double Y, double Z, double Height)[] Bodies;
}

/// <summary>A shot somebody says they took, waiting to be judged.</summary>
sealed class Trigger
{
    public Fighter By;

    public double Yaw;
    public double Pitch;

    /// <summary>The moment the browser was drawing when it pulled.</summary>
    public double Saw;
}
