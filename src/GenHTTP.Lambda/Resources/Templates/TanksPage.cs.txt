// The browser half: builds the labyrinth it is told about, draws what the
// server says is in it, and sends what the keys and the mouse are doing.
//
// It never decides anything. Where the tank is, whether a shot hit, who is
// alive - all of that arrives from the server, and the only cleverness here is
// walking each shape towards where it was last said to be so that twenty
// snapshots a second look like movement rather than teleporting.

static class Site
{
    public const string Page =
"""

  #log div { background: #12151acc; padding: 3px 8px; margin-bottom: 3px; }

  #dead { position: fixed; inset: 0; display: grid; place-items: center; background: #07090cbb; font-size: 22px; }

  button { padding: 9px 20px; border: 0; background: #1a73e8; color: #fff; font: inherit; cursor: pointer; }
</style>

<div id="gate"><div>
  <h1>Tanks</h1>
  <p class="dim">
    A labyrinth, everybody in it at once. <kbd>W</kbd><kbd>S</kbd> to drive,
    <kbd>A</kbd><kbd>D</kbd> to turn, the mouse to aim and click to fire.
    Drive over a crate: green patches you up, yellow reloads faster, blue takes one hit for you.
  </p>
  <p><input id="name" maxlength="14" placeholder="Your name"> <button id="go">Roll out</button></p>

  <div><b id="score">0</b> <span class="dim">knocked out</span></div>
  <div class="bar"><i id="health" style="width:100%"></i></div>
  <div class="bar boost" hidden id="boostbar"><i id="boost" style="width:0"></i></div>

<div id="board" hidden><h2>In the maze</h2><ul id="list"></ul></div>
<div id="log"></div>
<div id="dead" hidden>knocked out</div>

<script type="module">
import * as THREE from 'https://cdn.jsdelivr.net/npm/three@0.170.0/build/three.module.js';

const base = location.pathname.replace(/\/$/, '');
const $ = (id) => document.getElementById(id);

let socket = null, me = null, cell = 4, width = 19;
let tanks = new Map(), shells = new Map(), crates = new Map();
const keys = new Set();
let aim = 0, firing = false;

const scene = new THREE.Scene();
scene.fog = new THREE.Fog(0x07090c, 34, 96);

const camera = new THREE.PerspectiveCamera(58, innerWidth / innerHeight, 0.1, 300);
const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
renderer.setSize(innerWidth, innerHeight);
document.body.appendChild(renderer.domElement);

scene.add(new THREE.HemisphereLight(0x94b8ff, 0x0b0d11, 1.15));
const sun = new THREE.DirectionalLight(0xffffff, 1.25);
sun.position.set(10, 22, 8);
scene.add(sun);

const ground = new THREE.Mesh(new THREE.PlaneGeometry(1, 1),
                              new THREE.MeshStandardMaterial({ color: 0x2e3440, roughness: .95 }));
ground.rotation.x = -Math.PI / 2;
scene.add(ground);

const aimPlane = new THREE.Plane(new THREE.Vector3(0, 1, 0), 0);
const pointer = new THREE.Vector2();
const caster = new THREE.Raycaster();
const spot = new THREE.Vector3();

function build(walls) {
  const size = width * cell;
  ground.scale.set(size, size, 1);

  const wall = new THREE.BoxGeometry(cell, 3, cell);
  const skin = new THREE.MeshStandardMaterial({ color: 0x11141a, roughness: .8 });

  let solid = 0;
  for (const c of walls) if (c === '1') solid++;

  // one instanced mesh for the whole labyrinth: a few hundred separate boxes
  // is a few hundred draw calls, and they never move
  const blocks = new THREE.InstancedMesh(wall, skin, solid);
  const at = new THREE.Object3D();

  // a line where each square of the floor meets the next, so movement has
  // something to read against
  const lines = new THREE.GridHelper(size, width, 0x3d4552, 0x39404c);
  lines.position.y = 0.02;
  scene.add(lines);

  let n = 0;
  for (let z = 0; z < width; z++) {
    for (let x = 0; x < width; x++) {
      if (walls[z * width + x] !== '1') continue;
      at.position.set((x - width / 2 + 0.5) * cell, 1.5, (z - width / 2 + 0.5) * cell);
      at.updateMatrix();
      blocks.setMatrixAt(n++, at.matrix);
    }
  }

  scene.add(blocks);
}

const hullGeometry = new THREE.BoxGeometry(2, 0.85, 2.6);
const turretGeometry = new THREE.BoxGeometry(1.1, 0.6, 1.1);
const barrelGeometry = new THREE.BoxGeometry(0.22, 0.22, 1.9);
const shellGeometry = new THREE.SphereGeometry(0.22, 8, 8);
const shellSkin = new THREE.MeshStandardMaterial({ color: 0xfff1b8, emissive: 0x8a6b00 });
const crateGeometry = new THREE.BoxGeometry(0.9, 0.9, 0.9);
const crateSkins = {
  repair: new THREE.MeshStandardMaterial({ color: 0x81c995, emissive: 0x13361f }),
  rapid: new THREE.MeshStandardMaterial({ color: 0xfdd663, emissive: 0x3d3000 }),
  shield: new THREE.MeshStandardMaterial({ color: 0x78d9ec, emissive: 0x0d3a44 }),
};

function tankFor(info) {
  const group = new THREE.Group();
  const skin = new THREE.MeshStandardMaterial({ color: info.colour, roughness: .45 });

  const hull = new THREE.Mesh(hullGeometry, skin);
  hull.position.y = 0.45;
  group.add(hull);

  const turret = new THREE.Group();
  const dome = new THREE.Mesh(turretGeometry, skin);
  dome.position.y = 1.05;
  turret.add(dome);

  const barrel = new THREE.Mesh(barrelGeometry, skin);
  barrel.position.set(0, 1.05, 1.1);
  turret.add(barrel);

  group.add(turret);
  group.userData.turret = turret;
  scene.add(group);

  return group;
}

function paint(state) {
  const seen = new Set();

  for (const info of state.tanks) {
    seen.add(info.id);

    let group = tanks.get(info.id);
    if (!group) { group = tankFor(info); tanks.set(info.id, group); }

    group.visible = info.alive;
    group.userData.to = { x: info.x, z: info.z, heading: info.heading, aim: info.aim };

    if (info.id === me) {
      $('score').textContent = info.score;
      $('health').style.width = `${Math.max(0, info.health) / 3 * 100}%`;
      $('dead').hidden = info.alive;
      $('boostbar').hidden = info.boost <= 0;
      $('boost').style.width = `${Math.min(100, info.boost / 8 * 100)}%`;
    }
  }

  for (const [id, group] of tanks) {
    if (!seen.has(id)) { scene.remove(group); tanks.delete(id); }
  }

  const flying = new Set();
  for (const info of state.shells) {
    flying.add(info.id);
    let mesh = shells.get(info.id);
    if (!mesh) {
      mesh = new THREE.Mesh(shellGeometry, shellSkin);
      scene.add(mesh);
      shells.set(info.id, mesh);
    }
    mesh.position.set(info.x, 0.9, info.z);
  }
  for (const [id, mesh] of shells) {
    if (!flying.has(id)) { scene.remove(mesh); shells.delete(id); }
  }

  const standing = new Set();
  for (const info of state.pickups) {
    standing.add(info.id);
    if (crates.has(info.id)) continue;
    const mesh = new THREE.Mesh(crateGeometry, crateSkins[info.sort] ?? crateSkins.repair);
    mesh.position.set(info.x, 0.6, info.z);
    scene.add(mesh);
    crates.set(info.id, mesh);
  }
  for (const [id, mesh] of crates) {
    if (!standing.has(id)) { scene.remove(mesh); crates.delete(id); }
  }

  $('list').innerHTML = state.tanks.map((t) =>
    `<li class="${t.alive ? '' : 'down'}">
       <span><span class="dot" style="background:${t.colour}"></span>${t.name.replace(/[<&]/g, '')}</span>
       <span>${t.score}</span>
     </li>`).join('');
}

function say(text) {
  const line = document.createElement('div');
  line.textContent = text;
  $('log').prepend(line);
  setTimeout(() => line.remove(), 5000);
  while ($('log').children.length > 4) $('log').lastChild.remove();
}

addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
});

addEventListener('keydown', (e) => { keys.add(e.key.toLowerCase()); if (e.key === ' ') firing = true; });
addEventListener('keyup', (e) => { keys.delete(e.key.toLowerCase()); if (e.key === ' ') firing = false; });
addEventListener('pointermove', (e) => {
  pointer.x = (e.clientX / innerWidth) * 2 - 1;
  pointer.y = -(e.clientY / innerHeight) * 2 + 1;
});
addEventListener('pointerdown', () => { firing = true; });
addEventListener('pointerup', () => { firing = false; });
addEventListener('contextmenu', (e) => e.preventDefault());

function frame(now) {
  requestAnimationFrame(frame);

  for (const group of tanks.values()) {
    const to = group.userData.to;
    if (!to) continue;
    group.position.x += (to.x - group.position.x) * 0.3;
    group.position.z += (to.z - group.position.z) * 0.3;
    group.rotation.y = to.heading;
    group.userData.turret.rotation.y = to.aim - to.heading;
  }

  for (const mesh of crates.values()) {
    mesh.rotation.y = now / 900;
    mesh.position.y = 0.6 + Math.sin(now / 500) * 0.12;
  }

  const mine = tanks.get(me);

  if (mine) {
    // the turret follows the mouse across the floor, which is what makes
    // aiming feel like aiming rather than like steering
    caster.setFromCamera(pointer, camera);
    if (caster.ray.intersectPlane(aimPlane, spot)) {
      aim = Math.atan2(spot.x - mine.position.x, spot.z - mine.position.z);
    }

    /*
     * Over the shoulder of the turret rather than of the hull: what a player
     * is doing with a tank is aiming, and a camera that swings with the hull
     * turns the screen every time they steer.
     */
    const behind = 11, above = 16;

    camera.position.set(mine.position.x - Math.sin(aim) * behind, above, mine.position.z - Math.cos(aim) * behind);
    camera.lookAt(mine.position.x + Math.sin(aim) * 4, 0.5, mine.position.z + Math.cos(aim) * 4);
  }

  renderer.render(scene, camera);
}

requestAnimationFrame(frame);

setInterval(() => {
  if (socket?.readyState !== 1 || !me) return;

  const throttle = (keys.has('w') || keys.has('arrowup') ? 1 : 0) + (keys.has('s') || keys.has('arrowdown') ? -1 : 0);
  const steer = (keys.has('d') || keys.has('arrowright') ? 1 : 0) + (keys.has('a') || keys.has('arrowleft') ? -1 : 0);

  socket.send(JSON.stringify({ kind: 'input', throttle, steer, aim, fire: firing }));
}, 50);

$('go').onclick = () => {
  const address = new URL(`${base}/join`, location.href);
  address.protocol = address.protocol === 'https:' ? 'wss:' : 'ws:';

  socket = new WebSocket(address);
  socket.onopen = () => socket.send(JSON.stringify({ kind: 'join', name: $('name').value }));

  socket.onmessage = (event) => {
    const message = JSON.parse(event.data);

    if (message.kind === 'welcome') {
      me = message.id;
      cell = message.cell;
      width = message.width;
      build(message.walls);
      $('gate').hidden = true;
      $('hud').hidden = false;
      $('board').hidden = false;
    } else if (message.kind === 'state') {
      paint(message);
    } else if (message.kind === 'event') {
      say(message.text);
    }
  };

  socket.onclose = () => say('lost the connection');
};
</script>
</html>
""";
}
