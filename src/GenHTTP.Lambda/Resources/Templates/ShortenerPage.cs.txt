// The page it is served as.

static class Site
{
    public const string Page =
"""
<!doctype html>
<html lang="en">
<meta charset="utf-8">
<meta name="viewport" content="width=device-width,initial-scale=1">
<title>Shortener</title>
<style>
  :root { color-scheme: light dark; --line: #d5d7da; --dim: #5f6368; }
  * { box-sizing: border-box; }
  body { margin: 0 auto; max-width: 40rem; padding: 3rem 1.25rem;
         font: 15px/1.6 -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, sans-serif; }
  h1 { font-size: 20px; margin: 0 0 .25rem; }
  p.dim { color: var(--dim); margin-top: 0; }
  form { display: flex; gap: .5rem; margin: 1.5rem 0; }
  input { flex: 1; min-width: 0; padding: .6rem .7rem; border: 1px solid var(--line);
          background: transparent; color: inherit; font: inherit; }
  button { padding: .6rem 1.1rem; border: 0; background: #1a73e8; color: #fff; font: inherit; cursor: pointer; }
  table { width: 100%; border-collapse: collapse; font-size: 14px; }
  th { text-align: left; font-weight: 500; color: var(--dim); font-size: 12px;
       text-transform: uppercase; letter-spacing: .05em; padding-bottom: .4rem; }
  td { padding: .45rem 0; border-top: 1px solid var(--line); }
  td.n { text-align: right; color: var(--dim); font-variant-numeric: tabular-nums; }
  a { color: #1a73e8; }
  code { font-family: ui-monospace, Menlo, Consolas, monospace; }
  .problem { color: #c5221f; font-size: 13px; }
  .target { color: var(--dim); display: block; max-width: 22rem; overflow: hidden;
            text-overflow: ellipsis; white-space: nowrap; font-size: 12px; }
</style>

<h1>Shortener</h1>
<p class="dim">Paste an address, get a shorter one. Everything here is public.</p>

<form id="make">
  <input id="address" type="url" placeholder="https://example.com/something/long" required>
  <button>Shorten</button>
</form>

<p class="problem" id="problem"></p>

<table>
  <thead><tr><th>Short</th><th>Points at</th><th class="n">Follows</th></tr></thead>
  <tbody id="rows"></tbody>
</table>

<script type="module">
const base = location.pathname.replace(/\/$/, '');
const $ = (id) => document.getElementById(id);
const escape = (s) => s.replace(/[&<>"]/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]));

async function show() {
  const links = await (await fetch(`${base}/api/links`)).json();
  $('rows').innerHTML = links.map((l) => `
    <tr>
      <td><a href="${base}/go/${l.key}"><code>${l.key}</code></a></td>
      <td><span class="target" title="${escape(l.address)}">${escape(l.address)}</span></td>
      <td class="n">${l.follows}</td>
    </tr>`).join('');
}

$('make').onsubmit = async (event) => {
  event.preventDefault();
  $('problem').textContent = '';

  const answer = await (await fetch(`${base}/api/links`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ address: $('address').value }),
  })).json();

  if (!answer.ok) { $('problem').textContent = answer.problem; return; }

  $('address').value = '';
  show();
};

show();
</script>
</html>
""";
}
