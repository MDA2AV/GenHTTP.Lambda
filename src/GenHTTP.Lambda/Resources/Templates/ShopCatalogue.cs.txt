// What is for sale, and what this will accept as something to sell.
//
// Money is counted in pence, never in pounds. A price is a whole number of the
// smallest unit there is, because 0.1 + 0.2 is not 0.3 in any language with
// floating point and a shop that adds up a basket wrongly is worse than a shop
// that will not take decimals.

/// <summary>Something on the shelf.</summary>
public record Item(string Id, string Name, string Blurb, int Pence, int Stock, string Tint, DateTime Added);

/// <summary>What is sent to put something on the shelf.</summary>
public record Listing(string Name, string Blurb, int Pence, int Stock, string Tint);

/// <summary>The answer to listing something.</summary>
public record Listed(bool Ok, string Problem, Item Item);

static class Catalogue
{
    /// <summary>How many things the shop will hold.</summary>
    public const int Shelf = 60;

    /// <summary>The dearest thing it will take, which is a thousand pounds.</summary>
    public const int Ceiling = 100_000;

    /// <summary>The colours a card can be given, so the shop stays a shop.</summary>
    public static readonly string[] Tints = ["slate", "rose", "amber", "green", "sky", "violet"];

    /// <summary>
    /// Pence as people write money, which is the only place this happens.
    /// </summary>
    /// <remarks>
    /// The browser formats its own, so this is for anything that reads the API
    /// directly. Both agree because both divide by a hundred and neither ever
    /// multiplies a price by anything.
    /// </remarks>
    public static string Money(int pence) => $"£{pence / 100}.{pence % 100:00}";

    /// <summary>
    /// Whether something may go on the shelf, and why not when it may not.
    /// </summary>
    public static string Refuse(Listing listing, int onShelf)
    {
        if (onShelf >= Shelf)
        {
            return "The shelf is full. Somebody has to buy something first.";
        }

        var name = (listing?.Name ?? "").Trim();

        if (name.Length is < 2 or > 40)
        {
            return "A name is between two and forty characters.";
        }

        if ((listing.Blurb ?? "").Trim().Length > 140)
        {
            return "A description is at most a hundred and forty characters.";
        }

        if (listing.Pence is < 1 or > Ceiling)
        {
            return $"A price is between one penny and {Money(Ceiling)}.";
        }

        if (listing.Stock is < 1 or > 999)
        {
            return "Stock is between one and nine hundred and ninety-nine.";
        }

        return null;
    }

    /// <summary>A tint that exists, whatever was asked for.</summary>
    public static string TintOr(string wanted, Random random)
        => Array.IndexOf(Tints, wanted) >= 0 ? wanted : Tints[random.Next(Tints.Length)];
}
