/* The width and height matter. A canvas is a replaced element: without them
   it lays out at its intrinsic size, which is the drawing buffer, which on a
   screen with two device pixels to the pixel is twice the window - so the
   window shows the top left quarter of the picture and the middle of the
   world sits somewhere off to the lower right. */
canvas {
  display: block;
  position: fixed;
  inset: 0;
  width: 100%;
  height: 100%;
}
