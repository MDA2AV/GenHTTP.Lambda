// A guestbook: anyone can sign it, everyone can read it.
//
//   GET  /lambda/{{KEY}}/             the page
//   GET  /lambda/{{KEY}}/api/entries  what has been written
//   POST /lambda/{{KEY}}/api/entries  add a line
//
// The entries live in Workspace, the private folder this lambda gets, so they
// are still here after a redeployment.
//
// There is no account and no login. That is the point of a guestbook, and it
// is also why nothing here trusts what it is sent: the length is capped, the
// book is capped, and what comes back is escaped by the page before it is put
// on screen.

var json = new JsonSerializerOptions
{
    PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
    PropertyNameCaseInsensitive = true
};

var book = new List<Entry>();
var writing = new object();

if (Workspace.Exists("guestbook.json"))
{
    book.AddRange(JsonSerializer.Deserialize<List<Entry>>(Workspace.ReadText("guestbook.json"), json) ?? []);
}

Signed Sign(Signature request)
{
    if (Guestbook.Refuse(request) is string problem)
    {
        return new Signed(false, problem, null);
    }

    var entry = new Entry(request.Name.Trim(), request.Message.Trim(), DateTime.UtcNow);

    lock (writing)
    {
        book.Add(entry);

        while (book.Count > Guestbook.Keep)
        {
            book.RemoveAt(0);
        }

        Workspace.WriteText("guestbook.json", JsonSerializer.Serialize(book, json));
    }

    return new Signed(true, null, entry);
}

var api = Inline.Create()
                .Get("entries", () =>
                {
                    lock (writing)
                    {
                        return book.AsEnumerable().Reverse().ToList();
                    }
                })
                .Post("entries", (Signature body) => Sign(body));

return Layout.Create()
             .Add("api", api)
             .Add(Content.From(Resource.FromString(Site.Page).Type(new ContentType("text/html; charset=utf-8"))));
