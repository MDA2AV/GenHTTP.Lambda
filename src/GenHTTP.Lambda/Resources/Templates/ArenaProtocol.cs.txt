// What the two halves say to each other.
//
// Every frame on the socket is one of these, tagged by its `kind`, so the
// client and the server can be read side by side without either having to
// guess what the other meant.

/// <summary>Sent by a browser as its first frame.</summary>
public record Join(string Kind, string Name);

/// <summary>Sent by a browser to say where it would like to be.</summary>
/// <remarks>
/// Would like to be, rather than is: the server decides. A client that asks
/// to be somewhere it could not have walked to is moved as far as it could
/// have got, which is the difference between a game and a suggestion box.
/// </remarks>
public record Move(string Kind, double X, double Z);

/// <summary>One player, as everybody else sees them.</summary>
public record Player(string Id, string Name, string Colour, double X, double Z, int Score);

/// <summary>Something to run into, worth a point.</summary>
public record Prize(string Id, double X, double Z);

/// <summary>Who you are, sent once when you arrive.</summary>
public record Welcome(string Kind, string Id, string Colour, double Half);

/// <summary>Where everything is. Sent to everybody whenever anything moves.</summary>
public record State(string Kind, List<Player> Players, List<Prize> Prizes);

/// <summary>The best runs the room remembers.</summary>
public record Standing(string Name, int Score, DateTime At);
