// What the two halves say to each other.
//
// Kept short on purpose. This is sent to everybody twenty times a second, so
// every name in it is paid for once per player per tick - which is why the
// fields are one letter and the numbers are rounded before they leave.

/// <summary>Sent by a browser as its first frame.</summary>
public record Join(string Kind, string Name);

/// <summary>Where a browser would like to be heading, sent continuously.</summary>
public record Steer(string Kind, double X, double Y);

/// <summary>A blob, as everybody else sees it.</summary>
public record Blob(string I, string N, string C, double X, double Y, double R);

/// <summary>One line of the leaderboard.</summary>
public record Score(string N, int S);

/// <summary>Who you are and how big the world is, sent once.</summary>
public record Welcome(string Kind, string Id, string Colour, double World, string[] Palette, double Well);

/// <summary>
/// What is near enough to matter, sent on every tick.
/// </summary>
/// <remarks>
/// Only what falls inside the player's view is sent. A world of four hundred
/// pellets broadcast whole to everybody is most of a megabyte a second before
/// anyone has done anything, and nobody can see it anyway.
///
/// P is flat, four numbers to a pellet: which one it is, where it is, and
/// what colour. The first of those is what lets the browser recognise a
/// pellet it has seen before, which it has to do now that pellets move.
///
/// W is how big the arena currently is, which travels with every frame
/// rather than only with the welcome because it grows and shrinks with the
/// crowd - a browser told once would draw the wrong edge for the rest of the
/// session.
///
/// Flat rather than a list of objects: the same pellets written out as
/// records cost several times as much, and they are the bulk of every frame.
/// The colour is a position in the palette sent with the welcome.
/// </remarks>
public record Frame(string Kind, List<Blob> B, List<int> P, List<Score> L, int Alive, double W);

/// <summary>
/// Sent to everybody when somebody takes the whole arena.
/// </summary>
/// <remarks>
/// The round is over at that point and the next one has already started -
/// everybody is back at the starting size somewhere fresh by the time this
/// arrives, so it is an announcement rather than a question.
/// </remarks>
public record Won(string Kind, string Who, int Score);

/// <summary>Sent to whoever has just been eaten.</summary>
public record Eaten(string Kind, string By, int Score);
