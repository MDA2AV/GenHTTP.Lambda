// What the two halves say to each other.
//
// Kept short on purpose. This is sent to everybody twenty times a second, so
// every name in it is paid for once per player per tick - which is why the
// fields are one letter and the numbers are rounded before they leave.

/// <summary>Sent by a browser as its first frame.</summary>
public record Join(string Kind, string Name);

/// <summary>
/// Where a browser would like to be heading, sent continuously.
/// </summary>
/// <remarks>
/// Whether the space bar is down rides along with it rather than being its
/// own message. It is a thing that is true or false for a stretch of time
/// rather than an event, so sending it with every steer means a lost or
/// late message cannot leave somebody rooted to the floor with nothing
/// holding them there.
/// </remarks>
public record Steer(string Kind, double X, double Y, bool Hold);

/// <summary>
/// A blob, as everybody else sees it.
/// </summary>
/// <remarks>
/// G is how hard it is pulling: nought for something too small to bend the
/// floor, one for standing there being large, up to three for holding on.
/// The browser draws a dent that deep, which is how everybody else finds out
/// somebody is holding - before the pull reaches them, because it ramps.
/// </remarks>
public record Blob(string I, string N, string C, double X, double Y, double R, double G);

/// <summary>
/// What a pickup grants. Nought is nothing in hand.
/// </summary>
public static class Perk
{
    public const int None = 0;

    /// <summary>Thrown at something large; takes a share of it off.</summary>
    public const int Harpoon = 1;

    /// <summary>Spends your own blob to hurt everything large nearby.</summary>
    public const int Blast = 2;

    public const int Count = 2;
}

/// <summary>Somebody used what they were carrying.</summary>
public record Use(string Kind);

/// <summary>One line of the leaderboard.</summary>
public record Score(string N, int S);

/// <summary>Who you are and how big the world is, sent once.</summary>
public record Welcome(string Kind, string Id, string Colour, double World, string[] Palette, double Well, double[] Girths);

/// <summary>
/// What is near enough to matter, sent on every tick.
/// </summary>
/// <remarks>
/// Only what falls inside the player's view is sent. A world of four hundred
/// pellets broadcast whole to everybody is most of a megabyte a second before
/// anyone has done anything, and nobody can see it anyway.
///
/// P is flat, four numbers to a pellet: which one it is, where it is, and
/// what it looks like. The first of those is what lets the browser recognise
/// a pellet it has seen before, which it has to do now that pellets move.
///
/// The last packs two things into one number, because pellets are the bulk
/// of a frame and a fifth number for every one of them is not worth what it
/// says: the low four bits are the colour, the rest is the grade. Both are
/// small and neither is ever going to need more room than that.
///
/// K is the pickups in sight, flat: which one, where, and which ability. S is
/// the harpoons in the air, flat: where, and which way they are pointing, so
/// they can be drawn as something thrown rather than as a dot. Mine and Fuse
/// are the only things in a frame that are about the player reading it - what
/// they are carrying and how long is left of their own fuse - because those
/// two are the only things nobody else is entitled to see.
///
/// W is how big the arena currently is, which travels with every frame
/// rather than only with the welcome because it grows and shrinks with the
/// crowd - a browser told once would draw the wrong edge for the rest of the
/// session.
///
/// Flat rather than a list of objects: the same pellets written out as
/// records cost several times as much, and they are the bulk of every frame.
/// The colour is a position in the palette sent with the welcome.
/// </remarks>
public record Frame(string Kind, List<Blob> B, List<int> P, List<Score> L, int Alive, double W,
                    List<int> K, List<int> S, int Mine, double Fuse);

/// <summary>
/// Sent to everybody when somebody takes the whole arena.
/// </summary>
/// <remarks>
/// The round is over at that point and the next one has already started -
/// everybody is back at the starting size somewhere fresh by the time this
/// arrives, so it is an announcement rather than a question.
/// </remarks>
public record Won(string Kind, string Who, int Score);

/// <summary>Sent to whoever has just been eaten.</summary>
public record Eaten(string Kind, string By, int Score);
