// The rules of the arena. All of them decided here, on the server.

static class World
{
    /// <summary>The smallest the arena ever is, which is how it starts.</summary>
    public const double Smallest = 9000;

    /// <summary>
    /// How much room one player is given.
    /// </summary>
    /// <remarks>
    /// About two screens' worth. Below this the arena stops being a game and
    /// becomes a scrum: every pellet that lands in the middle is eaten within
    /// a tick, and the only ones that survive are the ones out at the edge
    /// where nobody goes - so the middle wears bare and stays bare, while the
    /// total never drops far enough for the scatter to notice.
    /// </remarks>
    public const double Elbow = 5_760_000;

    /// <summary>
    /// How big the arena is for this many people.
    /// </summary>
    /// <remarks>
    /// It grows with the crowd rather than being fixed, which keeps the
    /// density of both players and pellets roughly constant however many
    /// turn up - and so keeps what one player is sent per tick constant too,
    /// because that is decided by how far they can see and not by how big
    /// the world is.
    /// </remarks>
    public static double SizeFor(int players) => Math.Max(Smallest, Math.Sqrt(players * Elbow));

    /// <summary>
    /// How many pellets belong on the floor of an arena this size.
    /// </summary>
    /// <remarks>
    /// A density rather than a count, so that it is what a player sees that
    /// stays the same: about forty on screen, which is enough that there is
    /// always something worth steering at.
    /// </remarks>
    public static int PelletsFor(double size) => (int)(size * size / 20_000);

    /// <summary>The radius everybody starts at.</summary>
    public const double Start = 14;

    /// <summary>
    /// The score that wins the arena and ends the round.
    /// </summary>
    /// <remarks>
    /// The round is as long as it was: what it costs is set by the radius it
    /// ends at, which is as large as the arena can hold and still have a game
    /// in it, and not by the number printed beside it. What this buys is the
    /// number, and the number is the whole of the endgame - a blob past the
    /// collapse cannot move and has nothing else to watch.
    ///
    /// Scores are doubles now, too. They were integers, which stop at two
    /// billion and change, so anything past the old ceiling reached the
    /// leaderboard as a negative number.
    /// </remarks>
    public const double Win = 1e200;

    /// <summary>
    /// The score at which somebody stops being a player and becomes a hole.
    /// </summary>
    /// <remarks>
    /// Below the win rather than at it, so the last stretch of a round is
    /// played out rather than announced. Whoever reaches this can no longer
    /// move at all and pulls the whole arena towards them instead - which
    /// makes the end of a round a thing everybody else has to survive rather
    /// than a number going up somewhere they cannot see.
    /// </remarks>
    public const double Collapse = 10_000_000;

    /// <summary>The plain area at which a blob collapses.</summary>
    /// <remarks>
    /// The collapse is one event with two numbers: an area, where it
    /// physically happens - a radius of five hundred and sixty, which the
    /// arena reads well at - and a score, which is what a player aims at.
    /// Keeping both lets the score be anything without moving the event.
    /// </remarks>
    public const double Hole = 5_000_000;

    /// <summary>What the score is multiplied by below the collapse.</summary>
    /// <remarks>
    /// Exactly the ratio of the two numbers above, so that a blob arriving at
    /// the collapse from below and one leaving it from above agree about what
    /// it scores. Anything else puts a step in the middle of the scale.
    /// </remarks>
    public const double Lift = Collapse / Hole;

    /// <summary>Whether this radius has collapsed.</summary>
    /// <remarks>
    /// On the radius rather than on the score, and so is the win. The score
    /// past the collapse is an exponent away from the radius, and the two
    /// disagree in the last digits - a blob clamped to exactly the winning
    /// radius scored a hair under the winning score, and so never won.
    /// </remarks>
    public static bool IsHole(double radius) => radius >= Threshold;

    /// <summary>The radius the collapse happens at.</summary>
    public static readonly double Threshold = Math.Sqrt((Hole + AreaOf(Start)) / Math.PI);

    /// <summary>
    /// What a hole scores at each of its grades.
    /// </summary>
    /// <remarks>
    /// Chosen on radius and then read off as scores, which is why they are
    /// not round: five hundred and sixty, eight hundred, eleven hundred,
    /// fifteen hundred, nineteen hundred. Spacing them evenly in score would
    /// crowd four of the five into the last few hundred units of growth.
    /// The last of them is well below the ceiling rather than at it,
    /// which was the trouble with the previous ladder: the top grade was
    /// reached and the round ended in the same breath, so nobody ever saw
    /// one. There is most of a grade's worth of arena left to eat after the
    /// last step, which is long enough to be somewhere you have to live.
    /// </remarks>
    // the first of them is the collapse itself, taken from the same number
    // rather than written out again: a grade one that was not yet a hole is a
    // blob with a hole's aura and none of a hole's rules
    public static readonly double[] Grades9 = [Threshold, 1900, 2700, 3700, 4700];

    /// <summary>Which grade of hole this is, nought for something still a blob.</summary>
    public static int HoleLevel(double radius)
    {
        var level = 0;

        foreach (var step in Grades9)
        {
            if (radius >= step)
            {
                level++;
            }
        }

        return level;
    }

    /// <summary>
    /// How much harder a hole pulls than the same size merely holding on.
    /// </summary>
    /// <remarks>
    /// Large enough that it is not a strong player, it is weather. The reach
    /// multiplier is the part that decides whether the arena feels it: a hole
    /// that pulls enormously but only nearby is just a big blob.
    /// </remarks>
    // two and a half times what it was, on the same reasoning that made
    // holding free: the strongest thing in the game was being too polite
    public const double HoleGrip = 22.0;

    public const double HoleReach = 4.0;

    /// <summary>
    /// How much harder each grade of hole pulls than the one below it.
    /// </summary>
    /// <remarks>
    /// Compounding, so the top of the scale is not a bigger version of the
    /// bottom of it - a grade five pulls a hundred and thirty times what a
    /// grade one does, and a grade one pulls twenty two times what the same
    /// size holding on would. The reach compounds far more gently, or the
    /// last grade would simply own the whole arena from wherever it happened
    /// to be standing.
    /// </remarks>
    public const double HoleStep = 3.4;

    public static double HolePull(double radius) => HoleGrip * Math.Pow(HoleStep, Math.Max(0, HoleLevel(radius) - 1));

    public static double HoleSpan(double radius) => HoleReach * Math.Pow(1.28, Math.Max(0, HoleLevel(radius) - 1));

    /// <summary>
    /// How many pellets a hole calls down around itself each second.
    /// </summary>
    /// <remarks>
    /// The point of it is not the food, it is the traffic. A hole cannot move
    /// and cannot chase, so without something to draw people in it is a
    /// hazard everybody simply walks around and the round never ends. A floor
    /// thick with pellets is worth the risk of going near one, which is what
    /// makes it dangerous again.
    /// </remarks>
    public static int Summons(double radius) => 34 * HoleLevel(radius) * HoleLevel(radius);

    /// <summary>How far from a hole its summoned pellets land.</summary>
    public static double SummonSpan(double radius) => radius * 2.6 + 460;

    /// <summary>
    /// The most company an arena will keep.
    /// </summary>
    /// <remarks>
    /// A hundred is a crowd rather than a handful, and the arena grows to hold
    /// them - it is sized by how many are in it, so a full house is about nine
    /// thousand across with ten thousand pellets on the floor rather than
    /// three and a half thousand with fifteen hundred.
    /// </remarks>
    public const int Crowd = 100;

    /// <summary>How long a beat of the arena clock is, in seconds.</summary>
    /// <remarks>
    /// Twenty a second: fast enough that the browser has something recent to
    /// interpolate between, slow enough that a hundred blobs and thirty
    /// thousand pellets fit inside one of them with room to spare.
    /// </remarks>
    public const double Beat = 0.05;

    /// <summary>
    /// The most pellets one frame will describe.
    /// </summary>
    /// <remarks>
    /// A budget rather than a distance: how far somebody sees already grows
    /// with how big they are, and holes make the floor under them dense, so
    /// the two together had one player being sent a megabyte a second. At
    /// that zoom the floor is texture rather than countable things.
    /// </remarks>
    public const int Crumbs = 1400;

    /// <summary>
    /// How long this blob waits between teleports.
    /// </summary>
    /// <remarks>
    /// A second when you are nothing, fifteen when you are about to collapse.
    /// Being small is the state where an escape is worth the most and the
    /// least worth defending against, and a fixed cooldown handed the same
    /// escape to the thing everybody else was running from.
    /// </remarks>
    public static double WarpWait(double radius)
    {
        var through = Math.Clamp(ScoreOf(radius) / Collapse, 0, 1);

        return 1 + 14 * Math.Sqrt(through);
    }

    /// <summary>
    /// What a teleport costs, as a share of the blob.
    /// </summary>
    /// <remarks>
    /// Small, and there so that it is not simply free movement. Vanishing and
    /// reappearing across the arena for nothing would make chasing anybody
    /// pointless, and being chased is most of the game.
    /// </remarks>
    public const double WarpToll = 0.04;

    /// <summary>How long a teleport takes to cross, in seconds.</summary>
    /// <remarks>
    /// A teleport that lands on the keypress is unanswerable, and worse than
    /// that it is unreadable: the blob is simply somewhere else, and nobody
    /// watching learns anything about what happened. Marking the ground
    /// first and crossing to it over the next fraction of a second costs the
    /// escape almost nothing - it is still an order of magnitude faster than
    /// walking, and nothing in the arena can cover the ground in time - and
    /// it buys the two things that make it a move rather than a trick: the
    /// blob leaving is visibly the blob that arrives, and everybody near
    /// where it is going gets a moment's warning that it is coming.
    /// </remarks>
    public const double WarpLead = 0.42;

    /// <summary>How far a teleport may carry somebody.</summary>
    public static double WarpReach(double radius) => 2200 + radius * 6;

    /// <summary>
    /// How fast a drawn line is run, in units a second.
    /// </summary>
    /// <remarks>
    /// Far faster than anything can move under its own steam, which is the
    /// whole of the pickup: it is not a speed boost, it is a route decided in
    /// advance and then taken faster than anybody can react to.
    /// </remarks>
    /// <summary>How long a hole's whirl lasts, in seconds.</summary>
    /// <remarks>
    /// A hole is the strongest thing in the arena and cannot chase anything,
    /// so the only way its round ends is if somebody volunteers. Calling
    /// pellets down is an offer, and people decline offers.
    ///
    /// This does not pull towards the hole. It pulls towards a spot the hole
    /// chooses, anywhere it can see: on yourself to feed, beside somebody to
    /// take them off whatever they were doing. Either way, coming near a hole
    /// stops being entirely their decision.
    /// </remarks>
    public const double WhirlTime = 4.0;

    /// <summary>How long between one whirl and the next, in seconds.</summary>
    public const double WhirlWait = 11.0;

    /// <summary>How far a whirl reaches, in units.</summary>
    public const double WhirlReach = 6000;

    /// <summary>How fast it hauls, at the middle of it, in units a second.</summary>
    /// <remarks>
    /// Several times what anything can walk, because half measures here are
    /// worse than nothing: a pull somebody can walk out of is a nuisance that
    /// teaches them the hole is harmless. It falls away to nothing at the
    /// edge, so the escape is to already be near the edge.
    /// </remarks>
    public const double WhirlPull = 1700;

    /// <summary>How hard a whirl hauls something this far from the middle.</summary>
    public static double WhirlSpeed(double distance) =>
        WhirlPull * Math.Pow(Math.Clamp(1 - distance / WhirlReach, 0, 1), 0.7);

    /// <summary>How long between one drawn line and the next, in seconds.</summary>
    /// <remarks>
    /// It used to be a thing lying on the floor that one player in twenty
    /// happened to walk over, which made the best move in the game a matter
    /// of where you were standing. Everybody has it now, and what it costs
    /// is the wait - the same bargain the teleport makes, and the reason
    /// both of them are decisions rather than habits.
    /// </remarks>
    public const double DrawWait = 18.0;

    public const double StreakSpeed = 2600;

    /// <summary>The longest line that may be drawn, in units.</summary>
    public const double StreakLength = 9000;

    /// <summary>The most points a drawn line may be made of.</summary>
    public const int StreakPoints = 64;

    /// <summary>
    /// The radius nobody grows past, which is the radius that scores exactly
    /// <see cref="Win" />.
    /// </summary>
    /// <remarks>
    /// Derived rather than picked, so that the cap and the target cannot
    /// drift apart: a cap below the target is a game that can never be won,
    /// and one above it is a target reached while still growing.
    /// </remarks>
    public static readonly double Biggest = RadiusScoring(Win);

    /// <summary>
    /// What a pellet is worth, by grade.
    /// </summary>
    /// <remarks>
    /// Graded rather than free-form so that a pellet's worth costs nothing to
    /// send: the grade is a number under sixteen, and it rides in the spare
    /// bits of the colour it was already carrying. Ordinary scattered pellets
    /// are all grade zero; the larger ones are what falls out of something
    /// big when it is hurt.
    /// </remarks>
    public static readonly double[] Grades = [300, 900, 2700, 8100];

    /// <summary>What a pellet of this grade adds to the area of whoever eats it.</summary>
    public static double Worth(int grade) => Grades[Math.Clamp(grade, 0, Grades.Length - 1)];

    /// <summary>
    /// How big a pellet of this grade is, which is the radius its own worth
    /// comes to - so a pellet that is worth nine times as much looks three
    /// times as wide, and what it is worth can be judged by looking at it.
    /// </summary>
    public static double Girth(int grade) => Math.Sqrt(Worth(grade) / Math.PI);

    /// <summary>The grades, as the browser needs them to draw one.</summary>
    public static double[] Girths => [.. Grades.Select((_, g) => Girth(g))];

    /// <summary>How much bigger you have to be to eat somebody.</summary>
    public const double EatRatio = 1.10;

    #region Gravity

    /*
     * Anything big enough to eat you also pulls you towards it, harder the
     * closer you are. It is what stops a big blob being something you can
     * simply walk past, and it is the only thing in here that acts on a
     * player who is not touching anything.
     *
     * It is a velocity rather than an acceleration on purpose. Nothing else
     * in this arena stores momentum - a blob goes where it is pointed at the
     * speed its size allows - and integrating an inverse square acceleration
     * at twenty ticks a second with no substeps is the textbook way to get
     * blobs slingshotting through each other and gaining energy. A velocity
     * field cannot do that: what it adds in a tick is bounded by the cap
     * below, so a blob can never move further in a tick than it could under
     * its own steam.
     */

    /// <summary>How strong the pull is, before distance and size temper it.</summary>
    public const double Pull = 360;

    /// <summary>
    /// Keeps the pull finite where the distance is not.
    /// </summary>
    /// <remarks>
    /// Without it the force runs away as two centres converge, and the tick
    /// that lands closest decides the whole encounter.
    /// </remarks>
    public const double Soften = 80;

    /// <summary>
    /// The most of your own speed that can be taken away from you.
    /// </summary>
    /// <remarks>
    /// The number that decides whether this is a mechanic or a punishment.
    /// Holding on raises it - see Clutch - but never to one.
    /// A new blob at 260 a second fleeing the biggest possible chaser, which
    /// moves at about 102, keeps 130 against their 102 and pulls away at 28 -
    /// slowly, and only if it commits. Above about six tenths that margin
    /// goes negative and being spawned near somebody large is simply death,
    /// so this sits below it with room to spare. It also keeps half of the
    /// player's own steering authority, without which the drag stops reading
    /// as gravity and starts reading as a broken connection.
    /// </remarks>
    public const double Hand = 0.5;

    /// <summary>
    /// The size at which a blob starts bending things around it.
    /// </summary>
    /// <remarks>
    /// One number for all of it: below this a blob pulls nothing and has no
    /// dent drawn, above it the dent appears and everything loose nearby
    /// starts falling towards it. Keeping the drawing and the physics on the
    /// same threshold is what makes the picture trustworthy - a dent means a
    /// pull, always, and no pull ever happens without one.
    ///
    /// It is sent to the browser rather than agreed with it, because it also
    /// decides which blobs are sent to a player who cannot see them but can
    /// see the floor bending around them.
    /// </remarks>
    public const double WellSize = 40;

    /// <summary>
    /// How much of the pull a pellet feels, against what a player feels.
    /// </summary>
    /// <remarks>
    /// Far less, and the reason is not subtlety but supply. Whatever drifts
    /// in gets eaten, and the scatter that replaces it lands anywhere in the
    /// arena, so the ground around a big blob empties faster than a uniform
    /// scatter refills it. How bare it ends up is set by how long a pellet
    /// takes to cross the gap: slow the drift by half and twice as many are
    /// in the air at any moment. Dragged hard enough, a blob simply sits in
    /// a clearing with nothing visibly falling in at all, which is the
    /// opposite of the thing being drawn.
    ///
    /// This sits where it does because the scatter now aims at the thinnest
    /// ground it can find, so the clearing is being refilled as fast as it
    /// is being swept and the drift can afford to be seen.
    /// </remarks>
    public const double Dust = 0.08;

    /// <summary>The fastest a pellet is ever dragged, in units a second.</summary>
    public const double Sweep = 18;

    #region Holding on

    /*
     * The space bar. A player roots themselves to the floor and pulls
     * everything near them far harder than standing there would. It is the
     * one thing in the game a player presses rather than steers, and what it
     * costs is standing still - which in an arena where everything is
     * hunting everything is a great deal to give up.
     */

    /// <summary>How much harder a freshly held pull drags than an idle one.</summary>
    /// <remarks>
    /// Where it starts, not where it ends. A hold that arrives at full
    /// strength in half a second is a button rather than a commitment, and
    /// there is no decision if the first moment is already the best one.
    /// </remarks>
    public const double Grasp = 1.8;

    /// <summary>
    /// The most of your own speed a held pull can take from you.
    /// </summary>
    /// <remarks>
    /// Higher than the idle cap and deliberately still under one. At eight
    /// tenths, somebody running straight away from a held pull always nets a
    /// fifth of their own speed outwards, so leaving is always possible and
    /// always slow - which is the difference between a trap and a death
    /// sentence. Past one, two overlapping pulls make a patch of floor that
    /// nobody who wanders into it can ever leave.
    /// </remarks>
    public const double Clutch = 0.8;

    /// <summary>
    /// How long a held pull takes to come up to full strength.
    /// </summary>
    /// <remarks>
    /// Long enough that tapping it is pure loss, and long enough that the
    /// floor visibly caves in before the pull arrives - which is the warning
    /// everybody else gets.
    /// </remarks>
    public const double Wind = 0.6;

    /// <summary>How long before it can be held again.</summary>
    public const double Rest = 2.0;

    /// <summary>
    /// How long a hold keeps getting stronger for, after it has wound up.
    /// </summary>
    /// <remarks>
    /// The wind up is the warning; this is the commitment. A hold that is
    /// merely on is worth not quite two of standing there; ten seconds later
    /// it is worth nine. Ten seconds is a long time to spend rooted to one
    /// spot with the floor caving in under you, visibly, for everybody.
    /// </remarks>
    public const double Swell = 10.0;

    /// <summary>How many times its initial strength a hold can reach.</summary>
    public const double Zenith = 5.0;

    /// <summary>What a held pull does to loose pellets instead of the usual.</summary>
    public const double HeldDust = 0.24;

    /// <summary>The fastest a held pull drags a pellet, in units a second.</summary>
    public const double HeldSweep = 60;

    /// <summary>How far past the usual count the arena will carry refunds.</summary>
    public const double Glut = 1.5;

    /// <summary>
    /// How hard somebody is pulling, given whether and how long they have held.
    /// </summary>
    /// <remarks>
    /// One number carries the whole thing. Nought is a blob too small to
    /// bend anything, one is standing there being large, three is a hold
    /// that has just wound up, and nine is one that has been paid for. The physics multiplies by it, the browser draws a dent
    /// that deep, and neither needs to know anything else - so a blob too
    /// small to have a field of its own grows one the moment it holds, and
    /// the floor says so before the pull arrives.
    /// </remarks>
    public static double Grip(double radius, bool held, double wound)
    {
        if (!held)
        {
            return radius >= WellSize ? 1 : 0;
        }

        // up to full strength over the wind up, and then on past it for as
        // long as they are willing to keep paying
        var wound_up = Math.Clamp(wound / Wind, 0, 1);

        var deeper = 1 + Math.Clamp((wound - Wind) / Swell, 0, 1) * (Zenith - 1);

        return Grasp * wound_up * deeper;
    }

    #endregion


    /// <summary>
    /// How far a blob of this size drags loose pellets.
    /// </summary>
    /// <remarks>
    /// It widens with the grip, the same as the pull on players does - a
    /// hold that reaches further for people and not for dust looks broken,
    /// because the floor near you stays put while everything else comes in.
    ///
    /// What is being tuned here is not the strength of anything, it is how
    /// many pellets are in the air at once - which is the inflow times how
    /// long the crossing takes, and nothing else. Too tight and the gap is
    /// crossed in a second, so whatever lands in it is gone before it can be
    /// seen moving. This is wide enough that the crossing takes the better
    /// part of half a minute, and the ring around a big blob is always full
    /// of dust on its way in.
    /// </remarks>
    public static double Drag(double radius, double grip) => radius * 3.5 * Math.Sqrt(Math.Max(1, grip));

    /// <summary>How far a blob of this size reaches, past which it pulls nothing.</summary>
    /// <remarks>
    /// Four times its own radius, so the reach of the largest possible blob
    /// is two of the spatial cells the floor is bucketed into, and the last
    /// half radius of it is faded out - otherwise a blob crossing the line
    /// would feel the pull switch on all at once.
    /// </remarks>
    /// <remarks>
    /// It widens with the grip as well as the size, or a hold that has been
    /// on for six seconds would be pulling everything inside a fixed circle
    /// at the cap already and the extra strength would buy nothing. The
    /// square root keeps it from running away: nine times the grip is three
    /// times the reach, not nine.
    /// </remarks>
    public static double Reach(double radius, double grip) => radius * 4 * Math.Sqrt(Math.Max(1, grip));

    /// <summary>
    /// How fast <paramref name="theirs" /> drags <paramref name="mine" /> in,
    /// at this distance between their centres.
    /// </summary>
    /// <remarks>
    /// Only something that could actually eat you pulls at all, and it ramps
    /// up from there rather than switching on: two blobs near enough in size
    /// to be rivals have almost no speed between them, so a pull at that
    /// ratio would be inescapable exactly where it is least deserved.
    /// </remarks>
    public static double PullSpeed(double mine, double theirs, double distance, double grip)
    {
        var reach = Reach(theirs, grip);

        if (grip <= 0 || distance >= reach || mine <= 0)
        {
            return 0;
        }

        var ratio = theirs / mine;

        if (ratio <= EatRatio)
        {
            return 0;
        }

        // full strength once they are comfortably more than twice your size
        var bite = Smoothstep((ratio - EatRatio) / (2.5 - EatRatio));

        // and faded away over the last half radius of the reach
        var edge = Math.Clamp((reach - distance) / (theirs * 0.5), 0, 1);

        return Pull * theirs * theirs / (distance * distance + Soften * Soften) * bite * edge * grip;
    }

    /// <summary>The usual ease in and out, flat at both ends.</summary>
    public static double Smoothstep(double t)
    {
        var x = Math.Clamp(t, 0, 1);

        return x * x * (3 - 2 * x);
    }

    #endregion

    /// <summary>How much of the loser's area the winner keeps.</summary>
    public const double Spoils = 0.85;

    /// <summary>
    /// How far either side of a player is worth sending to them.
    /// </summary>
    /// <remarks>
    /// This follows the browser's zoom instead of being a fixed distance,
    /// because the browser zooms out as a blob grows. A fixed distance is
    /// necessarily wrong in both directions: too tight for a big blob, which
    /// then watches things appear out of nothing at the edge of its screen,
    /// and too generous for a small one, which pays every tick for pellets it
    /// could not possibly see.
    ///
    /// The browser fits Extent(radius) across the shorter side of the window.
    /// Nine tenths of that either way covers a wide screen with a little to
    /// spare, so nothing arrives late.
    /// </remarks>
    public static double ViewOf(double radius) => Extent(radius) * 0.9;

    /// <summary>
    /// How much of the arena the browser fits across the shorter side of the
    /// window, for a player of this size.
    /// </summary>
    /// <remarks>
    /// This has to match what the browser does, and the browser has its own
    /// copy - it is the one number both halves need and neither can be told
    /// by the other without asking every frame.
    ///
    /// It climbs well short of the radius on purpose. Straight off the radius
    /// and a blob near the cap looks at nine thousand units through a window
    /// eight hundred pixels tall, where a starter is two pixels across. At
    /// this exponent it sees three thousand, which still fills the screen
    /// and leaves everything in it a size you can aim at.
    /// </remarks>
    public static double Extent(double radius) => 600 + 260 * Math.Pow(radius / Start, 0.7);

    /// <summary>
    /// Everything here is one of these nine colours, and the browser is told
    /// what they are on arrival rather than being expected to already agree.
    /// </summary>
    public static readonly string[] Palette =
    [
        "#8ab4f8", "#f28b82", "#fdd663", "#81c995", "#c58af9", "#78d9ec", "#ff8bcb", "#ffa26b", "#a0e7a0"
    ];

    public static string ColourFor(int seat) => Palette[seat % Palette.Length];

    /// <summary>A colour to scatter a pellet in.</summary>
    public static int AnyTint(Random random) => random.Next(Palette.Length);

    /// <summary>
    /// How fast a blob of this size travels, in units a second.
    /// </summary>
    /// <remarks>
    /// Falling off with the radius, because a game where the biggest player is
    /// also the fastest has one player and then it is over. The exponent is
    /// gentle enough that being large is still worth it.
    /// </remarks>
    public static double Speed(double radius) => 260.0 / Math.Pow(radius / Start, 0.42);

    /// <summary>How long a kill leaves you charged, in seconds.</summary>
    /// <remarks>
    /// Long enough to reach the next one and no longer. The whole of the
    /// mechanic is in that: five seconds is not a reward for having eaten
    /// somebody, it is an invitation to go and do it again immediately,
    /// while whoever is near you has just watched what you did to the last
    /// one and is deciding whether to run.
    /// </remarks>
    /// <remarks>
    /// Not given to holes. A charge is extra speed and a hole has no legs, so
    /// on one it would be lightning drawn around something that cannot use it
    /// - a warning about a danger that is not the danger.
    /// </remarks>
    public const double SparkTime = 5.0;

    /// <summary>The most kills a charge can be stacked to.</summary>
    public const int SparkMost = 5;

    /// <summary>What being charged does to your legs.</summary>
    /// <remarks>
    /// Compounding with each kill taken while the last is still lit, which
    /// is what makes a run a run: fourteen per cent for one, seventy for a
    /// chain of five. Capped, because the arena is only so wide and a blob
    /// crossing it faster than the frames can show it is a bug, not a
    /// reward.
    /// </remarks>
    public static double SparkBoost(int sparks) => 1 + 0.14 * Math.Clamp(sparks, 0, SparkMost);

    /// <summary>How fast a blob of this size and this charge travels.</summary>
    public static double Legs(double radius, int sparks) => Speed(radius) * SparkBoost(sparks);

    /// <summary>
    /// How big a square of the arena one bucket of the pellet grid covers.
    /// </summary>
    /// <remarks>
    /// A fixed size rather than one tied to how big a blob can get. A cell
    /// the width of the largest possible blob would be five hundred units
    /// across, which puts most of the arena in a handful of buckets and
    /// gives back everything the bucketing was for. Anything larger than a
    /// cell simply reads more of them - see KeysWithin, which walks the
    /// whole range rather than assuming one neighbour is enough.
    /// </remarks>
    public const double Cell = 260;

    /// <summary>Which bucket something at this position belongs in.</summary>
    /// <remarks>
    /// Two cell numbers packed into one, so the grid is an ordinary
    /// dictionary rather than something that has to be sized in advance and
    /// rebuilt every time the arena grows.
    /// </remarks>
    public static long KeyOf(double x, double y) => ((long)(x / Cell) << 24) | (long)(y / Cell);

    /// <summary>
    /// Every bucket that a square of this reach around a point touches.
    /// </summary>
    /// <remarks>
    /// This is what keeps the tick affordable. Both of the loops that used to
    /// walk every pellet for every player - who is eating what, and what each
    /// player can see - now walk the handful of buckets they actually reach.
    /// At a few hundred of each that is the difference between a tick that
    /// fits in fifty milliseconds and one that does not.
    /// </remarks>
    public static IEnumerable<long> KeysWithin(double x, double y, double reach)
    {
        var fromX = (long)(Math.Max(0, x - reach) / Cell);
        var toX = (long)(Math.Max(0, x + reach) / Cell);
        var fromY = (long)(Math.Max(0, y - reach) / Cell);
        var toY = (long)(Math.Max(0, y + reach) / Cell);

        for (var cx = fromX; cx <= toX; cx++)
        {
            for (var cy = fromY; cy <= toY; cy++)
            {
                yield return (cx << 24) | cy;
            }
        }
    }

    /// <summary>The area of a blob, which is what actually accumulates.</summary>
    public static double AreaOf(double radius) => Math.PI * radius * radius;

    /// <summary>The radius an area comes to, capped.</summary>
    public static double RadiusOf(double area) => Math.Min(Biggest, Math.Sqrt(area / Math.PI));

    /// <summary>What a blob is worth on the leaderboard.</summary>
    /// <remarks>
    /// Ten times the area, up to the collapse, and far more than that past
    /// it. The two have to come apart: a hundred quintillion of plain area
    /// is a radius of five and a half billion in an arena nine thousand
    /// across. It is also the right shape for the thing it draws, since what
    /// makes a hole frightening is its mass and not its floor space.
    /// </remarks>
    public static double ScoreOf(double radius)
    {
        var plain = AreaOf(radius) - AreaOf(Start);

        if (plain <= Hole)
        {
            return Math.Round(plain * Lift);
        }

        return Math.Round(Collapse * Math.Pow(plain / Hole, Steepen));
    }

    /// <summary>How much faster score climbs than area, once collapsed.</summary>
    /// <remarks>
    /// Picked so both ends of the scale land at radii the arena can hold: the
    /// collapse at about five hundred and sixty, the ceiling at twenty one
    /// hundred. A hundred quintillion of plain area would be a radius of five
    /// and a half billion, in an arena nine thousand across.
    ///
    /// Steep, and meant to be: the thirteen orders of magnitude between the
    /// collapse and the ceiling are the whole of what a hole has to look at.
    /// </remarks>
    public const double Steepen = 156.90;

    /// <summary>The radius that scores this much, which is ScoreOf run backwards.</summary>
    public static double RadiusScoring(double score)
    {
        var plain = score <= Collapse
            ? score / Lift
            : Hole * Math.Pow(score / Collapse, 1 / Steepen);

        return Math.Sqrt((plain + AreaOf(Start)) / Math.PI);
    }

    /// <summary>Whether this blob has just taken the whole arena.</summary>
    public static bool HasWon(double radius) => radius >= Biggest - 0.001;

    /// <summary>Whether one blob may eat another.</summary>
    public static bool Eats(double mine, double theirs) => mine > theirs * EatRatio;

    /// <summary>
    /// Whether a blob is touching a pellet, which is edge to edge rather
    /// than centre to edge - a pellet worth eight thousand is fifty units
    /// across and being told to swallow its middle before it counts is the
    /// same lie the drawing used to tell about blobs.
    /// </summary>
    public static bool Reaches(double px, double py, double radius, double x, double y, int grade)
    {
        var dx = px - x;
        var dy = py - y;

        var touch = radius + Girth(grade);

        return dx * dx + dy * dy < touch * touch;
    }

    /// <summary>Whether two blobs are touching enough for that to happen.</summary>
    public static bool Overlaps(double ax, double ay, double ar, double bx, double by)
    {
        var dx = ax - bx;
        var dy = ay - by;

        // Half of the smaller one has to be inside the larger, which is what
        // its middle being inside means: at that distance exactly half of its
        // width is covered. It is the ratio above, not this, that used to
        // make a capture feel like it had failed - at eighteen per cent
        // bigger you could sit squarely on top of somebody nearly your size
        // and nothing would happen.
        return dx * dx + dy * dy < ar * ar;
    }

    /// <summary>
    /// How fast a pellet at this distance is dragged towards a blob.
    /// </summary>
    /// <remarks>
    /// The same law the players feel, tempered and capped. A pellet has no
    /// size to compare against, so nothing is gated on the ratio: anything
    /// with a dent pulls dust, and the dust falls in.
    /// </remarks>
    public static double DriftSpeed(double theirs, double distance, double grip, bool held)
    {
        var reach = Drag(theirs, grip);

        if (grip <= 0 || distance >= reach)
        {
            return 0;
        }

        var edge = Math.Clamp((reach - distance) / (theirs * 0.5), 0, 1);

        // holding on drags dust far harder than standing there does, and the
        // wind up is already in the grip
        var temper = held ? HeldDust : Dust;
        var most = held ? HeldSweep : Sweep;

        var speed = Pull * theirs * theirs / (distance * distance + Soften * Soften) * edge * temper * grip;

        return Math.Min(speed, most);
    }

    /// <summary>Whether something is near enough to whoever is looking.</summary>
    public static bool InView(double px, double py, double radius, double x, double y)
    {
        var reach = ViewOf(radius);

        return Math.Abs(x - px) < reach && Math.Abs(y - py) < reach;
    }

    public static double Clamp(double value, double size) => Math.Clamp(value, 0, size);

    public static string Called(string name)
    {
        var trimmed = (name ?? "").Trim();

        if (trimmed.Length == 0)
        {
            return "a blob";
        }

        return trimmed.Length > 12 ? trimmed[..12] : trimmed;
    }
}
