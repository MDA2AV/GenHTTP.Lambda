// The room, and the rules about what may happen in it.
//
// Everything here is decided on the server. A browser says where it would like
// to be and this decides where it is, which is the only arrangement where one
// player's altered client cannot spoil the room for everybody else.
//
// lambda.cs owns the socket and the storage; this owns the truth.

static class Arena
{
    /// <summary>Half the width of the floor. The room runs from -Half to +Half.</summary>
    public const double Half = 14.0;

    /// <summary>How far a player may get between two messages.</summary>
    /// <remarks>
    /// Generous enough for a slow connection sending few updates, small enough
    /// that nobody crosses the room in one frame. A client that asks for more
    /// is not refused, it is moved this far in the direction it wanted.
    /// </remarks>
    public const double Step = 1.6;

    /// <summary>How close counts as running into a prize.</summary>
    public const double Reach = 1.1;

    /// <summary>How many prizes are on the floor at once.</summary>
    public const int Prizes = 6;

    private static readonly string[] Palette =
    [
        "#8ab4f8", "#f28b82", "#fdd663", "#81c995", "#c58af9", "#78d9ec", "#fbbc04", "#ff8bcb"
    ];

    /// <summary>A colour for a player, spread around the palette.</summary>
    public static string ColourFor(int seat) => Palette[seat % Palette.Length];

    /// <summary>Somewhere on the floor, away from the very edge.</summary>
    public static (double X, double Z) Somewhere(Random random)
        => ((random.NextDouble() * 2 - 1) * (Half - 1), (random.NextDouble() * 2 - 1) * (Half - 1));

    /// <summary>
    /// Where a player actually ends up, given where they asked to be.
    /// </summary>
    public static (double X, double Z) Settle(double fromX, double fromZ, double wantX, double wantZ)
    {
        var dx = wantX - fromX;
        var dz = wantZ - fromZ;

        var distance = Math.Sqrt(dx * dx + dz * dz);

        // further than a step is walked as far as a step goes, in the
        // direction that was asked for
        if (distance > Step && distance > 0)
        {
            dx = dx / distance * Step;
            dz = dz / distance * Step;
        }

        return (Clamp(fromX + dx), Clamp(fromZ + dz));
    }

    /// <summary>Whether a player is close enough to a prize to have taken it.</summary>
    public static bool Reaches(double x, double z, Prize prize)
    {
        var dx = x - prize.X;
        var dz = z - prize.Z;

        return Math.Sqrt(dx * dx + dz * dz) <= Reach;
    }

    /// <summary>A name that can be shown, or one chosen for whoever gave none.</summary>
    public static string Called(string name)
    {
        var trimmed = (name ?? "").Trim();

        if (trimmed.Length == 0)
        {
            return "someone";
        }

        return trimmed.Length > 14 ? trimmed[..14] : trimmed;
    }

    private static double Clamp(double value) => Math.Clamp(value, -Half, Half);
}
