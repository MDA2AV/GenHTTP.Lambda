// Adds the tasks in tasks.json to the list, for trying the page with more
// than a handful of them.
//
//   node seed.mjs <address>         the preview of a feature - never the list people use
//
// The tasks are kept in the app's database, which only the app writes to, so
// test data goes in the way everybody's tasks do: through the API, one
// POST tasks/ each. Run it twice and they are there twice. Ends with 1 when a
// task was refused. Needs Node 18 or later and nothing else.

import { readFile } from 'node:fs/promises';

const address = process.argv[2];

if (!address) {
  console.error('Usage: node seed.mjs <address of the app>');
  process.exit(2);
}

const base = address.endsWith('/') ? address : `${address}/`;

const tasks = JSON.parse(await readFile(new URL('tasks.json', import.meta.url), 'utf8'));

let refused = 0;

for (const task of tasks) {
  const response = await fetch(new URL('tasks/', base), {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(task),
  });

  if (response.status !== 201) {
    refused++;
    console.log(`FAIL ${task.title} - status ${response.status}: ${await response.text()}`);
  }
}

console.log(`${tasks.length - refused} of ${tasks.length} tasks added.`);

process.exit(refused === 0 ? 0 : 1);
