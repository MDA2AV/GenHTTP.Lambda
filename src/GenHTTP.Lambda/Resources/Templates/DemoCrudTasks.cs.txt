// The API, as a class whose methods are the routes.
//
// [ResourceMethod] names the HTTP method and the path of each one. A parameter
// named like a segment of the path (:id) is taken from the path, a record from
// the JSON body, anything else from the query. What a method returns is sent
// back as JSON in camel case - the page reads task.title, not task.Title.
//
// Problems are thrown as ProviderException with the status to answer, which
// the platform turns into a response with that status and the message.
//
// [Summary] and [Remarks] say what a route does in the API's description,
// openapi.json - and so in scalar/, where it is tried. A parameter or a field
// of a record can have one too.

[Summary("What a client sends to add or change a task.")]
public record TaskInput([Summary("What is to be done, up to 120 characters")] string Title,
                        [Summary("More about it, up to 1000 characters")] string Notes,
                        [Summary("Whether it is done")] bool Done);

public class Tasks(TaskStore store)
{
    internal const int MaxTitle = 120;

    internal const int MaxNotes = 1000;

    [ResourceMethod]
    [Summary("Lists the tasks")]
    [Remarks("Open tasks first, the newest first. With search, only those whose title or notes contain it, whatever the case.")]
    public List<TaskItem> List([Summary("Text to look for")] string search = null) => store.List(search);

    [ResourceMethod(":id")]
    [Summary("Reads a task")]
    public TaskItem Get(string id) => store.Find(id) ?? throw Missing(id);

    [ResourceMethod(Method.Post)]
    [Summary("Adds a task")]
    public Result<TaskItem> Create(TaskInput input)
    {
        var (title, notes) = Validate(input);

        // 201 rather than 200: something new exists now
        return new Result<TaskItem>(Add(store, title, notes, input.Done)).Status(ResponseStatus.Created);
    }

    [ResourceMethod(Method.Put, ":id")]
    [Summary("Changes a task")]
    public TaskItem Update(string id, TaskInput input)
    {
        var (title, notes) = Validate(input);

        return store.Change(id, title, notes, input.Done) ?? throw Missing(id);
    }

    [ResourceMethod(Method.Delete, ":id")]
    [Summary("Removes a task")]
    public void Delete(string id)
    {
        // nothing returned is answered with 204 No Content
        if (!store.Remove(id))
        {
            throw Missing(id);
        }
    }

    /// <summary>
    /// Makes a task of what was checked, and keeps it.
    /// </summary>
    internal static TaskItem Add(TaskStore store, string title, string notes, bool done)
    {
        var now = DateTime.UtcNow;

        var task = new TaskItem { Id = Guid.NewGuid().ToString("n")[..10], Title = title, Notes = notes, Done = done, Created = now, Updated = now };

        store.Add(task);

        return task;
    }

    /// <summary>
    /// Checks what was sent, and answers 400 with the reason where it is wrong.
    /// </summary>
    internal static (string Title, string Notes) Validate(TaskInput input)
    {
        var title = (input?.Title ?? "").Trim();

        var notes = (input?.Notes ?? "").Trim();

        if (title.Length == 0)
        {
            throw new ProviderException(ResponseStatus.BadRequest, "A task needs a title.");
        }

        if (title.Length > MaxTitle)
        {
            throw new ProviderException(ResponseStatus.BadRequest, $"A title has at most {MaxTitle} characters.");
        }

        if (notes.Length > MaxNotes)
        {
            throw new ProviderException(ResponseStatus.BadRequest, $"Notes have at most {MaxNotes} characters.");
        }

        return (title, notes);
    }

    internal static ProviderException Missing(string id)
        => new(ResponseStatus.NotFound, $"There is no task '{id}'.");

    /// <summary>
    /// Something to look at on the very first start.
    /// </summary>
    public static void Seed(TaskStore store)
    {
        var now = DateTime.UtcNow;

        store.Add(new TaskItem { Id = "welcome1", Title = "Read Tasks.cs to see the routes", Notes = "Every public method with [ResourceMethod] is one of them.", Created = now, Updated = now });
        store.Add(new TaskItem { Id = "welcome2", Title = "Try the API in scalar/", Created = now.AddSeconds(1), Updated = now.AddSeconds(1) });
        store.Add(new TaskItem { Id = "welcome3", Title = "Open the page", Done = true, Created = now.AddSeconds(2), Updated = now.AddSeconds(2) });
    }
}
