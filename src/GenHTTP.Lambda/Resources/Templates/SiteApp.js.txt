// Served as it is: this file is never compiled and never looked at by the
// server. It is only what the folder happens to contain.

const $ = (id) => document.getElementById(id);

const api = new URL('api/notes', location.href.replace(/[^/]*$/, ''));

function show(notes) {
  $('notes').innerHTML = notes
    .map((n) => `<li>${escape(n.text)}<time>${new Date(n.written).toLocaleString()}</time></li>`)
    .join('');

  $('state').textContent = notes.length ? `${notes.length} kept` : 'Nothing yet.';
}

const escape = (s) => s.replace(/[<>&]/g, (c) => ({ '<': '&lt;', '>': '&gt;', '&': '&amp;' }[c]));

async function load() {
  try {
    show(await (await fetch(api)).json());
  } catch {
    $('state').textContent = 'Could not reach the api.';
  }
}

$('writing').onsubmit = async (e) => {
  e.preventDefault();

  const text = $('text').value.trim();

  if (!text) return;

  const button = e.target.querySelector('button');

  button.disabled = true;

  try {
    const answer = await fetch(api, {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({ text }),
    });

    if (!answer.ok) throw new Error();

    $('text').value = '';

    await load();
  } catch {
    $('state').textContent = 'That did not save.';
  } finally {
    button.disabled = false;
  }
};

load();
