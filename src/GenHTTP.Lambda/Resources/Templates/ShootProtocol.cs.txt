// What the two halves say to each other.
//
// Twenty of these a second to every player, so the field names are one letter
// and the numbers are whole: centimetres for positions, hundredths of a degree
// for angles. A body is about forty bytes and twenty of them fit in under a
// kilobyte, which is the budget.

/// <summary>What a browser holds down, packed into one number.</summary>
public static class Buttons
{
    public const int Forward = 1;
    public const int Back = 2;
    public const int Left = 4;
    public const int Right = 8;
    public const int Jump = 16;
    public const int Crouch = 32;
    public const int Fire = 64;
    public const int Reload = 128;
}

/// <summary>What the server says about somebody, packed the same way.</summary>
public static class Flags
{
    public const int Crouched = 1;
    public const int Airborne = 2;
    public const int Sheltered = 4;
    public const int Firing = 8;
    public const int Dead = 16;
}

/// <summary>Sent by a browser as its first message.</summary>
public record Join(string Kind, string Name);

/// <summary>
/// What a browser is doing, sent twenty times a second.
/// </summary>
/// <param name="I">Which input this is, echoed back so the browser knows what has landed</param>
/// <param name="B">What is held down</param>
/// <param name="Y">Where it is looking, in degrees</param>
/// <param name="P">And how far up or down</param>
/// <param name="S">
/// The moment it was drawing when it fired, in seconds since it arrived. Only
/// meaningful on a tick where the trigger went down.
/// </param>
public record Input(string Kind, int I, int B, double Y, double P, double S);

/// <summary>Somebody, as everybody else sees them.</summary>
/// <remarks>
/// H says whether they are up, and nothing more. How nearly dead somebody is
/// across the map is not something the game means to tell you, so the number
/// that would say it is only in Mine, where only they can read it.
///
/// K and D are here rather than in Mine because the scoreboard is everybody's
/// and it would be a strange scoreboard that only had one row filled in.
/// </remarks>
public record Body(string I, string N, int T, int X, int Y, int Z, int A, int P, int H, int F, int K, int D);

/// <summary>A shot, drawn as a line and heard as a crack.</summary>
public record Tracer(string I, int X0, int Y0, int Z0, int X1, int Y1, int Z1, int K);

/// <summary>Somebody killed somebody.</summary>
public record Fell(string By, string Who, bool Head);

/// <summary>Who you are and what the map looks like, sent once.</summary>
public record Welcome(string Kind, string Id, int Side, double[][] Solids, string[] Teams, double Width, double Length);

/// <summary>
/// One tick, as one player sees it.
/// </summary>
/// <param name="B">Everybody</param>
/// <param name="S">Shots taken since the last one</param>
/// <param name="K">Anybody killed since the last one</param>
/// <param name="M">Your own health, ammo and reload, which is nobody else's business</param>
/// <param name="D">Which way you were hurt from, or nothing</param>
/// <param name="I">The last input of yours that was acted on</param>
public record Frame(string Kind, double T, List<Body> B, List<Tracer> S, List<Fell> K, Mine M, double[] D, int I);

/// <summary>The part of a frame that is only for the one reading it.</summary>
/// <param name="T">
/// A heading to adopt, sent once on the tick somebody arrives and null on
/// every other one. The aim belongs to the browser the rest of the time.
/// </param>
public record Mine(int Health, int Ammo, double Reloading, double Returns, int Kills, int Deaths, double? T);
