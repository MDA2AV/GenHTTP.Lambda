// Accounts, passwords and sessions.
//
// Passwords are never stored: an account keeps a random salt and a PBKDF2
// hash of the password with it, and signing in hashes the attempt the same way
// and compares. Sessions are random tokens the browser keeps in a cookie; only
// a hash of each is stored, so the database cannot be used to sign in as
// anybody.
//
// Where the lambda has one, a pepper goes into every hash as well: a secret
// kept in the lambda's secrets rather than beside the hashes, so a copy of the
// database on its own cannot even be guessed at. An account remembers whether
// it was peppered, so accounts made before there was a pepper still sign in -
// data outlives the version that wrote it.
//
// Both live in the lambda's database, in the tables migrations/ makes, read
// and written through Entity Framework Core. Every call makes a context on a
// connection of its own, uses it synchronously and lets it go.

using System.Globalization;
using System.Security.Cryptography;

using GenHTTP.Api.Content.Authentication;

using Microsoft.EntityFrameworkCore.Storage.ValueConversion;

/// <summary>What is sent to register or to sign in.</summary>
public record Credentials(string Name, string Password);

/// <summary>What a successful registration or sign in answers with - the token travels in the cookie.</summary>
public record SignedIn(string Name);

/// <summary>What the members page shows about you.</summary>
public record Profile(string Name, DateTime Joined, int Members);

/// <summary>Somebody on the members page's list.</summary>
public record Newcomer(string Name, DateTime Joined);

/// <summary>
/// The cookie a session travels in.
/// </summary>
public static class Session
{
    public const string Cookie = "session";

    public static readonly TimeSpan Lifetime = TimeSpan.FromDays(7);

    /// <summary>
    /// HttpOnly: no script can read it, so a script injected into a page cannot
    /// steal it. SameSite Lax: another site cannot make the browser send it with
    /// a form of its own. Secure where the page came over HTTPS, so it never
    /// travels in the clear. No path: see lambda.cs.
    /// </summary>
    public static CookieOptions Options(IRequest request, bool expire = false) => new()
    {
        HttpOnly = true,
        SameSite = SameSite.Lax,
        Secure = request.Client.Protocol == ClientProtocol.Https,
        MaxAge = expire ? TimeSpan.Zero : Lifetime
    };
}

/// <summary>
/// The signed in user, as the Authentication module hands it to a route.
/// </summary>
public record Member(string Name) : IUser
{
    public string DisplayName => Name;
}

#region Records

/// <summary>An account, as the table holds it.</summary>
public sealed class Account
{
    /// <summary>The name in capitals, which is what makes an account - the key of the table.</summary>
    public string Folded { get; set; }

    /// <summary>The name as it was first registered, which is how it is shown.</summary>
    public string Name { get; set; }

    public string Salt { get; set; }

    public string Hash { get; set; }

    public DateTime Joined { get; set; }

    public bool Peppered { get; set; }
}

/// <summary>A session, as the table holds it: the hash of its token, never the token.</summary>
public sealed class StoredSession
{
    public string TokenHash { get; set; }

    /// <summary>The account it signs in, by its key.</summary>
    public string Account { get; set; }

    /// <summary>The account itself, which a query follows rather than joins by hand.</summary>
    public Account Owner { get; set; }

    public DateTime Expires { get; set; }
}

/// <summary>The lambda's database, as Entity Framework Core sees it.</summary>
public sealed class MemberContext(SqliteConnection connection) : DbContext
{
    public DbSet<Account> Accounts => Set<Account>();

    public DbSet<StoredSession> Sessions => Set<StoredSession>();

    protected override void OnConfiguring(DbContextOptionsBuilder options)
        => options.UseSqlite(connection, contextOwnsConnection: true);

    // every time as ISO 8601 text in UTC, which sorts and compares as it should
    protected override void ConfigureConventions(ModelConfigurationBuilder configuration)
        => configuration.Properties<DateTime>().HaveConversion<UtcText>();

    // the tables and their columns are the migration's; this only maps them
    protected override void OnModelCreating(ModelBuilder model)
    {
        model.Entity<Account>().ToTable("accounts").HasKey(a => a.Folded);

        var session = model.Entity<StoredSession>().ToTable("sessions");

        session.HasKey(s => s.TokenHash);
        session.Property(s => s.TokenHash).HasColumnName("token_hash");
        session.HasOne(s => s.Owner).WithMany().HasForeignKey(s => s.Account);
    }
}

/// <summary>A time as the tables keep it: ISO 8601 text in UTC.</summary>
public sealed class UtcText() : ValueConverter<DateTime, string>(
    value => value.ToUniversalTime().ToString("O", CultureInfo.InvariantCulture),
    text => DateTime.Parse(text, CultureInfo.InvariantCulture, DateTimeStyles.RoundtripKind));

#endregion

/// <param name="readPepper">Reads the pepper, or null where there is none - asked each time, so a new one counts at once</param>
public sealed class Accounts(Func<string> readPepper)
{
    // anybody can register on a demo, so it keeps the newest accounts only;
    // your own lambda would rather refuse than forget somebody
    private const int MaxAccounts = 300;

    // what SQLite answers when a key is taken
    private const int ConstraintFailed = 19;

    #region Routes

    public (SignedIn Member, string Token) Register(Credentials credentials)
    {
        var name = (credentials?.Name ?? "").Trim();
        var password = credentials?.Password ?? "";

        if (name.Length is < 2 or > 24 || !name.All(c => char.IsLetterOrDigit(c) || c is '-' or '_' or '.'))
        {
            throw new ProviderException(ResponseStatus.BadRequest, "A name has 2 to 24 letters, digits, dashes, dots or underscores.");
        }

        if (password.Length is < 8 or > 200)
        {
            throw new ProviderException(ResponseStatus.BadRequest, "A password needs at least 8 characters.");
        }

        var salt = RandomNumberGenerator.GetBytes(16);

        var pepper = readPepper();

        using var db = new MemberContext(Database.GetConnection());

        db.Accounts.Add(new Account
        {
            Folded = Fold(name),
            Name = name,
            Salt = Convert.ToBase64String(salt),
            Hash = Hash(password, salt, pepper),
            Joined = DateTime.UtcNow,
            Peppered = pepper != null
        });

        try
        {
            db.SaveChanges();
        }
        catch (DbUpdateException e) when (e.InnerException is SqliteException { SqliteErrorCode: ConstraintFailed })
        {
            // the account is the name in capitals, the key of the table - so
            // two people registering the same name at once cannot both win.
            // 409: the request was fine, it just clashes with what is there
            throw new ProviderException(ResponseStatus.Conflict, "That name is taken.");
        }

        Forget(db);

        return Open(db, name);
    }

    public (SignedIn Member, string Token) Login(Credentials credentials)
    {
        var name = (credentials?.Name ?? "").Trim();

        var folded = Fold(name);

        using var db = new MemberContext(Database.GetConnection());

        var account = db.Accounts.AsNoTracking().FirstOrDefault(a => a.Folded == folded);

        // an unknown name and a wrong password get the same answer, so
        // this cannot be used to find out who has an account
        if (account == null || !Matches(credentials?.Password ?? "", account.Salt, account.Hash, account.Peppered))
        {
            throw new ProviderException(ResponseStatus.Unauthorized, "That name and password do not go together.");
        }

        return Open(db, account.Name);
    }

    public Profile Describe(Member member)
    {
        var folded = Fold(member.Name);

        using var db = new MemberContext(Database.GetConnection());

        var account = db.Accounts.AsNoTracking().FirstOrDefault(a => a.Folded == folded)
                   ?? throw new ProviderException(ResponseStatus.NotFound, "That account is gone.");

        return new Profile(account.Name, account.Joined, db.Accounts.Count());
    }

    public List<Newcomer> Newest()
    {
        using var db = new MemberContext(Database.GetConnection());

        return db.Accounts.OrderByDescending(a => a.Joined)
                          .Take(10)
                          .Select(a => new Newcomer(a.Name, a.Joined))
                          .ToList();
    }

    /// <summary>Ends the session of the token presented, if it is one.</summary>
    public void Logout(string token)
    {
        var hash = HashToken(token);

        using var db = new MemberContext(Database.GetConnection());

        db.Sessions.Where(s => s.TokenHash == hash).ExecuteDelete();
    }

    #endregion

    #region Sessions

    /// <summary>
    /// The member a session cookie belongs to, or null - which the
    /// Authentication module answers with 403.
    /// </summary>
    public Member Authenticate(string token)
    {
        using var db = new MemberContext(Database.GetConnection());

        var hash = HashToken(token);

        var now = DateTime.UtcNow;

        // following Owner joins the account in; a session of an account that is gone is no session
        var name = db.Sessions.Where(s => s.TokenHash == hash && s.Expires > now)
                              .Select(s => s.Owner.Name)
                              .FirstOrDefault();

        return name != null ? new Member(name) : null;
    }

    private (SignedIn Member, string Token) Open(MemberContext db, string name)
    {
        var token = Convert.ToHexString(RandomNumberGenerator.GetBytes(32));

        var now = DateTime.UtcNow;

        // the expired ones go whenever a new one comes
        db.Sessions.Where(s => s.Expires <= now).ExecuteDelete();

        db.Sessions.Add(new StoredSession { TokenHash = HashToken(token), Account = Fold(name), Expires = now + Session.Lifetime });

        db.SaveChanges();

        return (new SignedIn(name), token);
    }

    /// <summary>Drops the oldest accounts once there are too many - a demo's housekeeping. Their sessions go with them.</summary>
    private static void Forget(MemberContext db)
        => db.Accounts.OrderByDescending(a => a.Joined).Skip(MaxAccounts).ExecuteDelete();

    #endregion

    #region Hashing

    // slow on purpose: a stolen database of hashes should take years to guess,
    // not minutes; the pepper, mixed in first, is what the database does not contain
    private static string Hash(string password, byte[] salt, string pepper)
    {
        var input = pepper == null
            ? password
            : Convert.ToBase64String(HMACSHA256.HashData(Encoding.UTF8.GetBytes(pepper), Encoding.UTF8.GetBytes(password)));

        return Convert.ToBase64String(Rfc2898DeriveBytes.Pbkdf2(input, salt, 100_000, HashAlgorithmName.SHA256, 32));
    }

    // compared in constant time, so how long the answer takes says nothing
    // about how close a guess was; an account made without a pepper is
    // checked without one, and one made with it cannot be checked without it
    private bool Matches(string password, string salt, string hash, bool peppered)
    {
        var pepper = peppered ? readPepper() : null;

        if (peppered && pepper == null)
        {
            return false;
        }

        return CryptographicOperations.FixedTimeEquals(Convert.FromBase64String(Hash(password, Convert.FromBase64String(salt), pepper)),
                                                       Convert.FromBase64String(hash));
    }

    // a token is already random and long, so a fast hash is enough here
    private static string HashToken(string token)
        => Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(token ?? "")));

    #endregion

    // what makes an account of a name: the same for "Jürgen" and "JÜRGEN", as
    // string.Equals with OrdinalIgnoreCase would say
    private static string Fold(string name) => name.ToUpperInvariant();

}
