// A game in the browser, with a scoreboard the server keeps.
//
//   GET  /lambda/{{KEY}}/            the game
//   GET  /lambda/{{KEY}}/api/scores  the ten best runs
//   POST /lambda/{{KEY}}/api/scores  add one
//
// Three files. This one is the wiring and the only one that touches storage;
// Scores.cs holds the rules about what may go on the board; Page.cs is the
// game, which runs in the browser and which this lambda never sees.
//
// The board is a file in Workspace, the private folder this lambda gets, which
// is why it is still there after a redeployment.

var json = new JsonSerializerOptions
{
    PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
    PropertyNameCaseInsensitive = true
};

var board = new List<Score>();
var writing = new object();

if (Workspace.Exists("scores.json"))
{
    board.AddRange(JsonSerializer.Deserialize<List<Score>>(Workspace.ReadText("scores.json"), json) ?? []);
}

List<Score> Best()
{
    lock (writing)
    {
        return Scoreboard.Best(board);
    }
}

Posted Submit(Attempt attempt)
{
    if (Scoreboard.Refuse(attempt) is string problem)
    {
        return new Posted(false, problem, Best());
    }

    lock (writing)
    {
        board.Add(new Score(attempt.Name.Trim(), attempt.Points, DateTime.UtcNow));

        var kept = Scoreboard.Trim(board);

        board.Clear();
        board.AddRange(kept);

        Workspace.WriteText("scores.json", JsonSerializer.Serialize(board, json));
    }

    return new Posted(true, null, Best());
}

var api = Inline.Create()
                .Get("scores", () => Best())
                .Post("scores", (Attempt body) => Submit(body));

return Layout.Create()
             .Add("api", api)
             .Add(Content.From(Resource.FromString(Site.Page).Type(new ContentType("text/html; charset=utf-8"))));
