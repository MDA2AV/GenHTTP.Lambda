// What the server keeps about a fight while it is happening.
//
// Mutable on purpose, unlike the records that go over the socket: this is
// stepped twenty times a second, and rebuilding every tank to move it a metre
// would be a lot of rubbish for the collector to sweep up. What leaves the
// process is still the immutable shape in Protocol.cs.

sealed class Crew
{
    public string Id;
    public string Name;
    public string Colour;

    public double X;
    public double Z;
    public double Heading;
    public double Aim;

    public int Health = Battle.Health;
    public int Score;

    /// <summary>Null while alive; when it is due to come back otherwise.</summary>
    public DateTime? Down;

    /// <summary>What the browser last said its controls were doing.</summary>
    public double Throttle;
    public double Steer;
    public bool Firing;

    /// <summary>When it may shoot again.</summary>
    public DateTime Ready = DateTime.MinValue;

    /// <summary>Which crate is running, and until when.</summary>
    public string Boost;
    public DateTime BoostUntil;

    public bool Alive => Down == null;

    public bool Shielded => Boost == "shield" && DateTime.UtcNow < BoostUntil;

    public double Reload => Boost == "rapid" && DateTime.UtcNow < BoostUntil ? Battle.FastReload : Battle.Reload;

    /// <summary>Seconds of boost left, which is what the browser draws a bar from.</summary>
    public double Remaining => Boost == null || DateTime.UtcNow >= BoostUntil
                             ? 0
                             : (BoostUntil - DateTime.UtcNow).TotalSeconds;
}

sealed class Round
{
    public string Id;
    public string Owner;

    public double X;
    public double Z;
    public double Heading;

    public DateTime Gone;
}
