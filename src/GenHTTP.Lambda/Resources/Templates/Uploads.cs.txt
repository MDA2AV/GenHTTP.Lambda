// A single page application served from files on disk, using the module that
// does it rather than the shortcut.
//
//   GET /lambda/{{KEY}}/            whatever index.html in the workspace says
//   GET /lambda/{{KEY}}/api/files   what is in the workspace right now
//
// Nothing here is shipped with the code. Every file this serves lives in the
// workspace: this snippet writes a few on its first run so there is something
// to look at, and after that they are yours to replace. Open Storage, go to
// the Workspace tab, and upload over them. The change is live immediately,
// with no deploy, because what is being served is the directory itself.
//
// Two lines do the serving:
//
//   Workspace.Tree()                     the directory, as something servable
//   SinglePageApplication.From(tree)     index.html as the shell for all of it
//
// The second is the GenHTTP module itself, used exactly as it would be in any
// project. The first is not ResourceTree.FromDirectory, which is what you
// would reach for outside here and which this platform refuses: it takes a
// path, and nothing about the way it is written tells the difference between
// your own directory and somebody else's. Tree() hands back the same kind of
// thing for the one directory you are allowed to have.
//
// Workspace.App() is these two lines and nothing else. Worth seeing apart
// once, because ServerSideRouting is what people go looking for when a deep
// link gives them a 404 on reload.

if (!Workspace.Exists("index.html"))
{
    Seed();
}

/*
 * The tree is built once, when the handler is. It reads the directory as
 * requests arrive rather than taking a copy, which is why a file uploaded a
 * minute from now is served a minute from now without anything being redeployed.
 */
var tree = Workspace.Tree();

var site = SinglePageApplication.From(tree)
                                .ServerSideRouting();

var api = Inline.Create()
                .Get("files", () => Workspace.List()
                                             .Select(name => new Listed(name, Workspace.ReadBytes(name).Length))
                                             .OrderBy(f => f.Name));

return Layout.Create()
             .Add("api", api)
             .Add(site);

/// <summary>
/// Writes something to look at, once, so the example is not an empty folder.
/// </summary>
void Seed()
{
    Workspace.WriteText("index.html", """
        <!doctype html>
        <html lang="en">
        <meta charset="utf-8">
        <meta name="viewport" content="width=device-width,initial-scale=1">
        <title>Served from the workspace</title>
        <link rel="stylesheet" href="app.css">

        <main>
          <h1>This page is a file on disk</h1>
          <p>
            It is not shipped with the code. It sits in this lambda's workspace, and the
            snippet serves that directory with
            <code>SinglePageApplication.From(Workspace.Tree())</code>.
          </p>
          <p>
            Open <b>Storage</b>, go to the <b>Workspace</b> tab, and upload something over
            <code>index.html</code>. Reload. No deploy.
          </p>

          <h2>What is in there</h2>
          <ul id="files"><li>Loading...</li></ul>

          <p class="dim">
            Try <a href="somewhere/that/does/not/exist">a path that is not a file</a>. You get
            this page rather than a 404, which is what <code>ServerSideRouting</code> is for:
            a front end doing its own routing still works when somebody reloads on a deep link.
          </p>
        </main>

        <script src="app.js" type="module"></script>
        """);

    Workspace.WriteText("app.css", """
        :root { color-scheme: light dark; --ink: #1f2328; --dim: #6a737d; --line: #d8dee4; --card: #fff; --page: #f6f8fa; }

        @media (prefers-color-scheme: dark) {
          :root { --ink: #e6edf3; --dim: #8b949e; --line: #30363d; --card: #161b22; --page: #0d1117; }
        }

        body {
          margin: 0;
          background: var(--page);
          color: var(--ink);
          font: 16px/1.6 -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, sans-serif;
        }

        main { max-width: 36rem; margin: 0 auto; padding: 3rem 1.25rem 4rem; }

        h1 { font-size: 1.6rem; margin: 0 0 .75rem; }
        h2 { font-size: 1rem; margin: 2rem 0 .5rem; }

        .dim { color: var(--dim); font-size: .9rem; }

        code {
          font-family: ui-monospace, SFMono-Regular, Menlo, monospace;
          font-size: .82em;
          background: var(--card);
          border: 1px solid var(--line);
          border-radius: 4px;
          padding: .1em .35em;
        }

        ul { list-style: none; margin: 0; padding: 0; }

        li {
          display: flex;
          justify-content: space-between;
          gap: 1rem;
          background: var(--card);
          border: 1px solid var(--line);
          border-radius: 6px;
          padding: .5rem .75rem;
          margin-bottom: .4rem;
          font-family: ui-monospace, SFMono-Regular, Menlo, monospace;
          font-size: .85rem;
        }

        li span:last-child { color: var(--dim); }
        """);

    Workspace.WriteText("app.js", """
        // Also a file in the workspace. Replace it and this page does something else.

        const list = document.getElementById('files');

        const size = (n) => (n < 1024 ? `${n} B` : `${(n / 1024).toFixed(1)} kB`);

        try {
          const files = await (await fetch(new URL('api/files', location.href.replace(/[^/]*$/, '')))).json();

          list.innerHTML = files
            .map((f) => `<li><span>${f.name}</span><span>${size(f.size)}</span></li>`)
            .join('') || '<li>nothing here</li>';
        } catch {
          list.innerHTML = '<li>could not read the workspace</li>';
        }
        """);
}

record Listed(string Name, int Size);
