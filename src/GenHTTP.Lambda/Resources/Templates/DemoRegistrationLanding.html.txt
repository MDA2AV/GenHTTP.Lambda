<script type="module">
  import { call } from './session.js';

  // somebody signed in has no business here - the cookie says whether they are
  try {
    await call('api/me');
    location.replace('members.html');
  } catch { }

  const form = document.getElementById('form');
  const submit = document.getElementById('submit');
  const error = document.getElementById('error');

  let mode = 'register';

  document.querySelectorAll('[data-mode]').forEach((tab) => tab.addEventListener('click', () => {
    mode = tab.dataset.mode;
    document.querySelectorAll('[data-mode]').forEach((t) => t.setAttribute('aria-selected', t === tab));
    submit.textContent = tab.textContent;
    form.password.autocomplete = mode === 'register' ? 'new-password' : 'current-password';
    error.textContent = '';
  }));

  form.addEventListener('submit', async (event) => {
    event.preventDefault();
    error.textContent = '';
    submit.disabled = true;

    try {
      // the answer sets the session cookie; nothing to keep on this side
      await call(mode, {
        method: 'POST',
        body: { name: form.name.value, password: form.password.value }
      });

      location.replace('members.html');
    } catch (e) {
      error.textContent = e.message;
    } finally {
      submit.disabled = false;
    }
  });
</script>
