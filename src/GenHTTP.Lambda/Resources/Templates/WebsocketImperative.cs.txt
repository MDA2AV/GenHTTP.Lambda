// The same websocket, read frame by frame. Connect to it with
//
//   wss://<this-host>/lambda/{{KEY}}/
//
// Nothing is called for you here: the loop below owns the connection and
// decides what every frame means, which is the flavour to pick when the order
// of messages matters or the protocol has a handshake of its own.
//
// FrameType is one of the names every lambda is given, so no import is needed.

return Websocket.Imperative().Handler(new Echo());

public class Echo : IImperativeHandler
{
    public async ValueTask HandleAsync(IImperativeConnection connection)
    {
        await connection.WritePayloadAsync("welcome");

        while (connection.Request.Server.Running)
        {
            var frame = await connection.ReadFrameAsync();

            if (frame.Type == FrameType.Ping)
            {
                await connection.PongAsync();
            }
            else if (frame.Type == FrameType.Text)
            {
                var text = await frame.ReadPayloadAsync<string>();

                await connection.WritePayloadAsync($"you said: {text}");
            }
            else if (frame.Type == FrameType.Close)
            {
                await connection.CloseAsync();
                break;
            }
        }
    }
}
