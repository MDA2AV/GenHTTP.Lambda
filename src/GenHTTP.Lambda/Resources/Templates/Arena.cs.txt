// An arena everybody shares: eat the pellets, grow, and eat anybody smaller.
//
//   GET /lambda/{{KEY}}/            the game
//   GET /lambda/{{KEY}}/api/best    the best runs it remembers
//   WS  /lambda/{{KEY}}/play        everything that happens
//
// Seven files: this one is the socket, the clock and the storage, World.cs
// the rules, Protocol.cs what the two halves say, State.cs a player as the
// server holds one; index.html, game.js and style.css ship as assets.
//
// Written to be leaned on. Everything that moves is decided here and sent to
// everybody twenty times a second, so the number that matters is not the
// frame rate but how many fit in before the socket gives. Only what falls
// inside a player's view is sent to them.

var json = new JsonSerializerOptions
{
    PropertyNamingPolicy = JsonNamingPolicy.CamelCase,

    // the three newest fields in a frame are empty on almost every tick -
    // nobody is teleporting, nobody has just been eaten - and a null is left
    // out altogether where an empty list still costs its name and brackets
    DefaultIgnoreCondition = JsonIgnoreCondition.WhenWritingNull,
    PropertyNameCaseInsensitive = true
};

var players = new ConcurrentDictionary<IReactiveConnection, Player>();

/*
 * Blobs with nobody behind them.
 *
 * Kept apart from the players because the players are keyed by the socket
 * they arrived on and these have no socket - but everything that decides what
 * the arena does treats the two the same, which is what Souls is for. They
 * are eaten, they grow, they appear on the board, and the only thing they
 * never get is a frame.
 */
var bots = new ConcurrentDictionary<string, Player>();

/// <summary>Everybody in the arena, driven or not.</summary>
Player[] Souls()
{
    var driven = players.Values;
    var idle = bots.Values;

    var all = new List<Player>(driven.Count + idle.Count);

    all.AddRange(driven);
    all.AddRange(idle);

    return [.. all];
}

/*
 * The connections part way through joining.
 *
 * A message handler is asynchronous, so two messages from the same socket can
 * both be inside it at once. A browser sends join and then starts steering
 * twenty times a second, and if a steer arrives before the join has finished
 * writing the welcome, the second one finds nobody registered, decides it is
 * a join too, and writes a second welcome into a socket already being written
 * to. That is a fault, and the player is dropped without being told why.
 *
 * One join at a time per connection. Anything else arriving meanwhile is
 * dropped, which costs a steer that would have been replaced fifty
 * milliseconds later anyway.
 */
var arriving = new ConcurrentDictionary<IReactiveConnection, byte>();

// one write at a time per socket, see Send
var gates = new ConcurrentDictionary<IReactiveConnection, SemaphoreSlim>();

/*
 * When each socket arrived and how much has gone out on it.
 *
 * Only for saying what a connection was doing just before it stopped. The
 * engine reports a faulted handler with the exception's message and nothing
 * else - no socket, no age, no idea whether it had been serving happily for an
 * hour or had never worked - and that one line is all there was to go on.
 */
var since = new ConcurrentDictionary<IReactiveConnection, (DateTime At, int Sent)>();

/// <summary>
/// Says something about one connection, once, in a form the panel can fold.
/// </summary>
/// <remarks>
/// Deliberately not a line per frame. What is wanted is the handful of events
/// around a failure - the write that threw, the error frame that arrived, the
/// close that followed - and a running commentary would bury them.
/// </remarks>
void Note(IReactiveConnection connection, string what)
{
    var life = since.TryGetValue(connection, out var seen)
        ? $"{(DateTime.UtcNow - seen.At).TotalSeconds:N1}s, {seen.Sent} sent"
        : "unseen";

    Console.WriteLine($"[socket {connection.GetHashCode():x8}] {what} ({life})");
}
var pellets = new ConcurrentDictionary<int, Pellet>();

/*
 * Slot numbers, handed out and taken back. A pellet has to be recognisable
 * across two frames now that they move - the browser draws between the last
 * two things it was told, and cannot interpolate what it cannot match up -
 * and recycling keeps those numbers short enough to be worth sending.
 *
 * A freed slot is left alone for a while before it is handed out again. The
 * browser is drawing a tenth of a second in the past, so a slot reused
 * immediately would be the old pellet in one of the two frames it is reading
 * between and a brand new one somewhere else in the other, and it would draw
 * the difference: a dot sliding the width of the arena. A second of quiet is
 * several times longer than that window.
 */
var spare = new Queue<(int Slot, long Tick)>();

var nextSlot = 0;

var tick = 0L;

/*
 * Teleports part-way across, and blobs eaten, this tick.
 *
 * Both are moments rather than things, so both are built from scratch every
 * tick and read once, by the broadcast that follows it.
 */
var marks = new List<Mark>();

/*
 * What has been said since the last broadcast, and where everybody is.
 *
 * Both are the same shape as the two above: built during a tick, read once
 * by the broadcast that follows it, and kept nowhere afterwards.
 */
/*
 * A queue rather than a list: this one is filled from whatever thread the
 * socket happened to deliver on and drained by the clock, which is not true
 * of the other two - they are both written and read inside a tick.
 */
var chatter = new ConcurrentQueue<Said>();

List<int> map = null;

var feasts = new List<Feast>();

/// <summary>Seconds since the arena started, which is what the bruise window is measured against.</summary>
var at = 0.0;

const int Quiet = 20;

int TakeSlot()
{
    if (spare.Count > 0 && tick - spare.Peek().Tick > Quiet)
    {
        return spare.Dequeue().Slot;
    }

    return nextSlot++;
}

void Drop(Pellet pellet)
{
    if (pellets.TryRemove(pellet.Slot, out _))
    {
        spare.Enqueue((pellet.Slot, tick));
    }
}

// clock and from every socket at once. A Random used from two threads has its

// empties out everywhere else. Nothing throws; it just goes wrong.
var random = Random.Shared;

/*
 * How big the arena is right now. It follows the crowd, so it is read once a
 * tick and then used everywhere for that tick rather than recomputed - a
 * world that changed size halfway through a tick would clamp some players
 * against one edge and scatter pellets past it.
 */
var size = World.Smallest;

/// <summary>
/// The pellets, in buckets, so that nobody has to look at all of them.
/// </summary>
/// <remarks>
/// Rebuilt once a tick and then read by everything for the rest of it. It is
/// held rather than passed because the broadcast needs it too, and it is
/// replaced whole rather than edited so the threads reading it during the
/// fan out are never looking at one being changed underneath them.
/// </remarks>
var floor = new Dictionary<long, List<Pellet>>();

/// <summary>The crowd, in the same shape and for the same reason.</summary>
var herd = new Dictionary<long, List<Player>>();

/*
 * The few blobs big enough to bend the floor around them.
 *
 * The only thing that wants the whole crowd at once is the search for
 * somewhere to put somebody, and all it asks is whether the spot is inside
 * somebody's pull - which nothing under the well size has. At ten thousand
 * blobs that is a handful read rather than all of them, twelve times, per
 * spawn.
 */
var heavies = new List<Player>();

/// <summary>When the arena last had nobody in it at all.</summary>
var emptied = DateTime.UtcNow;

/*
 * Blobs whose player is not connected just now.
 *
 * Keyed by a secret the welcome gave them, which is the only thing that can
 * claim one back. They are out of the arena while they wait - not standing
 * there to be eaten by whatever happens past - and they expire, so a size is
 * a thing you come back to in the next few seconds rather than bank.
 */
var adrift = new ConcurrentDictionary<string, (double Radius, string Colour, DateTime Since)>();

void Weigh(Player[] everyone)
{
    heavies.Clear();

    foreach (var soul in everyone)
    {
        if (!soul.Gone && soul.Radius >= World.WellSize)
        {
            heavies.Add(soul);
        }
    }
}

/// <summary>Puts the crowd back in its squares, once, at the top of a tick.</summary>
/// <remarks>
/// Everything that follows asks this rather than everybody: ten thousand
/// adds against the hundred million comparisons it saves.
/// </remarks>
void Reherd(Player[] everyone)
{
    // emptied and refilled rather than rebuilt: at ten thousand blobs twenty
    // times a second, throwing the squares away and making new ones is ten
    // thousand allocations a tick for a dictionary that is nearly always the
    // same shape as it was
    foreach (var bucket in herd.Values)
    {
        bucket.Clear();
    }

    foreach (var soul in everyone)
    {
        if (soul.Gone)
        {
            continue;
        }

        var key = World.PatchOf(soul.X, soul.Y);

        if (!herd.TryGetValue(key, out var bucket))
        {
            herd[key] = bucket = [];
        }

        bucket.Add(soul);
    }
}

/// <summary>Everybody within a square of this reach, without walking the rest.</summary>
/// <remarks>
/// Falls back to the whole crowd when the reach covers most of the arena: a
/// hole's pull does, and walking sixty thousand empty squares to find ten
/// thousand blobs is slower than walking the blobs.
/// </remarks>
IEnumerable<Player> Around(double x, double y, double reach, Player[] everyone)
{
    if (reach >= size * 0.4)
    {
        return everyone;
    }

    return Gathered(x, y, reach);
}

IEnumerable<Player> Gathered(double x, double y, double reach)
{
    foreach (var key in World.PatchesWithin(x, y, reach + World.Slip))
    {
        if (!herd.TryGetValue(key, out var bucket))
        {
            continue;
        }

        foreach (var soul in bucket)
        {
            yield return soul;
        }
    }
}

/// <summary>Puts the floor back in its squares, once, at the top of a tick.</summary>
/// <remarks>
/// Emptied and refilled rather than rebuilt. Sixty thousand pellets is sixty
/// thousand adds either way, but rebuilding also threw away the dictionary
/// and every list in it and made them again - sixty thousand allocations a
/// tick, twenty times a second, for a structure that is almost always the
/// same shape it was. At five hundred players this and its sibling were
/// sixty milliseconds of a tick that has fifty to spend.
/// </remarks>
void Rebucket()
{
    foreach (var bucket in floor.Values)
    {
        bucket.Clear();
    }

    foreach (var pellet in pellets.Values)
    {
        var key = World.KeyOf(pellet.X, pellet.Y);

        if (!floor.TryGetValue(key, out var bucket))
        {
            floor[key] = bucket = [];
        }

        bucket.Add(pellet);
    }
}

/// <summary>
/// Somewhere to put a pellet: a few guesses, and whichever of them lands
/// where the floor is thinnest wins.
/// </summary>
/// <remarks>
/// Scattering uniformly sounds fair and is not, because what is eaten is
/// not eaten uniformly. Players strip the ground they stand on and big ones
/// drag in everything around them, so a uniform scatter refills the empty
/// patches at the same rate as the full ones and the empty patches stay
/// empty - the arena ends up with a bare middle and a well stocked border
/// that nobody visits. Three guesses is enough to level it out and costs a
/// dictionary lookup each.
/// </remarks>
(double X, double Y) Thinnest()
{
    var bestX = random.NextDouble() * size;
    var bestY = random.NextDouble() * size;

    var fewest = int.MaxValue;

    for (var guess = 0; guess < 3; guess++)
    {
        var x = random.NextDouble() * size;
        var y = random.NextDouble() * size;

        var held = floor.TryGetValue(World.KeyOf(x, y), out var bucket) ? bucket.Count : 0;

        if (held < fewest)
        {
            fewest = held;
            bestX = x;
            bestY = y;
        }
    }

    return (bestX, bestY);
}

void Scatter()
{
    var wanted = World.PelletsFor(size);

    // whatever is now outside the arena comes off the floor first. A world
    // that has just shrunk is otherwise left holding its whole count out
    // past the edge, where nobody can reach it and where it still counts
    // against the target, so nothing is ever scattered anywhere useful again
    foreach (var pellet in pellets.Values)
    {
        if (pellet.X > size || pellet.Y > size)
        {
            Drop(pellet);
        }
    }

    while (pellets.Count < wanted)
    {
        var slot = TakeSlot();

        var spot = Thinnest();

        pellets[slot] = new Pellet
        {
            Slot = slot,
            X = spot.X,
            Y = spot.Y,
            C = World.AnyTint(random),
            Grade = 0
        };
    }
}

Scatter();

/// <summary>
/// Somewhere to arrive that is not already inside somebody's pull.
/// </summary>
/// <remarks>
/// Landing next to a giant is not a bad start, it is no start at all: the
/// drag is on before the first frame is drawn and there is nothing the
/// player could have done. A handful of goes is enough - the arena grows
/// with the crowd, so there is always somewhere.
/// </remarks>
(double X, double Y) Somewhere()
{
    for (var attempt = 0; attempt < 12; attempt++)
    {
        var x = random.NextDouble() * size;
        var y = random.NextDouble() * size;

        var clear = true;

        foreach (var other in heavies)
        {
            var reach = World.Reach(other.Radius, 1);

            if (Math.Abs(other.X - x) < reach && Math.Abs(other.Y - y) < reach)
            {
                clear = false;

                break;
            }
        }

        if (clear)
        {
            return (x, y);
        }
    }

    return (random.NextDouble() * size, random.NextDouble() * size);
}

void Remember(Player player)
{
    var score = World.ScoreOf(player.Radius);

    if (score <= 0)
    {
        return;
    }

    lock (writing)
    {
        best.Add(new Score(player.Name, (int)score));

        var kept = best.OrderByDescending(s => s.S).Take(10).ToList();

        best.Clear();
        best.AddRange(kept);

        Workspace.WriteText("best.json", JsonSerializer.Serialize(best, json));
    }
}


/// <summary>
/// Moves a blob along the line it was given, far faster than it can walk.
/// </summary>
/// <remarks>
/// Leg by leg rather than towards the far end, because the shape is the
/// point: a line drawn round the back of somebody has to actually go round
/// the back of them. Whatever is left of a tick's travel after finishing one
/// leg is spent on the next, so a line made of many short segments is run as
/// fast as one made of two long ones.
/// </remarks>
void Run(Player player, double seconds)
{
    var budget = World.StreakSpeed * seconds;

    while (budget > 0 && player.StreakX != null)
    {
        var next = player.StreakLeg + 1;

        if (next >= player.StreakX.Length)
        {
            player.StreakX = null;
            player.StreakY = null;
            return;
        }

        var toX = player.StreakX[next] - player.X;
        var toY = player.StreakY[next] - player.Y;

        var leg = Math.Sqrt(toX * toX + toY * toY);

        if (leg < 0.5)
        {
            player.StreakLeg = next;
            continue;
        }

        var step = Math.Min(budget, leg);

        player.X = World.Clamp(player.X + toX / leg * step, size);
        player.Y = World.Clamp(player.Y + toY / leg * step, size);

        budget -= step;

        if (step >= leg - 0.5)
        {
            player.StreakLeg = next;
        }
    }
}

/// <summary>
/// Puts somebody somewhere else, if they have it to spend.
/// </summary>
/// <remarks>
/// Bounded by distance and by a share of the blob, and refused outright to a
/// hole - something that has stopped being able to move should not be able to
/// cross the arena either.
/// </remarks>
bool Jump(Player player, double x, double y)
{
    if (at < player.WarpReady || World.IsHole(player.Radius) || player.StreakX != null || player.WarpLands > 0)
    {
        return false;
    }

    var toX = World.Clamp(x, size) - player.X;
    var toY = World.Clamp(y, size) - player.Y;

    var far = Math.Sqrt(toX * toX + toY * toY);

    if (far < 1)
    {
        return false;
    }

    var reach = Math.Min(far, World.WarpReach(player.Radius));

    // where, and from when: the ground is marked now and the blob crosses to
    // it over the lead, rather than being somewhere else by the time the key
    // comes back up
    player.WarpToX = World.Clamp(player.X + toX / far * reach, size);
    player.WarpToY = World.Clamp(player.Y + toY / far * reach, size);

    player.WarpLands = at + World.WarpLead;

    // both the toll and the wait are charged on the press rather than on the
    // arrival, or the crossing would be a window to press it again in
    player.WarpReady = at + World.WarpLead + World.WarpWait(player.Radius);

    var area = Math.PI * player.Radius * player.Radius * (1 - World.WarpToll);

    player.Radius = Math.Max(World.Start, Math.Sqrt(area / Math.PI));

    return true;
}

/// <summary>
/// Opens a whirl where a hole asks for one.
/// </summary>
/// <remarks>
/// Only a hole, and only one at a time. It is refused rather than queued -
/// a hole that presses twice should get the second one where it pressed the
/// second time, not both at once - and it is clamped to the arena, since
/// hauling things towards a point outside it would only pin them to a wall.
/// </remarks>
bool Open(Player hole, double x, double y)
{
    if (!World.IsHole(hole.Radius) || at < hole.WarpReady || hole.WhirlEnds > 0)
    {
        return false;
    }

    hole.WhirlX = World.Clamp(x, size);
    hole.WhirlY = World.Clamp(y, size);
    hole.WhirlEnds = at + World.WhirlTime;
    hole.WarpReady = at + World.WhirlTime + World.WhirlWait;

    return true;
}

/// <summary>
/// Hauls everything smaller towards every whirl that is turning.
/// </summary>
/// <remarks>
/// Applied before the blobs move rather than after, so that a tick spent in
/// a whirl is a tick of being dragged and not a tick of walking followed by
/// a correction. Nothing bigger than the hole that opened it is touched, and
/// neither is the hole itself: a whirl is a thing done to the arena, not a
/// thing the arena does back.
/// </remarks>
void Whirls(Player[] everyone, double seconds)
{
    foreach (var hole in everyone)
    {
        if (hole.WhirlEnds <= 0)
        {
            continue;
        }

        if (at >= hole.WhirlEnds || hole.Gone || !World.IsHole(hole.Radius))
        {
            hole.WhirlEnds = 0;

            continue;
        }

        foreach (var soul in Around(hole.WhirlX, hole.WhirlY, World.WhirlReach, everyone))
        {
            if (ReferenceEquals(soul, hole) || soul.Gone || soul.Radius >= hole.Radius)
            {
                continue;
            }

            var dx = hole.WhirlX - soul.X;
            var dy = hole.WhirlY - soul.Y;

            var far = Math.Sqrt(dx * dx + dy * dy);

            if (far < 1 || far > World.WhirlReach)
            {
                continue;
            }

            var haul = Math.Min(far, World.WhirlSpeed(far) * seconds);

            soul.X = World.Clamp(soul.X + dx / far * haul, size);
            soul.Y = World.Clamp(soul.Y + dy / far * haul, size);
        }

        // and the floor with them, because a whirl that moved the players
        // and left the food where it was would look like a bug in the food
        foreach (var key in World.KeysWithin(hole.WhirlX, hole.WhirlY, World.WhirlReach))
        {
            if (!floor.TryGetValue(key, out var bucket))
            {
                continue;
            }

            foreach (var pellet in bucket)
            {
                var dx = hole.WhirlX - pellet.X;
                var dy = hole.WhirlY - pellet.Y;

                var far = Math.Sqrt(dx * dx + dy * dy);

                if (far < 1 || far > World.WhirlReach)
                {
                    continue;
                }

                var haul = Math.Min(far, World.WhirlSpeed(far) * seconds);

                pellet.X = World.Clamp(pellet.X + dx / far * haul, size);
                pellet.Y = World.Clamp(pellet.Y + dy / far * haul, size);
            }
        }
    }
}

/// <summary>
/// Carries a blob the next tick's worth of the way through a teleport.
/// </summary>
/// <remarks>
/// The far end was fixed when the key went down, so this is not steering:
/// it takes the share of what is left to cross that belongs to this tick,
/// which puts the blob on its mark on the beat however long the tick
/// happened to run for, and holds a steady speed while it does it.
///
/// Nothing is made untouchable on the way. Crossing at several thousand
/// units a second there is nothing that can catch a blob mid-flight, but
/// there is plenty it can be sent into, and landing on something that eats
/// you is a consequence of where you aimed rather than of bad luck.
/// </remarks>
void Dash(Player player, double seconds)
{
    var left = player.WarpLands - at;

    if (left <= seconds)
    {
        player.X = player.WarpToX;
        player.Y = player.WarpToY;

        // steering follows, or the blob turns round and walks straight back
        player.WantX = player.X;
        player.WantY = player.Y;

        player.WarpLands = 0;

        return;
    }

    var share = seconds / left;

    player.X = World.Clamp(player.X + (player.WarpToX - player.X) * share, size);
    player.Y = World.Clamp(player.Y + (player.WarpToY - player.Y) * share, size);
}

/// <summary>
/// What a bot does next.
/// </summary>
/// <remarks>
/// Deliberately simple, and deliberately not very good. It runs from anything
/// big enough to eat it, chases anything it could eat, and otherwise goes to
/// the nearest food - which is enough to make an empty arena worth opening
/// without making it a place a person cannot win.
/// </remarks>
void Think(Player bot, double seconds)
{
    bot.Rethink -= seconds;

    var danger = double.MaxValue;
    Player from = null;
    Player prey = null;
    var nearest = double.MaxValue;

    // a bot never cared about anything further off than the larger of the two
    // bounds below, so asking for that square rather than for the whole arena
    // changes nothing about what it decides
    foreach (var other in Gathered(bot.X, bot.Y, 1400))
    {
        if (ReferenceEquals(other, bot) || other.Gone)
        {
            continue;
        }

        var gap = Math.Sqrt((other.X - bot.X) * (other.X - bot.X) + (other.Y - bot.Y) * (other.Y - bot.Y));

        if (other.Radius > bot.Radius * World.EatRatio && gap < danger && gap < 900)
        {
            danger = gap;
            from = other;
        }
        else if (bot.Radius > other.Radius * World.EatRatio && gap < nearest && gap < 1400)
        {
            nearest = gap;
            prey = other;
        }
    }

    // running away is decided every tick; everything else is decided rarely,
    // so a bot commits to a direction instead of shivering between two
    if (from != null)
    {
        var awayX = bot.X - from.X;
        var awayY = bot.Y - from.Y;

        var far = Math.Max(0.001, Math.Sqrt(awayX * awayX + awayY * awayY));

        bot.AimX = World.Clamp(bot.X + awayX / far * 1200, size);
        bot.AimY = World.Clamp(bot.Y + awayY / far * 1200, size);
        bot.Rethink = 0.4;

        // a cornered bot uses the thing everybody has
        if (danger < bot.Radius * 6)
        {
            Jump(bot, bot.AimX, bot.AimY);
        }
    }
    else if (prey != null)
    {
        bot.AimX = prey.X;
        bot.AimY = prey.Y;
        bot.Rethink = 0.3;
    }
    else if (bot.Rethink <= 0)
    {
        var food = Nearest(bot);

        bot.AimX = food.X;
        bot.AimY = food.Y;
        // the shared one, not the arena's: this runs on every core at once
        // and System.Random is not something two threads may hold at a time
        bot.Rethink = 0.8 + Random.Shared.NextDouble();
    }

    bot.WantX = bot.AimX;
    bot.WantY = bot.AimY;
}

/// <summary>
/// Somewhere near the middle, for everybody at once.
/// </summary>
/// <remarks>
/// Used when a round restarts. The arena is sized for its population, so with
/// a crowd in it a blob at starting size can see a few per cent of the floor -
/// and being scattered uniformly across that is being alone in it.
/// </remarks>
(double X, double Y) Together()
{
    var span = size * 0.33;
    var edge = (size - span) / 2;

    return (edge + random.NextDouble() * span, edge + random.NextDouble() * span);
}

/// <summary>The closest pellet, or somewhere to wander if the floor is bare.</summary>
(double X, double Y) Nearest(Player bot)
{
    var best = double.MaxValue;
    var found = (X: bot.X, Y: bot.Y);
    var any = false;

    foreach (var key in World.KeysWithin(bot.X, bot.Y, 1100))
    {
        if (!floor.TryGetValue(key, out var dust))
        {
            continue;
        }

        foreach (var pellet in dust)
        {
            var gap = (pellet.X - bot.X) * (pellet.X - bot.X) + (pellet.Y - bot.Y) * (pellet.Y - bot.Y);

            if (gap < best)
            {
                best = gap;
                found = (pellet.X, pellet.Y);
                any = true;
            }
        }
    }

    return any ? found : Somewhere();
}

/// <summary>
/// Puts some company in the arena.
/// </summary>
/// <remarks>
/// Asked for rather than always present, and capped, because an arena that
/// fills itself is one nobody can ever have to themselves. They are cleared
/// when the last person leaves, so an empty arena does not quietly keep a
/// dozen of them running.
/// </remarks>
void Populate(int wanted)
{
    Weigh(Souls());

    var names = new[] { "Mote", "Husk", "Drift", "Cinder", "Wisp", "Ember", "Shard", "Vessel", "Plume", "Quill", "Ash", "Fleck" };

    for (var i = 0; i < wanted && bots.Count < World.Crowd; i++)
    {
        var spot = Somewhere();

        var bot = new Player
        {
            Id = NewId(),
            Name = names[random.Next(names.Length)] + " " + (char)('A' + random.Next(26)),
            Colour = World.ColourFor(Interlocked.Increment(ref seat)),
            X = spot.X,
            Y = spot.Y,
            Radius = World.Start * (1 + random.NextDouble()),
            Bot = true
        };

        bot.WantX = bot.AimX = bot.X;
        bot.WantY = bot.AimY = bot.Y;

        bots[bot.Id] = bot;
    }
}

void Step(double seconds)
{
    var everyone = Souls();

    tick++;

    at += seconds;

    /*
     * Everything that is only true for one tick starts the tick empty.
     *
     * The broadcast that reads these runs after this returns, so a mark or a
     * death recorded below is sent exactly once and then forgotten - which
     * is the whole of their lifetime and the reason they are not state.
     */
    feasts.Clear();

    // blobs nobody came back for
    if (tick % 40 == 0 && !adrift.IsEmpty)
    {
        foreach (var (ticket, waiting) in adrift)
        {
            if ((DateTime.UtcNow - waiting.Since).TotalSeconds > World.Adrift)
            {
                adrift.TryRemove(ticket, out _);
            }
        }
    }

    foreach (var soul in everyone)
    {
        soul.Ate = null;

        if (soul.Sparks > 0 && (at >= soul.Spark || World.IsHole(soul.Radius)))
        {
            soul.Sparks = 0;
        }

        /*
         * A hole holds on always.
         *
         * It cannot move, cannot let go of anything, and the pull is the only
         * thing it does - so asking somebody to keep a key down for the rest
         * of the round in order to be what they already are is a control that
         * does nothing except wear out a finger. On a phone it is worse than
         * that: it is a control there is no finger spare for.
         */
        if (World.IsHole(soul.Radius))
        {
            soul.Hold = true;
            soul.Rest = 0;
        }
    }

    // the arena is sized for whoever is in it, once, before anything moves
    size = World.SizeFor(everyone.Length);

    /*
     * What every bot has decided, worked out on all of them at once.
     *
     * This is the whole of the tick at a crowd worth having: at ten thousand
     * it was a hundred and thirty five milliseconds of the hundred and
     * eighty eight, and the other four fifths of the machine sat watching
     * one core do it.
     *
     * It parallelises because a bot shares nothing: it reads the two grids,
     * which nothing writes until this pass is over, and writes only its own
     * fields. The one thing that was not safe was the arena's random number
     * generator, which is a line further down. Chunked rather than one task
     * per bot - the body is microseconds.
     */
    if (bots.Count >= 200)
    {
        Parallel.ForEach(Partitioner.Create(0, everyone.Length), span =>
        {
            for (var i = span.Item1; i < span.Item2; i++)
            {
                if (everyone[i].Bot && !everyone[i].Gone)
                {
                    Think(everyone[i], seconds);
                }
            }
        });
    }
    else
    {
        foreach (var soul in everyone)
        {
            if (soul.Bot && !soul.Gone)
            {
                Think(soul, seconds);
            }
        }
    }

    foreach (var soul in everyone)
    {
        /*
         * A hole pulls the floor down around itself.
         *
         * Not for the food - it is already the biggest thing in the arena -
         * but for the traffic. Something that cannot move and cannot chase is
         * a hazard everybody walks around, and a round that nobody will
         * approach does not end. A floor thick with pellets is worth the risk
         * of going near one, which is what puts it back in the game.
         */
        if (!soul.Gone && World.IsHole(soul.Radius))
        {
            var owed = World.Summons(soul.Radius) * seconds;

            var many = (int)owed + (random.NextDouble() < owed - (int)owed ? 1 : 0);

            // capped against the floor it is adding to, or a grade five
            // summoning a hundred and fifty a second with nobody coming to
            // eat them fills the arena and never stops
            var room = (int)(World.PelletsFor(size) * World.Glut * 2);

            for (var i = 0; i < many && pellets.Count < room; i++)
            {
                var angle = random.NextDouble() * Math.Tau;
                var far = soul.Radius + Math.Sqrt(random.NextDouble()) * World.SummonSpan(soul.Radius);

                var slot = TakeSlot();

                /*
                 * Straight onto the floor. Drop is the other direction - it
                 * takes one off - so calling it with a pellet that had never
                 * been on there did nothing except hand the slot number
                 * straight back, and the floor never grew at all.
                 */
                pellets[slot] = new Pellet
                {
                    Slot = slot,
                    X = World.Clamp(soul.X + Math.Cos(angle) * far, size),
                    Y = World.Clamp(soul.Y + Math.Sin(angle) * far, size),
                    C = World.AnyTint(random),
                    Grade = 0
                };
            }
        }
    }

    // and the floor is made good and put into buckets before anybody walks
    // over it, so that the eating below and the broadcast afterwards are both
    // reading the same arena rather than one a tick apart
    Scatter();

    /*
     * The dust falls in as well. Anything with a dent drags the pellets
     * around it towards itself, which is what makes the arena look like it
     * is draining into the big blobs rather than merely bending around them.
     *
     * This runs against the buckets as they were at the end of the last tick
     * rather than rebuilding them first. A pellet moves at most three or four
     * units in a tick against buckets two hundred and sixty across, so being
     * one tick stale about which bucket a pellet is in costs nothing, and the
     * alternative is bucketing the whole floor twice every tick.
     */
    foreach (var heavy in everyone)
    {
        var grip = World.Grip(heavy.Radius, heavy.Hold, heavy.Wound);

        if (World.IsHole(heavy.Radius))
        {
            grip = Math.Max(grip, 1) * World.HolePull(heavy.Radius);
        }

        if (grip <= 0)
        {
            continue;
        }

        /*
         * Bounded by the arena, and it has to be.
         *
         * The sweep is the radius times the square root of the grip, and the
         * grip of a high grade hole is hundreds - which put the sweep at a
         * hundred and fifteen thousand units across a thirty six hundred
         * arena, and the buckets inside it at three quarters of a million per
         * tick. The server stopped sending frames at all. Nothing outside the
         * arena can be pulled anyway, so clamping costs nothing and is the
         * difference between a grade four working and the game hanging.
         */
        var sweep = Math.Min(Math.Min(size, World.Dredge),
                                 World.Drag(heavy.Radius, grip) * (World.IsHole(heavy.Radius) ? World.HoleSpan(heavy.Radius) : 1));

        foreach (var key in World.KeysWithin(heavy.X, heavy.Y, sweep))
        {
            if (!floor.TryGetValue(key, out var dust))
            {
                continue;
            }

            foreach (var pellet in dust)
            {
                var toX = heavy.X - pellet.X;
                var toY = heavy.Y - pellet.Y;

                var gap = Math.Sqrt(toX * toX + toY * toY);

                if (gap < 0.001)
                {
                    continue;
                }

                var speed = World.DriftSpeed(heavy.Radius, gap, grip, heavy.Hold);

                if (speed <= 0)
                {
                    continue;
                }

                var step = Math.Min(speed * seconds, gap);

                pellet.X = World.Clamp(pellet.X + toX / gap * step, size);
                pellet.Y = World.Clamp(pellet.Y + toY / gap * step, size);
            }
        }
    }

    Reherd(everyone);

    Weigh(everyone);

    Rebucket();

    /*
     * How long everybody holding on has been holding on, which is what the
     * grip is computed from. Done before the pulls are worked out, so that
     * the grip is this tick's rather than last tick's.
     *
     * It used to cost area, and this loop used to be the accounting for
     * that: what each holder burned, whether it had burned them down to
     * nothing, and what the floor was owed back. None of that survived
     * holding on becoming free, and what was left of it was arithmetic on a
     * quantity fixed at zero.
     */
    foreach (var player in everyone)
    {
        if (player.Hold && player.Rest <= 0)
        {
            player.Wound += seconds;
        }
        else
        {
            // letting go starts the wait, whether it was let go of or ran out
            if (player.Wound > 0)
            {
                player.Rest = World.Rest;
            }

            player.Wound = 0;
            player.Rest = Math.Max(0, player.Rest - seconds);
        }
    }

    Whirls(everyone, seconds);

    /*
     * What everybody is being dragged towards, worked out for all of them
     * from where they all are now. It has to happen before anybody moves, or
     * the order the dictionary happens to hand players back in would decide
     * who gets pulled towards where somebody was and who towards where they
     * ended up.
     */
    foreach (var player in everyone)
    {
        player.PullX = 0;
        player.PullY = 0;
    }

    foreach (var heavy in everyone)
    {
        var grip = World.Grip(heavy.Radius, heavy.Hold, heavy.Wound);

        // a collapsed blob is not a big player, it is weather: it pulls what
        // the rest of the rules would have given it, several times over and
        // compounding with its grade, from much further away
        if (World.IsHole(heavy.Radius))
        {
            grip = Math.Max(grip, 1) * World.HolePull(heavy.Radius);
        }

        if (grip <= 0)
        {
            continue;
        }

        // the same bound, for the same reason: this one is compared against
        // rather than walked, but a reach larger than the arena means every
        // blob in it is inside every other one's pull, which is not a game
        var reach = Math.Min(Math.Min(size, World.Dredge),
                                 World.Reach(heavy.Radius, grip) * (World.IsHole(heavy.Radius) ? World.HoleSpan(heavy.Radius) : 1));

        foreach (var player in Around(heavy.X, heavy.Y, reach, everyone))
        {
            if (ReferenceEquals(player, heavy) || player.Gone)
            {
                continue;
            }

            var toX = heavy.X - player.X;
            var toY = heavy.Y - player.Y;

            if (Math.Abs(toX) >= reach || Math.Abs(toY) >= reach)
            {
                continue;
            }

            var gap = Math.Sqrt(toX * toX + toY * toY);

            if (gap < 0.001)
            {
                continue;
            }

            var speed = World.PullSpeed(player.Radius, heavy.Radius, gap, grip);

            if (speed <= 0)
            {
                continue;
            }

            player.PullX += toX / gap * speed;
            player.PullY += toY / gap * speed;

            // being dragged by somebody holding on loosens the cap below
            if (heavy.Hold)
            {
                player.Clutched = true;
            }
        }
    }

    foreach (var player in everyone)
    {
        var dx = player.WantX - player.X;
        var dy = player.WantY - player.Y;

        var distance = Math.Sqrt(dx * dx + dy * dy);

        var goX = 0.0;
        var goY = 0.0;

        /*
         * Running a drawn line.
         *
         * It replaces steering rather than adding to it: the line was chosen
         * in advance and the whole point is that it is taken exactly, so
         * whatever the pointer is doing meanwhile is ignored until it ends.
         */
        if (player.WarpLands > 0)
        {
            Dash(player, seconds);
        }
        else if (player.StreakX != null)
        {
            Run(player, seconds);
        }
        // a hole has no legs. It is pulled by nothing and pulls everything,
        // and the only way it moves is if something bigger turns up
        else if (World.IsHole(player.Radius))
        {
            goX = goY = 0;
        }
        // holding on roots you: the pull still moves you, your own legs do not
        else if (distance > 1 && !(player.Hold && player.Rest <= 0))
        {
            var step = Math.Min(World.Legs(player.Radius, player.Sparks) * seconds, distance);

            goX = dx / distance * step;
            goY = dy / distance * step;
        }

        /*
         * However much everything bigger adds up to, it can only ever take
         * the same fraction of this player's own speed away. Capping the sum
         * rather than each puller is the point: two giants either side would
         * otherwise add up to more than anybody can walk, and whoever was
         * between them would simply stop.
         */
        var drag = Math.Sqrt(player.PullX * player.PullX + player.PullY * player.PullY);

        if (drag > 0)
        {
            var most = (player.Clutched ? World.Clutch : World.Hand) * World.Legs(player.Radius, player.Sparks);

            var scale = drag > most ? most / drag : 1;

            goX += player.PullX * scale * seconds;
            goY += player.PullY * scale * seconds;
        }

        player.X = World.Clamp(player.X + goX, size);
        player.Y = World.Clamp(player.Y + goY, size);

        player.Clutched = false;

        // pellets first: growing before the fight is resolved is what makes a
        // near miss feel like a near miss
        foreach (var key in World.KeysWithin(player.X, player.Y, player.Radius + World.Girth(World.Grades.Length - 1)))
        {
            if (!floor.TryGetValue(key, out var bucket))
            {
                continue;
            }

            foreach (var pellet in bucket)
            {
                if (!World.Reaches(player.X, player.Y, player.Radius, pellet.X, pellet.Y, pellet.Grade))
                {
                    continue;
                }

                if (pellets.TryRemove(pellet.Slot, out _))
                {
                    spare.Enqueue((pellet.Slot, tick));

                    player.Radius = World.RadiusOf(World.AreaOf(player.Radius) + World.Worth(pellet.Grade));
                }
            }
        }
    }

    foreach (var player in everyone)
    {
        if (player.Gone)
        {
            continue;
        }

        /*
         * Only what is close enough to be swallowed, rather than everybody.
         *
         * Eating needs the other one's middle inside this one, so the square
         * asked for is this blob's own radius - which for the ten thousand
         * blobs that are the size of a pellet is one square, and for the one
         * that is a quarter of the arena wide is the arena.
         */
        foreach (var other in Around(player.X, player.Y, player.Radius, everyone))
        {
            if (ReferenceEquals(player, other) || other.Gone || player.Gone)
            {
                continue;
            }

            if (!World.Eats(player.Radius, other.Radius) || !World.Overlaps(player.X, player.Y, player.Radius, other.X, other.Y))
            {
                continue;
            }

            player.Radius = World.RadiusOf(World.AreaOf(player.Radius) + World.AreaOf(other.Radius) * World.Spoils);

            /*
             * A kill charges you. The teleport comes back at once - the
             * moment after a kill is the moment you most want to be
             * elsewhere, since everything that was circling the fight is now
             * circling you - and each kill taken while the last is still lit
             * stacks. It compounds deliberately: the way to stop a run is to
             * not be the next one, which is a decision people get to make
             * because the charge is drawn for everybody to see.
             */
            player.WarpReady = at;

            // a hole gets neither: it has nowhere to teleport from and no
            // legs to be made faster, so the charge would be lightning drawn
            // around something that cannot use it
            if (!World.IsHole(player.Radius))
            {
                player.Sparks = at < player.Spark ? Math.Min(World.SparkMost, player.Sparks + 1) : 1;

                player.Spark = at + World.SparkTime;
            }

            feasts.Add(new Feast(Math.Round(other.X, 1), Math.Round(other.Y, 1),
                                 Math.Round(other.Radius, 1), other.Colour));

            (player.Ate ??= []).Add(other.Name);

            other.Gone = true;
            other.EatenBy = player.Name;
        }
    }

    /*
     * Somebody taking the whole arena ends the round for everybody. It is
     * done here, after the eating, so that the blob that just swallowed its
     * way over the line is the one that gets the credit.
     */
    foreach (var player in everyone)
    {
        if (player.Gone || !World.HasWon(player.Radius))
        {
            continue;
        }

        winner = new Won("won", player.Name, World.ScoreOf(player.Radius));

        Remember(player);

        // everybody starts again, including whoever just won - the alternative
        // is an arena that stops, and a game nobody can rejoin without
        // reloading the page
        /*
         * Everybody starts again in the middle third rather than anywhere at
         * all. Scattered over the whole arena, a round began with every blob
         * alone in a field it was far too small to see across - which read as
         * everybody else having vanished, and was the most common thing
         * anybody said about a reset.
         */
        foreach (var soul in everyone)
        {
            var spot = Together();

            soul.Radius = World.Start;
            soul.X = spot.X;
            soul.Y = spot.Y;
            soul.WantX = soul.AimX = spot.X;
            soul.WantY = soul.AimY = spot.Y;
            soul.Gone = false;
            soul.EatenBy = null;
            soul.Hold = false;
            soul.StreakX = null;
            soul.StreakY = null;
            soul.Wound = 0;
            soul.WarpReady = 0;
            soul.WarpLands = 0;
            soul.DrawReady = 0;
            soul.Sparks = 0;
            soul.Spark = 0;
            soul.WhirlEnds = 0;
        }

        pellets.Clear();

        break;
    }
}

/// <summary>Where every teleport still crossing is headed.</summary>
/// <remarks>
/// Gathered once for the whole arena rather than per viewer, because the
/// mark is the same mark whoever is looking at it - the filtering by what
/// each of them can see happens in <c>Picture</c>, on a list this short.
/// </remarks>
/// <summary>Where everybody is, small enough to send to everybody.</summary>
/// <remarks>
/// Built once for the whole arena rather than per viewer: a minimap showing
/// only what you can already see is a picture of your own screen. One number
/// for the blob - a size band below the collapse, a grade above - which is
/// all a dot a few pixels across can carry.
/// </remarks>
void Chart(Player[] everyone)
{
    if (tick % World.MapEvery != 0)
    {
        map = null;

        return;
    }

    /*
     * Every hole, and as many of everybody else as the budget allows.
     *
     * Ten thousand dots is a hundred and ninety kilobytes six times a second
     * to every player, for a picture a hundred and sixty pixels across on
     * which they would be four to a pixel. The holes are never thinned: they
     * are the thing the map is for.
     */
    var thin = Math.Max(1, (int)Math.Ceiling(everyone.Length / (double)World.Dots));

    map = new List<int>(Math.Min(everyone.Length, World.Dots + 64) * 3);

    for (var i = 0; i < everyone.Length; i++)
    {
        var soul = everyone[i];
        var hole = World.IsHole(soul.Radius);

        if (!hole && thin > 1 && i % thin != 0)
        {
            continue;
        }

        map.Add((int)soul.X);
        map.Add((int)soul.Y);
        map.Add(hole ? 3 + World.HoleLevel(soul.Radius) : World.Tier(soul.Radius));
    }
}

void Marks(Player[] everyone)
{
    marks.Clear();

    foreach (var soul in everyone)
    {
        if (soul.WarpLands <= 0 || soul.Gone)
        {
            continue;
        }

        var through = Math.Clamp(1 - (soul.WarpLands - at) / World.WarpLead, 0, 1);

        marks.Add(new Mark(Math.Round(soul.WarpToX, 1), Math.Round(soul.WarpToY, 1),
                           Math.Round(soul.Radius, 1), soul.Colour, Math.Round(through, 2), soul.Id));
    }
}

string Picture(Player viewer, Blob[] blobs, Score[] board, int alive, Player[] turning)
{
    List<Blob> near = [];

    foreach (var blob in blobs)
    {
        if (World.InView(viewer.X, viewer.Y, viewer.Radius, blob.X, blob.Y))
        {
            near.Add(blob);

            continue;
        }

        // something heavy enough to bend the floor is sent even when it is
        // itself past the edge of the screen, because the bend it makes is
        // not - and a bend that appeared only once its cause did would look
        // like the floor snapping rather than like anything being there
        if (blob.G > 0)
        {
            var edge = World.ViewOf(viewer.Radius) + World.Reach(blob.R, blob.G);

            if (Math.Abs(blob.X - viewer.X) < edge && Math.Abs(blob.Y - viewer.Y) < edge)
            {
                near.Add(blob);
            }
        }
    }

    /*
     * The same budget the floor has, for the same reason: a hole sees the
     * whole arena, and at a full house that is ten thousand blobs. Anything
     * heavy enough to bend the floor is kept whatever happens - a dent whose
     * cause was thinned away looks like the floor is broken - and the rest
     * are thinned evenly.
     */
    if (near.Count > World.Faces)
    {
        var every = (int)Math.Ceiling(near.Count / (double)World.Faces);

        var kept = new List<Blob>(World.Faces + 64);

        for (var i = 0; i < near.Count; i++)
        {
            if (near[i].G > 0 || i % every == 0)
            {
                kept.Add(near[i]);
            }
        }

        near = kept;
    }

    // whole units. A pellet does move now, but only ever towards something,
    // and half a unit of rounding on a dot five units wide is well under a
    // pixel at any zoom the game is played at
    var crumbs = new List<int>();

    /*
     * How much of the floor to leave out, worked out from how much of it is
     * in front of this viewer before any of it is gathered.
     */
    var sighted = 0;

    foreach (var key in World.KeysWithin(viewer.X, viewer.Y, World.ViewOf(viewer.Radius)))
    {
        if (floor.TryGetValue(key, out var counted))
        {
            sighted += counted.Count;
        }
    }

    /*
     * Thinned by distance rather than by slot number.
     *
     * It used to keep every nth pellet, n being the sighted count over the
     * budget. Deterministic, which was the point - but n is a whole number
     * computed from a count that changes every tick, so whenever the count
     * sat near a multiple of the budget, n flipped between one and two and
     * the kept set changed almost entirely. Measured: five hundred and
     * ninety nine of six hundred and fourteen pellets vanishing between one
     * frame and the next, twenty times a second. That is the flicker.
     *
     * So keep a fixed share of them instead, chosen by scrambling the slot
     * number and comparing it against a threshold. That has the property the
     * modulo lacked: it is monotone. Lifting the share from a third to a
     * third and a bit keeps everything it was already keeping and adds a few,
     * so a share that drifts makes the floor thicken and thin rather than
     * change its mind. Easing the share as well means it drifts even when the
     * count does not.
     *
     * Not a radius, which was the first thing tried. A hole hauls the floor
     * into itself, so the pellets near it are dense and the ones further out
     * are what it is visibly dragging - and the radius holding the budget
     * collapsed to inside the hole's own mouth, spending every pellet on the
     * ones about to be eaten and drawing nothing anywhere else.
     */
    var want = sighted > World.Crumbs ? World.Crumbs / (double)sighted : 1.0;

    viewer.Share = viewer.Share <= 0 ? want : viewer.Share * 0.85 + want * 0.15;

    var share = Math.Clamp(viewer.Share, 0.02, 1.0);

    var gate = (uint)(share * uint.MaxValue);

    foreach (var key in World.KeysWithin(viewer.X, viewer.Y, World.ViewOf(viewer.Radius)))
    {
        if (!floor.TryGetValue(key, out var bucket))
        {
            continue;
        }

        foreach (var pellet in bucket)
        {
            if (World.InView(viewer.X, viewer.Y, viewer.Radius, pellet.X, pellet.Y))
            {
                if (share < 1.0 && World.Scramble(pellet.Slot) > gate)
                {
                    continue;
                }

                crumbs.Add(pellet.Slot);
                crumbs.Add((int)pellet.X);
                crumbs.Add((int)pellet.Y);

                // colour in the low four bits, grade above them
                crumbs.Add(pellet.C | (pellet.Grade << 4));
            }
        }
    }

    /*
     * The marks and the deaths, cut down to what this screen reaches.
     *
     * Null rather than an empty list when there is nothing, which on the
     * overwhelming majority of ticks is the case: the serialiser leaves a
     * null field out of the frame altogether.
     */
    List<Mark> crossing = null;

    foreach (var mark in marks)
    {
        if (World.InView(viewer.X, viewer.Y, viewer.Radius, mark.X, mark.Y))
        {
            (crossing ??= []).Add(mark);
        }
    }

    List<Eddy> eddies = null;

    foreach (var hole in turning)
    {
        if (World.InView(viewer.X, viewer.Y, viewer.Radius, hole.WhirlX, hole.WhirlY))
        {
            var through = Math.Clamp(1 - (hole.WhirlEnds - at) / World.WhirlTime, 0, 1);

            (eddies ??= []).Add(new Eddy(Math.Round(hole.WhirlX, 1), Math.Round(hole.WhirlY, 1),
                                         World.WhirlReach, hole.Colour, Math.Round(through, 2)));
        }
    }

    List<Feast> gone = null;

    foreach (var feast in feasts)
    {
        if (World.InView(viewer.X, viewer.Y, viewer.Radius, feast.X, feast.Y))
        {
            (gone ??= []).Add(feast);
        }
    }

    return JsonSerializer.Serialize(
        new Frame("f", near, crumbs, board.ToList(), alive, size,
                  Math.Round(Math.Max(0, viewer.WarpReady - at), 2),
                  Math.Round(Math.Max(0, viewer.DrawReady - at), 2), World.ScoreOf(viewer.Radius),
                  World.IsHole(viewer.Radius),
                  crossing, gone, viewer.Ate, eddies), json);
}

async ValueTask Broadcast()
{
    var everyone = players.ToArray();

    // drawn from everybody rather than from the sockets: a bot is in the
    // arena and has to be seen, it simply has nowhere to be sent to
    var souls = Souls().Where(p => !p.Gone).ToArray();

    Marks(souls);

    Chart(souls);

    // short, and usually empty: one per hole with a whirl open
    var turning = souls.Where(p => p.WhirlEnds > 0).ToArray();

    var blobs = souls.Select(p => new Blob(p.Id, p.Name, World.TintOf(p.Colour),
                                           Math.Round(p.X, 1), Math.Round(p.Y, 1),
                                           Math.Round(p.Radius, 1),
                                           Math.Round(World.IsHole(p.Radius)
                                                      ? World.Grip(p.Radius, p.Hold, p.Wound) * World.HolePull(p.Radius)
                                                      : World.Grip(p.Radius, p.Hold, p.Wound), 2),
                                           p.Sparks > 0 ? p.Sparks : null))
                     .ToArray();

    var board = souls.OrderByDescending(p => p.Radius)
                     .Take(8)
                     .Select(p => new Score(p.Name, World.ScoreOf(p.Radius)))
                     .ToArray();

    // a bot that was eaten starts again rather than leaving the arena emptier
    // than whoever asked for company wanted it
    foreach (var (id, bot) in bots)
    {
        if (!bot.Gone)
        {
            continue;
        }

        var spot = Somewhere();

        bot.X = bot.WantX = bot.AimX = spot.X;
        bot.Y = bot.WantY = bot.AimY = spot.Y;
        bot.Radius = World.Start;
        bot.Gone = false;
        bot.EatenBy = null;
        bot.Hold = false;
        bot.StreakX = null;
        bot.StreakY = null;
        bot.Wound = 0;
    }

    // whoever went this tick is told so and dropped. It touches the shared
    // state and happens to a handful of people at most, so it stays here
    // rather than joining the fan out below.
    foreach (var (connection, player) in everyone)
    {
        if (!player.Gone)
        {
            continue;
        }

        await Send(connection, JsonSerializer.Serialize(
            new Eaten("eaten", player.EatenBy ?? "somebody", World.ScoreOf(player.Radius)), json));

        Remember(player);

        players.TryRemove(connection, out _);
    }

    /*
     * What anybody said goes to everybody, ahead of the frames, and is then
     * forgotten. Nothing about chat is filtered by distance: an arena where
     * you can only talk to what you can already eat is not one people talk
     * in.
     */
    if (!chatter.IsEmpty)
    {
        var lines = new List<string>();

        while (chatter.TryDequeue(out var said))
        {
            lines.Add(JsonSerializer.Serialize(said, json));
        }

        foreach (var (connection, _) in everyone)
        {
            foreach (var line in lines)
            {
                await Send(connection, line);
            }
        }
    }

    /*
     * A round ending is told to everybody at once, ahead of the frame that
     * shows the arena it has already been reset to. Sending it first is what
     * lets the browser put the result up over a world that is visibly
     * starting again rather than over a frozen one.
     */
    if (winner != null)
    {
        var news = JsonSerializer.Serialize(winner, json);

        winner = null;

        foreach (var (connection, _) in everyone)
        {
            await Send(connection, news);
        }
    }

    var living = everyone.Where(e => !e.Value.Gone).ToArray();

    if (living.Length == 0)
    {
        return;
    }

    /*
     * The map, written once and handed to everybody.
     *
     * It is the only thing in what a player receives that does not depend on
     * who they are, so serialising it inside each of five hundred frames was
     * five hundred copies of one picture, twenty times a second.
     */
    if (map != null)
    {
        var picture = JsonSerializer.Serialize(new Chart("chart", map), json);

        foreach (var (connection, _) in living)
        {
            await Send(connection, picture);
        }
    }

    /*
     * Everybody sees something different, so there is one of these per player
     * and together they are the whole tick. They are also completely
     * independent of each other, which is worth spending: a loop that builds
     * them one after another is one core no matter what the machine has, and
     * on a four core box that is the number that decides how many people fit
     * in the arena rather than anything about the socket.
     */
    var payloads = new string[living.Length];

    Parallel.For(0, living.Length,
                 i => payloads[i] = Picture(living[i].Value, blobs, board, blobs.Length, turning));

    var writes = new List<Task>(living.Length);

    for (var i = 0; i < living.Length; i++)
    {
        writes.Add(Send(living[i].Key, payloads[i]));
    }

    await Task.WhenAll(writes);
}

/// <summary>
/// Sends one payload, and forgets whoever is no longer there to receive it.
/// </summary>
/// <remarks>
/// A websocket connection is not thread safe, and GenHTTP says so: keeping
/// writes to one from colliding is the caller's job. This is that, per
/// connection rather than the single static lock the documentation's example
/// uses, so no two players are ever queued behind each other.
///
/// The other writer was the pong. Of the eight things a reactive handler can
/// be asked to do, two write by default - OnPing answers with a pong and
/// OnClose answers with a close - and both of them run on the read loop,
/// which is a different thread from the one broadcasting. This arena replaced
/// OnClose long ago and left OnPing alone, on the grounds that no ping had
/// been seen; that was an argument about traffic rather than about the code,
/// and traffic changes. A ping arriving while a frame is going out is two
/// flushes on a writer that permits one, which the engine reports as
/// "FlushAsync already in progress" and the player sees as a reconnect.
///
/// So every write to a connection goes through here now, the pong included.
/// A faulted write still costs that player their place rather than taking the
/// connection handler down with it.
/// </remarks>
Task<bool> Send(IReactiveConnection connection, string payload)
    => Guarded(connection, c => c.WritePayloadAsync(payload));

/// <summary>
/// Performs one write on a connection, alone.
/// </summary>
async Task<bool> Guarded(IReactiveConnection connection, Func<IReactiveConnection, ValueTask> write)
{
    var gate = gates.GetOrAdd(connection, _ => new SemaphoreSlim(1, 1));

    await gate.WaitAsync();

    var ours = true;

    try
    {
        var writing = write(connection).AsTask();

        // one socket that has stopped reading must not stop the arena
        if (!writing.IsCompleted && await Task.WhenAny(writing, Task.Delay(World.Patience)) != writing)
        {
            // the write is still out there somewhere, so its gate is never
            // handed back: nothing may touch this connection again
            ours = false;

            Note(connection, $"write did not finish within {World.Patience / 1000}s; abandoned and the gate kept");

            players.TryRemove(connection, out _);

            return false;
        }

        await writing;

        since.AddOrUpdate(connection, _ => (DateTime.UtcNow, 1), (_, was) => (was.At, was.Sent + 1));

        return true;
    }
    catch (Exception e)
    {
        /*
         * The kind and the message, because "the handler faulted" on its own
         * never said which of the several things that can go wrong did.
         *
         * Matched rather than reflected: GetType is not available here, and
         * rightly - this runs inside the guard that keeps a lambda from
         * reaching for the runtime. The four below are the ones a socket
         * actually produces; anything else is reported as what it says.
         */
        // the derived one first: ObjectDisposedException is an
        // InvalidOperationException, and the other order never reaches it
        var kind = e switch
        {
            ObjectDisposedException => "ObjectDisposedException",
            InvalidOperationException => "InvalidOperationException",
            OperationCanceledException => "OperationCanceledException",
            IOException => "IOException",
            _ => "Exception"
        };

        Note(connection, $"write failed — {kind}: {e.Message}");

        players.TryRemove(connection, out _);

        return false;
    }
    finally
    {
        if (ours)
        {
            gate.Release();
        }
    }
}

// the clock, started by whoever arrives to an empty arena and stopped by the
// last of them leaving: an empty arena should not cost a thread
void StartClock()
{
    if (Interlocked.Exchange(ref ticking, 1) == 1)
    {
        return;
    }

    _ = Task.Run(async () =>
    {
        var last = DateTime.UtcNow;

        /*
         * Fifty milliseconds between beats, not fifty milliseconds of doing
         * nothing between them.
         *
         * The wait used to be a flat delay after the work, so the period was
         * fifty plus the tick - harmless at a millisecond, and not at twenty,
         * where the rate had quietly fallen to fourteen a second. Subtracting
         * the work keeps it flat until the work will not fit at all, and then
         * it runs late rather than slow.
         */
        try
        {
            while (!players.IsEmpty)
            {
                var began = DateTime.UtcNow;

                var now = began;
                var seconds = Math.Min((now - last).TotalSeconds, 0.25);

                last = now;

                Step(seconds);

                await Broadcast();

                var spent = (DateTime.UtcNow - began).TotalMilliseconds;

                await Task.Delay(Math.Max(1, (int)(World.Beat * 1000 - spent)));
            }
        }
        finally
        {
            Interlocked.Exchange(ref ticking, 0);

            if (!players.IsEmpty)
            {
                StartClock();
            }
            else
            {
                // the clock has stopped with nobody here; whether the company
                // stays is decided by whoever arrives next, and how long after
                emptied = DateTime.UtcNow;
            }
        }
    });
}

var socket = Websocket.Functional()
                      .OnMessage(async (connection, message) =>
                      {
                          var text = await message.ReadPayloadAsync<string>() ?? "";

                          if (text.Length is 0 or > 300)
                          {
                              return;
                          }

                          if (!players.TryGetValue(connection, out var player))
                          {
                              if (!arriving.TryAdd(connection, 0))
                              {
                                  return;
                              }

                              try
                              {
                              var joining = JsonSerializer.Deserialize<Join>(text, json);

                              /*
                               * Only a join makes a player. This used to
                               * take whatever arrived, and a steer
                               * deserialises into a Join perfectly happily
                               * with no name at all - which the server then
                               * correctly called "a blob". Being eaten takes
                               * a player off the books and leaves the socket
                               * open, so the steer twenty milliseconds later
                               * put them back as one.
                               */
                              if (joining?.Kind != "join")
                              {
                                  return;
                              }

                              var spot = Somewhere();

                              /*
                               * A token from a welcome, still within the
                               * wait, gets that blob's size back. Anything
                               * else - no token, a stale one, one already
                               * claimed - starts over, so a guess is worth
                               * nothing and being eaten still costs you.
                               */
                              var back = joining?.Token is { Length: 16 } was
                                      && adrift.TryRemove(was, out var kept)
                                      && (DateTime.UtcNow - kept.Since).TotalSeconds <= World.Adrift
                                  ? kept
                                  : default;

                              var ticket = NewId() + NewId();

                              player = new Player
                              {
                                  Id = NewId(),
                                  Name = World.Called(joining?.Name),
                                  Colour = back.Colour ?? World.ColourFor(Interlocked.Increment(ref seat)),
                                  X = spot.X,
                                  Y = spot.Y,
                                  Radius = back.Radius > World.Start ? back.Radius : World.Start,
                                  Ticket = ticket
                              };

                              player.WantX = player.X;
                              player.WantY = player.Y;

                              /*
                               * The welcome goes out before they are on the
                               * books, and the order is the point: the clock
                               * writes to everybody on the books twenty times
                               * a second, and a second write while this one
                               * is going out is not survivable.
                               */
                              var greeted = await Send(connection, JsonSerializer.Serialize(
                                  new Welcome("welcome", player.Id, player.Colour, size, World.Palette, World.WellSize, World.Girths, ticket), json));

                              if (!greeted)
                              {
                                  return;
                              }

                              /*
                               * Company nobody came back for. Cleared on the
                               * way in rather than on the way out, so that a
                               * socket which drops and reconnects finds the
                               * arena it left rather than an empty one.
                               */
                              if (players.IsEmpty && !bots.IsEmpty
                                  && (DateTime.UtcNow - emptied).TotalSeconds > World.Linger)
                              {
                                  bots.Clear();
                              }

                              players[connection] = player;

                              StartClock();
                              }
                              finally
                              {
                                  arriving.TryRemove(connection, out _);
                              }

                              return;
                          }

                          if (text.Contains("\"say\""))
                          {
                              var spoken = JsonSerializer.Deserialize<Say>(text, json);

                              /*
                               * Trimmed, capped, stripped of anything that is
                               * not printable, and refused if it arrives too
                               * soon after the last one. None of that is about
                               * taste - it is about a line being a line: a
                               * message with newlines in it or three hundred
                               * characters of nothing is a way of pushing the
                               * arena off somebody else's screen.
                               */
                              var line = (spoken?.Text ?? "").Trim();

                              if (line.Length == 0 || at - player.Spoke < World.SayGap)
                              {
                                  return;
                              }

                              if (line.Length > World.SayLength)
                              {
                                  line = line[..World.SayLength];
                              }

                              // a space rather than nothing, or "line\nbreak"
                              // arrives as "linebreak"
                              line = new string(line.Select(c => char.IsControl(c) ? ' ' : c).ToArray()).Trim();

                              if (line.Length == 0)
                              {
                                  return;
                              }

                              player.Spoke = at;

                              chatter.Enqueue(new Said("said", player.Name, line));

                              return;
                          }

                          if (text.Contains("\"haul\""))
                          {
                              var wants = JsonSerializer.Deserialize<Haul>(text, json);

                              if (wants != null)
                              {
                                  Open(player, wants.X, wants.Y);
                              }

                              return;
                          }

                          if (text.Contains("\"warp\""))
                          {
                              var wants = JsonSerializer.Deserialize<Warp>(text, json);

                              if (wants != null)
                              {
                                  Jump(player, wants.X, wants.Y);
                              }

                              return;
                          }

                          if (text.Contains("\"path\""))
                          {
                              /*
                               * A drawn line, spent on arrival rather than on
                               * the press that started it: somebody who draws
                               * and then changes their mind has not spent
                               * their wait on a gesture that never finished.
                               */
                              var drawn = JsonSerializer.Deserialize<Path>(text, json);

                              if (at >= player.DrawReady && !World.IsHole(player.Radius) && player.StreakX == null
                                  && player.WarpLands <= 0 && drawn?.X is { Length: > 1 } && drawn.Y?.Length == drawn.X.Length)
                              {
                                  var count = Math.Min(drawn.X.Length, World.StreakPoints);

                                  var xs = new double[count];
                                  var ys = new double[count];

                                  var run = 0.0;
                                  var kept = 0;

                                  for (var i = 0; i < count; i++)
                                  {
                                      var px = World.Clamp(drawn.X[i], size);
                                      var py = World.Clamp(drawn.Y[i], size);

                                      if (kept > 0)
                                      {
                                          run += Math.Sqrt((px - xs[kept - 1]) * (px - xs[kept - 1])
                                                         + (py - ys[kept - 1]) * (py - ys[kept - 1]));

                                          // a line longer than the rules allow
                                          // is cut, not refused: the drawing
                                          // was honest, it just went too far
                                          if (run > World.StreakLength)
                                          {
                                              break;
                                          }
                                      }

                                      xs[kept] = px;
                                      ys[kept] = py;
                                      kept++;
                                  }

                                  if (kept > 1)
                                  {
                                      player.StreakX = xs[..kept];
                                      player.StreakY = ys[..kept];
                                      player.StreakLeg = 0;
                                      player.StreakGone = 0;
                                      player.Hold = false;
                                      player.DrawReady = at + World.DrawWait;
                                  }
                              }

                              return;
                          }

                          if (text.Contains("\"fill\""))
                          {
                              var asked = JsonSerializer.Deserialize<Fill>(text, json);

                              Populate(Math.Clamp(asked?.Count ?? 10, 1, World.Crowd));

                              return;
                          }

                          var steer = JsonSerializer.Deserialize<Steer>(text, json);

                          if (steer == null)
                          {
                              return;
                          }

                          player.WantX = World.Clamp(steer.X, size);
                          player.WantY = World.Clamp(steer.Y, size);
                          player.Hold = steer.Hold;
                      })
                      .OnError((connection, error) =>
                      {
                          /*
                           * Says what the default would have swallowed.
                           *
                           * Left alone, a frame error returns true, the read
                           * loop calls OnClose and breaks, and what reaches
                           * the log is a player leaving - identical to one who
                           * closed the tab. Which of those it was is the first
                           * question when sockets start dropping, and until
                           * now the arena could not answer it.
                           *
                           * Still returns true: the behaviour is unchanged,
                           * only now it is visible.
                           */
                          Note(connection, $"read error — {error.ErrorType}: {error.Message}");

                          return ValueTask.FromResult(true);
                      })
                      .OnPing(async (connection, message) =>
                      {
                          /*
                           * The pong the framework would send for us, sent
                           * through the same gate as everything else.
                           *
                           * Left to the default it goes out from the read
                           * loop while the broadcast may be halfway through a
                           * frame to the same socket, and the engine's writer
                           * takes one flush at a time. Copied first because
                           * the frame's buffer belongs to the reader and is
                           * handed back the moment this returns - and this
                           * may wait on the gate before it writes. A ping
                           * carries at most a hundred and twenty five bytes,
                           * so the copy costs nothing worth counting.
                           */
                          var echo = message.Data.ToArray();

                          await Guarded(connection, c => c.PongAsync(echo));
                      })
                      .OnClose((connection, _) =>
                      {
                          since.TryRemove(connection, out (DateTime, int) _);

                          if (players.TryRemove(connection, out var leaving))
                          {
                              Remember(leaving);

                              /*
                               * Parked rather than discarded. The socket may
                               * have been faulted rather than closed, and a
                               * browser that reconnects in a second should
                               * find the blob it had, not a new one.
                               */
                              if (!leaving.Gone && leaving.Radius > World.Start && leaving.Ticket != null)
                              {
                                  adrift[leaving.Ticket] = (leaving.Radius, leaving.Colour, DateTime.UtcNow);
                              }
                          }

                          if (players.IsEmpty)
                          {
                              emptied = DateTime.UtcNow;
                          }

                          // not disposed: a write may still be holding it.
                          // Named rather than discarded because the close
                          // handler already has a parameter called _
                          gates.TryRemove(connection, out var gate);

                          return ValueTask.CompletedTask;
                      });

var api = Inline.Create()
                .Get("best", () =>
                {
                    lock (writing)
                    {
                        return best.ToList();
                    }
                })
                .Get("here", () => new Here(players.Count, bots.Count, pellets.Count, tick));

return Layout.Create()
             .Add("api", api)
             .Add("play", socket)
             .Add(Assets.App());

// Ticks, because the difference between an arena that has stopped and one
// that is running but not reaching you is two requests, and without it is an
// afternoon.
record Here(int Players, int Bots, int Pellets, long Ticks);
