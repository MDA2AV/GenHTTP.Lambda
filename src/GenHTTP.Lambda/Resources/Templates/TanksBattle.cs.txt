// The rules of the fight: how fast a tank drives, what a shell does, and what
// driving over a crate gives you.
//
// All of it decided here, on the server. The browser draws what it is told and
// sends what its keys are doing; it is never asked where it is.

static class Battle
{
    /// <summary>Metres a second at full throttle.</summary>
    public const double Speed = 7.5;

    /// <summary>Radians a second at full lock.</summary>
    public const double Turn = 2.4;

    /// <summary>Metres a second for a shell, which has to outrun a tank.</summary>
    public const double ShellSpeed = 26.0;

    /// <summary>How long a shell lives before it gives up, in seconds.</summary>
    public const double ShellLife = 2.2;

    /// <summary>Seconds between shots.</summary>
    public const double Reload = 0.9;

    /// <summary>Seconds between shots while a rapid crate is running.</summary>
    public const double FastReload = 0.28;

    /// <summary>How long a crate's effect lasts.</summary>
    public const double BoostTime = 8.0;

    /// <summary>Hits a tank takes before it goes up.</summary>
    public const int Health = 3;

    /// <summary>Seconds on the floor before coming back.</summary>
    public const double Respawn = 3.0;

    /// <summary>How close a shell has to pass to count as a hit.</summary>
    public const double HitRadius = 1.2;

    /// <summary>How close is close enough to pick a crate up.</summary>
    public const double PickupRadius = 1.5;

    /// <summary>How many crates are out at once.</summary>
    public const int Crates = 5;

    private static readonly string[] Palette =
    [
        "#8ab4f8", "#f28b82", "#fdd663", "#81c995", "#c58af9", "#78d9ec", "#ff8bcb", "#ffa26b"
    ];

    /// <summary>The sorts of crate there are.</summary>
    public static readonly string[] Sorts = ["repair", "rapid", "shield"];

    public static string ColourFor(int seat) => Palette[seat % Palette.Length];

    /// <summary>A name that can be shown, or one for whoever gave none.</summary>
    public static string Called(string name)
    {
        var trimmed = (name ?? "").Trim();

        if (trimmed.Length == 0)
        {
            return "recruit";
        }

        return trimmed.Length > 14 ? trimmed[..14] : trimmed;
    }

    /// <summary>
    /// Where a tank ends up after a moment of driving.
    /// </summary>
    /// <remarks>
    /// The two axes are tried separately when the pair of them is blocked, so
    /// a tank held against a wall slides along it instead of stopping dead.
    /// Driving into a corner at an angle and coming to a halt feels like a
    /// bug even when it is arithmetic.
    /// </remarks>
    public static (double X, double Z, double Heading) Drive(
        bool[,] solid, double x, double z, double heading, double throttle, double steer, double seconds)
    {
        heading += Math.Clamp(steer, -1, 1) * Turn * seconds;

        var distance = Math.Clamp(throttle, -1, 1) * Speed * seconds;

        var dx = Math.Sin(heading) * distance;
        var dz = Math.Cos(heading) * distance;

        if (Maze.Fits(solid, x + dx, z + dz))
        {
            return (x + dx, z + dz, heading);
        }

        if (Maze.Fits(solid, x + dx, z))
        {
            return (x + dx, z, heading);
        }

        if (Maze.Fits(solid, x, z + dz))
        {
            return (x, z + dz, heading);
        }

        return (x, z, heading);
    }

    /// <summary>Whether two points are within a distance of each other.</summary>
    public static bool Within(double ax, double az, double bx, double bz, double distance)
    {
        var dx = ax - bx;
        var dz = az - bz;

        return dx * dx + dz * dz <= distance * distance;
    }

    /// <summary>What a crate does when it is driven over.</summary>
    public static string Effect(string sort) => sort switch
    {
        "repair" => "patched up",
        "rapid" => "loading fast",
        _ => "shielded"
    };
}
