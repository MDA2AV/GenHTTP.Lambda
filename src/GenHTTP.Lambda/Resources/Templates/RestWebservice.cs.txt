// The same idea as the inline service, written as a class. Handy once a route
// needs more than an expression, and the attributes read like documentation.
//
//   GET    /lambda/{{KEY}}/api/       every note
//   GET    /lambda/{{KEY}}/api/1      a single one
//   POST   /lambda/{{KEY}}/api/       add one, with a JSON body
//   DELETE /lambda/{{KEY}}/api/1      remove one
//
//   /lambda/{{KEY}}/openapi.json      the generated specification
//   /lambda/{{KEY}}/scalar/           Scalar, to try it out right away

return Layout.Create()
             .AddService<NoteService>("api")
             .AddOpenApi()
             .AddScalar();

public class NoteService
{
    private static readonly ConcurrentDictionary<int, Note> Notes = new()
    {
        [1] = new Note(1, "Write a lambda")
    };

    private static int _sequence = 1;

    [ResourceMethod]
    public IEnumerable<Note> Index() => Notes.Values.OrderBy(n => n.Id);

    [ResourceMethod(":id")]
    public Note Single(int id) => Notes.TryGetValue(id, out var note)
        ? note
        : throw new ProviderException(ResponseStatus.NotFound, $"There is no note with ID {id}.");

    [ResourceMethod(Method.Post)]
    public Note Add(Note note)
    {
        var created = note with { Id = Interlocked.Increment(ref _sequence) };

        Notes[created.Id] = created;

        return created;
    }

    [ResourceMethod(Method.Delete, ":id")]
    public void Remove(int id)
    {
        if (!Notes.TryRemove(id, out _))
        {
            throw new ProviderException(ResponseStatus.NotFound, $"There is no note with ID {id}.");
        }
    }
}

public record Note(int Id, string Text);
