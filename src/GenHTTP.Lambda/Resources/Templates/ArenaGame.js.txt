// The browser half. It draws what it is told and says where the pointer is;
// everything else - where anyone is, who ate whom, how big anybody got - is
// the server's, and this believes all of it.

const $ = (id) => document.getElementById(id);
const canvas = $('view');
const ctx = canvas.getContext('2d', { alpha: false });

/*
 * How far behind the newest thing the server said everything is drawn.
 *
 * The server speaks twenty times a second and the screen redraws sixty, so
 * two frames in three have nothing new in them. Rather than guess at what
 * happened in between, this keeps the last few snapshots and draws a fixed
 * moment in the past, reading every blob off between the two places it
 * actually was. The delay is what makes that possible: it is long enough
 * that the snapshot on the far side has almost always arrived by the time it
 * is wanted, so there is something to read between rather than nothing.
 *
 * A hundred milliseconds is two ticks and a margin. Shorter and a late
 * packet leaves nothing to interpolate towards, which shows up as a hitch;
 * much longer and steering starts to feel posted rather than driven.
 */
const DELAY = 100;

/** The last few snapshots, oldest first. */
const history = [];

let socket = null, me = null, world = 2400;
let palette = [], wellSize = 40;

/*
 * The floor is a sheet, and anything heavy enough to eat you sits in a dent
 * it has made in it. The dent is not decoration - it is the same pull the
 * server is applying to you, drawn.
 *
 * The depth is a quartic that reaches exactly as far as the pull does and
 * then is flatly zero, so a well can be rejected with one comparison and
 * there is no tail to fade out by hand. TILT is how much of that depth is
 * spent sliding things down the screen - deeper is lower, the way the foot
 * of a post is lower on screen than its top; FLOOR is as deep as the sheet
 * is ever drawn, so that two wells on top of each other bottom out rather
 * than falling through the floor.
 */
const TILT = 0.55;
const FLOOR = 400;

/*
 * How much the sheet is gathered inwards as well as pushed down.
 *
 * Sag on its own bends the lines that run across the screen and leaves the
 * ones running up it perfectly straight, because moving a point down does
 * not move it sideways - so it reads as a ripple rather than as a hole.
 * Pulling points in towards the well as well bends both sets and crowds the
 * squares together going in, which is what a funnel does to a grid.
 */
const GATHER = 0.5;

/** Where the player was when the last frame was drawn, which the camera and
 *  the steering both read so that the point under the cursor is the point
 *  that gets sent. */
let camera = null;

let pointer = { x: 0, y: 0 };

function size() {
  canvas.width = innerWidth * devicePixelRatio;
  canvas.height = innerHeight * devicePixelRatio;
  canvas.style.width = `${innerWidth}px`;
  canvas.style.height = `${innerHeight}px`;
}

addEventListener('resize', size);
size();

addEventListener('pointermove', (e) => { pointer = { x: e.clientX, y: e.clientY }; });

// the server is told where the pointer is in world units, twenty times a
// second - the same rate it answers at, because sending faster than it
// replies only builds a queue
setInterval(() => {
  if (socket?.readyState !== 1 || !camera) return;

  const scale = scaleFor(camera.r);
  const x = camera.x + (pointer.x - innerWidth / 2) / scale;
  const y = camera.y + (pointer.y - innerHeight / 2) / scale;

  socket.send(JSON.stringify({ kind: 'steer', x, y }));
}, 50);

/** How many pixels one world unit is. */
function scaleFor(radius) {
  return Math.min(innerWidth, innerHeight) / extentFor(radius);
}

/**
 * How much of the arena fits across the shorter side of the window.
 *
 * Bigger blobs see more, but a good deal less than their size would suggest.
 * Straight off the radius and a blob near the cap looks at nine thousand
 * units of arena through an eight hundred pixel window, where a starting
 * blob is two pixels across and the pellets are dust you cannot aim at. This
 * gets to about three thousand instead, which still fills the screen with
 * the arena and leaves everything in it a size worth steering at.
 *
 * The server keeps the same curve, because it decides what to send from it.
 */
function extentFor(radius) {
  return 260 + 224 * Math.pow(radius / 14, 0.7);
}

/** The blobs heavy enough to dent the floor, as the drawing wants them. */
function wellsOf(blobs) {
  const wells = [];

  for (const b of blobs) {
    if (b.r >= wellSize) {
      const reach = b.r * 4;

      // the dent stops deepening long before a blob stops growing. Past this
      // the well is already wider than the screen, and letting it go on would
      // gather everything visible into a single point rather than look like
      // anything - what keeps reading as bigger is the width, not the depth
      wells.push({ i: b.i, x: b.x, y: b.y, deep: Math.min(b.r * 2, 320), span: reach * reach });
    }
  }

  return wells;
}

/**
 * Where a point of the floor has moved to, and how far it has sunk.
 *
 * One pass over the wells doing both halves of the warp: in towards each
 * well, and down. `mine` leaves one well out, which is how a blob is given
 * the bend of everybody else's dent without being given its own.
 */
function sink(x, y, wells, mine) {
  let gx = x;
  let gy = y;
  let z = 0;

  for (const w of wells) {
    if (w.i === mine) continue;

    const dx = x - w.x;
    const dy = y - w.y;
    const d2 = dx * dx + dy * dy;

    if (d2 >= w.span) continue;

    // a quartic that reaches exactly as far as the pull does and is then
    // flatly zero, so a well is rejected with one comparison and there is
    // no tail left to fade out by hand
    const near = 1 - d2 / w.span;
    const dip = w.deep * near * near;

    z += dip;

    const d = Math.sqrt(d2);

    if (d > 0.001) {
      const inwards = Math.min(d * 0.85, dip * GATHER);

      gx -= dx / d * inwards;
      gy -= dy / d * inwards;
    }
  }

  return { x: gx, y: gy, z: FLOOR * Math.tanh(z / FLOOR) };
}

/**
 * Everything as it was DELAY milliseconds ago, read between the two
 * snapshots that bracket that moment.
 */
function moment() {
  if (history.length === 0) return null;

  const want = performance.now() - DELAY;

  // the newest pair whose older half is already behind us
  let i = history.length - 1;

  while (i > 0 && history[i - 1].at > want) i--;

  const after = history[i];
  const before = history[i - 1] ?? after;

  const span = after.at - before.at;
  const t = span > 0 ? Math.min(1, Math.max(0, (want - before.at) / span)) : 1;

  const was = new Map(before.b.map((b) => [b.i, b]));

  const blobs = after.b.map((b) => {
    const a = was.get(b.i);

    // somebody who has only just appeared has nowhere to come from
    if (!a) return b;

    return {
      i: b.i, n: b.n, c: b.c,
      x: a.x + (b.x - a.x) * t,
      y: a.y + (b.y - a.y) * t,
      r: a.r + (b.r - a.r) * t,
    };
  });

  /*
   * Pellets are read the same way now that they are dragged about: four
   * numbers each, the first of which says which pellet it is, so one seen in
   * both snapshots can be drawn between the two places it was. One that has
   * only just appeared is drawn where it is; one that has gone is simply not
   * in the newer half, and so is not drawn at all.
   */
  const dust = new Map();

  for (let i = 0; i < before.p.length; i += 4) dust.set(before.p[i], i);

  const pellets = [];

  for (let i = 0; i < after.p.length; i += 4) {
    const j = dust.get(after.p[i]);

    // a pellet is dragged at most a few units between two frames, so a slot
    // that appears to have crossed the arena is not the pellet it was - the
    // server leaves a freed slot alone long enough that this should never
    // fire, and drawing the streak if it ever did would be worse than not
    const jumped = j === undefined
                || Math.abs(after.p[i + 1] - before.p[j + 1]) > 40
                || Math.abs(after.p[i + 2] - before.p[j + 2]) > 40;

    if (jumped) {
      pellets.push(after.p[i + 1], after.p[i + 2], after.p[i + 3]);
    } else {
      pellets.push(
        before.p[j + 1] + (after.p[i + 1] - before.p[j + 1]) * t,
        before.p[j + 2] + (after.p[i + 2] - before.p[j + 2]) * t,
        after.p[i + 3]);
    }
  }

  return { blobs, pellets };
}

function draw() {
  requestAnimationFrame(draw);

  ctx.setTransform(devicePixelRatio, 0, 0, devicePixelRatio, 0, 0);
  ctx.fillStyle = '#0b0d11';
  ctx.fillRect(0, 0, innerWidth, innerHeight);

  const now = moment();

  if (!now) return;

  const mine = now.blobs.find((b) => b.i === me);

  if (!mine) return;

  // the camera sits on the same position the blob is drawn at, so the
  // player is exactly in the middle of the screen every frame rather than
  // swinging about it
  camera = mine;

  const scale = scaleFor(mine.r);
  const wells = wellsOf(now.blobs);

  /*
   * How deep the floor is under the player, which everything else is drawn
   * relative to. Without it, standing in a well would slide the whole world
   * off the screen instead of the floor rising around the player - which is
   * what standing in a hole actually looks like. Their own dent is left out,
   * so that growing does not drag the world out from under them.
   */
  const under = sink(mine.x, mine.y, wells, mine.i).z;

  const ox = innerWidth / 2 - mine.x * scale;
  const oy = innerHeight / 2 - mine.y * scale;

  /**
   * Where a point of the floor lands on the screen.
   *
   * For the floor only. Nothing that can be eaten goes through here - see
   * `flat` below, and the note on it, which is the whole reason this is two
   * functions rather than one.
   */
  const put = (x, y) => {
    const s = sink(x, y, wells);

    return {
      x: ox + s.x * scale,
      y: oy + s.y * scale + (s.z - under) * scale * TILT,
      z: s.z,
    };
  };

  /*
   * Where a thing standing on the floor lands on the screen: exactly where
   * it is, with none of the bending applied to it.
   *
   * This is not a shortcut, it is the point. The server decides who eats
   * whom from where things actually are, so if the drawing moved them the
   * two would disagree - and they disagree by a lot, because the gathering
   * is strongest exactly where the eating happens. Next to a blob of radius
   * ninety six, something eighty units too far away to be caught is drawn
   * comfortably inside it: you would aim at what you could see, miss, and
   * conclude the hit box was broken. It was; this is it fixed.
   *
   * The floor keeps bending underneath them, which is where the whole
   * effect lives anyway. A blob covers the deepest part of its own dent, so
   * what shows is the ring around it - and that reads as sitting in it.
   */
  const flat = (x, y) => ({ x: ox + x * scale, y: oy + y * scale });

  // the dark of each well, under the grid, so the lines read as lying on it
  for (const w of wells) {
    const at = put(w.x, w.y);
    const span = Math.sqrt(w.span) * scale;
    const sag = Math.min(w.deep, FLOOR) * scale * TILT;

    // the two focuses are offset, which tilts the cone for free and lights
    // it from the top of the screen without a second pass
    const shade = ctx.createRadialGradient(at.x, at.y, 0, at.x, at.y - sag * 0.45, span);

    shade.addColorStop(0, 'rgba(0,0,0,0.72)');
    shade.addColorStop(0.35, 'rgba(4,6,16,0.46)');
    shade.addColorStop(0.7, 'rgba(8,12,26,0.18)');
    shade.addColorStop(1, 'rgba(8,12,26,0)');

    ctx.fillStyle = shade;
    ctx.fillRect(0, 0, innerWidth, innerHeight);
  }

  drawSheet(put, scale,
            Math.max(0, mine.x - innerWidth / 2 / scale),
            Math.min(world, mine.x + innerWidth / 2 / scale),
            Math.max(0, mine.y - innerHeight / 2 / scale - FLOOR * TILT),
            Math.min(world, mine.y + innerHeight / 2 / scale + FLOOR * TILT),
            under);

  // pellets arrive flat - x, y, colour, x, y, colour - so they are walked
  // three at a time rather than unpacked into objects that live one frame
  ctx.globalAlpha = 0.75;

  const dot = Math.max(2, 5 * scale);
  const pellets = now.pellets;

  for (let i = 0; i < pellets.length; i += 3) {
    const at = flat(pellets[i], pellets[i + 1]);

    ctx.fillStyle = palette[pellets[i + 2]] ?? '#9aa0a6';
    ctx.beginPath();
    ctx.arc(at.x, at.y, dot, 0, Math.PI * 2);
    ctx.fill();
  }

  ctx.globalAlpha = 1;

  // smallest first, so being swallowed looks like being swallowed
  for (const b of [...now.blobs].sort((a, z) => a.r - z.r)) {
    const at = flat(b.x, b.y);

    const y = at.y;
    const r = b.r * scale;

    // a shadow on the sheet under it, which is what says it is resting on
    // something rather than floating above it
    ctx.fillStyle = 'rgba(0,0,0,0.32)';
    ctx.beginPath();
    ctx.ellipse(at.x, y + r * 0.2, r * 1.06, r * 0.36, 0, 0, Math.PI * 2);
    ctx.fill();

    // the body stays a circle whatever the sheet is doing, because how big
    // somebody is has to be readable at a glance and a squashed one is not
    ctx.fillStyle = b.c;
    ctx.beginPath();
    ctx.arc(at.x, y, r, 0, Math.PI * 2);
    ctx.fill();

    ctx.strokeStyle = b.i === me ? '#ffffff' : '#00000055';
    ctx.lineWidth = b.i === me ? 3 : 2;
    ctx.stroke();

    if (r > 14) {
      ctx.fillStyle = '#0b0d11';
      ctx.font = `${Math.min(r * 0.55, 22)}px -apple-system, system-ui, sans-serif`;
      ctx.textAlign = 'center';
      ctx.textBaseline = 'middle';
      ctx.fillText(b.n, at.x, y);
    }
  }
}

/**
 * The grid, as a sheet rather than as straight lines.
 *
 * Every line is walked in short steps and each step is put through the warp,
 * so a straight line through a well comes out as a curve and enough of them
 * together read as a surface. The steps are sized in screen pixels rather
 * than world units, so the cost is the same whether the player is small and
 * zoomed right in or huge and zoomed out.
 */
function drawSheet(put, scale, left, right, top, bottom, under) {
  const gap = scale < 0.7 ? 120 : 60;
  const step = Math.min(80, Math.max(16, 22 / scale));

  // four paths by depth, so the whole sheet is four strokes rather than one
  // per segment, and the deep parts can be fainter without a colour change
  const paths = [new Path2D(), new Path2D(), new Path2D(), new Path2D()];

  const run = (ax, ay, bx, by, into) => {
    const steps = Math.ceil(Math.hypot(bx - ax, by - ay) / step);

    let last = null;

    for (let i = 0; i <= steps; i++) {
      const x = ax + (bx - ax) * i / steps;
      const y = ay + (by - ay) * i / steps;
      const at = put(x, y);

      if (last) {
        const deep = Math.min(3, Math.max(0, at.z - under) / FLOOR * 4 | 0);

        (into ?? paths[deep]).moveTo(last.x, last.y);
        (into ?? paths[deep]).lineTo(at.x, at.y);
      }

      last = at;
    }
  };

  for (let x = Math.ceil(left / gap) * gap; x <= right; x += gap) run(x, top, x, bottom);
  for (let y = Math.ceil(top / gap) * gap; y <= bottom; y += gap) run(left, y, right, y);

  ctx.lineWidth = 1;

  const fade = [0.8, 0.58, 0.38, 0.22];

  for (let i = 3; i >= 0; i--) {
    ctx.strokeStyle = `rgba(54,64,82,${fade[i]})`;
    ctx.stroke(paths[i]);
  }

  // the edge of the arena, which warps with everything else so that it reads
  // as the rim of the same sheet rather than a frame laid over it
  const rim = new Path2D();

  if (top <= 0) run(left, 0, right, 0, rim);
  if (bottom >= world) run(left, world, right, world, rim);
  if (left <= 0) run(0, top, 0, bottom, rim);
  if (right >= world) run(world, top, world, bottom, rim);

  ctx.strokeStyle = '#46506a';
  ctx.lineWidth = 3;
  ctx.stroke(rim);
}

requestAnimationFrame(draw);

function connect(name) {
  const address = new URL('play', location.href.replace(/[^/]*$/, ''));
  address.protocol = address.protocol === 'https:' ? 'wss:' : 'ws:';

  socket = new WebSocket(address);
  socket.onopen = () => socket.send(JSON.stringify({ kind: 'join', name }));

  socket.onmessage = (event) => {
    const m = JSON.parse(event.data);

    if (m.kind === 'welcome') {
      me = m.id; world = m.world; palette = m.palette; wellSize = m.well ?? 40;
      history.length = 0; camera = null;
      $('gate').hidden = true; $('over').hidden = true;
      $('hud').hidden = false; $('board').hidden = false;
      return;
    }

    if (m.kind === 'won') {
      $('champion').textContent = `${m.who.replace(/[<&]/g, '')} took the arena`;
      $('tally').textContent = `${m.score.toLocaleString()} points.`;
      $('over').hidden = true;
      $('won').hidden = false;
      history.length = 0;
      return;
    }

    if (m.kind === 'eaten') {
      $('verdict').textContent = `Eaten by ${m.by}`;
      $('final').textContent = `You got to ${m.score}.`;
      $('over').hidden = false;
      camera = null;
      return;
    }

    world = m.w ?? world;

    history.push({ at: performance.now(), b: m.b, p: m.p });

    // a couple of seconds is far more than DELAY ever reaches back for, and
    // keeps this from growing for as long as somebody leaves the tab open
    while (history.length > 40) history.shift();

    const self = m.b.find((b) => b.i === me);

    $('score').textContent = self ? Math.round(Math.PI * self.r * self.r - Math.PI * 196) : 0;
    $('here').textContent = `${m.alive} in the arena`;
    $('list').innerHTML = m.l.map((s) =>
      `<li class="${self && s.n === self.n ? 'me' : ''}"><span>${s.n.replace(/[<&]/g, '')}</span><span>${s.s}</span></li>`).join('');
  };

  socket.onclose = () => { $('here').textContent = 'disconnected'; };
}

$('go').onclick = () => connect($('name').value);
$('again').onclick = () => connect($('name').value);

// a win does not put anybody out - the next round is already running
$('carry').onclick = () => { $('won').hidden = true; };
