// The browser half. It draws what it is told and says where the pointer is;
// everything else - where anyone is, who ate whom, how big anybody got - is
// the server's, and this believes all of it.

const ctx = canvas.getContext('2d', { alpha: false });

/*
 * How far behind the newest thing the server said everything is drawn.
 *
 * The server speaks twenty times a second and the screen redraws sixty, so
 * two frames in three have nothing new in them. Rather than guess at what
 * happened in between, this keeps the last few snapshots and draws a fixed
 * moment in the past, reading every blob off between the two places it
 * actually was. The delay is what makes that possible: it is long enough
 * that the snapshot on the far side has almost always arrived by the time it
 * is wanted, so there is something to read between rather than nothing.
 *
 * A hundred milliseconds is two ticks and a margin. Shorter and a late
 * packet leaves nothing to interpolate towards, which shows up as a hitch;
 * much longer and steering starts to feel posted rather than driven.
 */
const DELAY = 100;

/** The last few snapshots, oldest first. */

/*
 * Whether this player is out and waiting to come back.
 *
 * It exists because a round can end while somebody is dead. The win is sent
 * to everybody who was in the arena when the tick began, which includes the
 * player eaten during that same tick, so their "eaten" screen is replaced by
 * the winner's - and dismissing that used to leave them with no screen at
 * all, not in the game, and nothing to press. That reads as the game having
 * frozen, because from the inside it is indistinguishable from it.
 */
let out = false;

/*
 * The floor is a sheet, and anything heavy enough to eat you sits in a dent
 * it has made in it. The dent is not decoration - it is the same pull the
 * server is applying to you, drawn.
 *
 * The depth is a quartic that reaches exactly as far as the pull does and
 * then is flatly zero, so a well can be rejected with one comparison and
 * there is no tail to fade out by hand. TILT is how much of that depth is
 * spent sliding things down the screen - deeper is lower, the way the foot
 * of a post is lower on screen than its top; FLOOR is as deep as the sheet
 * is ever drawn, so that two wells on top of each other bottom out rather
 * than falling through the floor.
 */
const TILT = 0.55;
const FLOOR = 400;

/*
 * How much the sheet is gathered inwards as well as pushed down.
 *
 * Sag on its own bends the lines that run across the screen and leaves the
 * ones running up it perfectly straight, because moving a point down does
 * not move it sideways - so it reads as a ripple rather than as a hole.
 * Pulling points in towards the well as well bends both sets and crowds the
 * squares together going in, which is what a funnel does to a grid.
 */
const GATHER = 0.5;

/*
 * Chat.
 *
 * The box is hidden until somebody asks for it, and while it is open the
 * game's keys belong to it - otherwise typing "space" into a message roots
 * you to the floor, and typing an E teleports you into whatever you were
 * running from.

const chatbox = $('chatbox');

function talking() {
  return !chatbox.hidden && document.activeElement === chatbox;
}

function speak(open) {
  chatbox.hidden = !open;

  if (open) {
    chatbox.focus();
  } else {
    chatbox.value = '';
    chatbox.blur();
  }
}

function heard(who, what, mine) {
  const line = document.createElement('div');

  if (mine) line.className = 'mine';

  // text, never markup: both halves of this came from somebody's keyboard
  const name = document.createElement('b');

  name.textContent = `${who} `;

  line.append(name, document.createTextNode(what));
  $('chatlog').append(line);

  line.addEventListener('animationend', () => line.remove());

  while ($('chatlog').childElementCount > 8) $('chatlog').firstElementChild.remove();
}

chatbox.addEventListener('keydown', (e) => {
  e.stopPropagation();

  if (e.key === 'Escape') {
    speak(false);

    return;
  }

  if (e.key !== 'Enter') return;

  const said = chatbox.value.trim();

  if (said && socket?.readyState === 1) {
    socket.send(JSON.stringify({ kind: 'say', text: said.slice(0, 120) }));
  }

  speak(false);
});

chatbox.addEventListener('blur', () => { if (!chatbox.value.trim()) speak(false); });

  if (touching) return;

  touching = true;

  $('thumbs').hidden = false;
}

 * Text, never markup: a name comes from whoever typed it, and the only

function say(text) {

  line.className = 'said';
  line.textContent = text;

  feed.append(line);

  // the fading out is the stylesheet's; this is only the tidying up after it
  line.addEventListener('animationend', () => line.remove());

}

/** Where the pointer is, in world units. */
function aim() {
  if (!camera) return null;

  const scale = scaleFor(camera.r);

  return {
    x: camera.x + (pointer.x - innerWidth / 2) / scale,
    y: camera.y + (pointer.y - innerHeight / 2) / scale
  };
}

/** Teleports to wherever the pointer or the last finger was. */
function hop() {
  const to = aim();

  if (to && socket?.readyState === 1 && cool <= 0 && !collapsed) {
    socket.send(JSON.stringify({ kind: 'warp', x: to.x, y: to.y }));

    return true;
  }

  return false;
}

addEventListener('keydown', (e) => {
  // Enter opens the box, and while it is open nothing here fires
  if (talking()) return;

  if (e.key === 'Enter' && !e.repeat) {
    speak(true);
    e.preventDefault();

    return;
  }

  /*
   * Teleport. On its own key rather than on a mouse button, because both
   * buttons already mean something and the one thing everybody has should
   * not be the one thing that needs explaining.
   */
  if (e.code === 'KeyE' && !e.repeat) {
    hop();

    e.preventDefault();
    return;
  }

  if (e.code !== 'Space' || e.repeat) return;
  hold = true;
  e.preventDefault();
});

addEventListener('keyup', (e) => {
  if (talking() || e.code !== 'Space') return;
  hold = false;
  e.preventDefault();
});

// letting go of the window counts as letting go of the key, or a player who
// alt-tabs mid pull comes back to find themselves still burning
addEventListener('blur', () => { hold = false; });

// pressing draws a line, when the line is off its wait
addEventListener('pointerdown', (e) => {
  if (e.pointerType === 'touch') touchy();

  // the buttons and the overlays are not the arena, and a press that lands
  // on one of them has already been dealt with by whatever it landed on
  if (e.target?.closest?.('button, input, #thumbs, #gate, #over, #won')) return;

  if (e.button !== 0 || socket?.readyState !== 1 || !camera) return;

  /*
   * A finger reports where it is only while it is down, so the press is
   * also the first move: without this the first tap would be aimed at
   * wherever the previous one happened to end.
   */
  if (e.pointerType === 'touch') pointer = { x: e.clientX, y: e.clientY };

  /*
   * A hole has nowhere to walk to and nothing in hand to throw, so pressing
   * the floor means the one thing it can do: open a whirl there. This is the
   * whole of the endgame - it is how something that cannot chase anybody
   * gets anybody to come to it.
   */
  if (collapsed) {
    const to = aim();

    if (to && cool <= 0) socket.send(JSON.stringify({ kind: 'haul', x: to.x, y: to.y }));

    return;
  }

  // armed by the teleport button, which is how it is done without a keyboard
  if (aiming) {
    aiming = false;
    $('hop').classList.remove('on');
    hop();

    return;
  }

  /*
   * A line is started by pressing and spent when it is let go of, so
   * somebody who presses and changes their mind has not spent their wait
   * on a gesture that never finished.
   *
   * On glass it has to be armed first. A tap on the floor is how you steer
   * there, so without the arming step the drag that moves you would be the
   * same gesture as the one that spends the best move in the game.
   */
  if (pen <= 0 && (!touching || sketching)) {
    const at = aim();

    drawing = at ? [at] : null;
  }
});

/** How many pixels one world unit is. */
function scaleFor(radius) {
  return Math.min(innerWidth, innerHeight) / extentFor(radius);
}

/**
 * How much of the arena fits across the shorter side of the window.
 *
 * Bigger blobs see more, but a good deal less than their size would suggest.
 * Straight off the radius and a blob near the cap looks at nine thousand
 * units of arena through an eight hundred pixel window, where a starting
 * blob is two pixels across and the pellets are dust you cannot aim at. This
 * gets to about three thousand instead, which still fills the screen with
 * the arena and leaves everything in it a size worth steering at.
 *
 * The server keeps the same curve, because it decides what to send from it.
 */
function extentFor(radius) {
  return 600 + 260 * Math.pow(radius / 14, 0.7);
}

/** The blobs heavy enough to dent the floor, as the drawing wants them. */
function wellsOf(blobs, mine, spread) {
  const wells = [];

  for (const b of blobs) {
    /*
     * A blob bends the floor as hard as it is gripping, which the server
     * sends: nought for something too small to bend anything, one for
     * standing there being large, up to three for holding the space bar.
     * So somebody who holds grows a well even if they were too small to
     * have one, and everybody nearby watches the floor cave in under them
     * before the pull that is coming arrives.
     */
    const grip = b.g ?? (b.r >= wellSize ? 1 : 0);

    if (grip > 0) {
      // the dent widens with the grip as well as the size, the same way the
      // pull does - see Reach in World.cs, which is the half that matters
      const reach = b.r * 4 * Math.sqrt(Math.max(1, grip));

      // the dent stops deepening long before a blob stops growing. Past this
      // the well is already wider than the screen, and letting it go on would
      // gather everything visible into a single point rather than look like
      // anything - what keeps reading as bigger is the width, not the depth
      wells.push({
        i: b.i, x: b.x, y: b.y,
        deep: Math.min(b.r * 2 * grip, 320 * Math.max(1, grip)),
        span: reach * reach,
      });
    }
  }

  return wells;
}

/**
 * Where a point of the floor has moved to, and how far it has sunk.
 *
 * One pass over the wells doing both halves of the warp: in towards each
 * well, and down. `mine` leaves one well out, which is how a blob is given
 * the bend of everybody else's dent without being given its own.
 */
function sink(x, y, wells, mine) {
  let gx = x;
  let gy = y;
  let z = 0;

  for (const w of wells) {
    if (w.i === mine) continue;

    const dx = x - w.x;
    const dy = y - w.y;
    const d2 = dx * dx + dy * dy;

    if (d2 >= w.span) continue;

    // a quartic that reaches exactly as far as the pull does and is then
    // flatly zero, so a well is rejected with one comparison and there is
    // no tail left to fade out by hand
    const near = 1 - d2 / w.span;
    const dip = w.deep * near * near;

    z += dip;

    const d = Math.sqrt(d2);

    if (d > 0.001) {
      const inwards = Math.min(d * 0.85, dip * GATHER);

      gx -= dx / d * inwards;
      gy -= dy / d * inwards;
    }
  }

  return { x: gx, y: gy, z: FLOOR * Math.tanh(z / FLOOR) };
}

/**
 * Everything as it was DELAY milliseconds ago, read between the two
 * snapshots that bracket that moment.
 */
function moment() {
  if (history.length === 0) return null;

  const want = performance.now() - DELAY;

  // the newest pair whose older half is already behind us
  let i = history.length - 1;

  while (i > 0 && history[i - 1].at > want) i--;

  const after = history[i];
  const before = history[i - 1] ?? after;

  const span = after.at - before.at;
  const t = span > 0 ? Math.min(1, Math.max(0, (want - before.at) / span)) : 1;

  const was = new Map(before.b.map((b) => [b.i, b]));

  const blobs = after.b.map((b) => {
    const a = was.get(b.i);

    // somebody who has only just appeared has nowhere to come from
    if (!a) return b;

    return {
      i: b.i, n: b.n, c: b.c,
      x: a.x + (b.x - a.x) * t,
      y: a.y + (b.y - a.y) * t,
      r: a.r + (b.r - a.r) * t,
      g: (a.g ?? 0) + ((b.g ?? 0) - (a.g ?? 0)) * t,
      // a whole number of kills, which is not a thing to slide between
      s: b.s ?? 0,
    };
  });

  /*
   * Pellets are read the same way now that they are dragged about: four
   * numbers each, the first of which says which pellet it is, so one seen in
   * both snapshots can be drawn between the two places it was. One that has
   * only just appeared is drawn where it is; one that has gone is simply not
   * in the newer half, and so is not drawn at all.
   */
  const dust = new Map();

  for (let i = 0; i < before.p.length; i += 4) dust.set(before.p[i], i);

  const pellets = [];

  for (let i = 0; i < after.p.length; i += 4) {
    const j = dust.get(after.p[i]);

    // a pellet is dragged at most a few units between two frames, so a slot
    // that appears to have crossed the arena is not the pellet it was - the
    // server leaves a freed slot alone long enough that this should never
    // fire, and drawing the streak if it ever did would be worse than not
    const jumped = j === undefined
                || Math.abs(after.p[i + 1] - before.p[j + 1]) > 40
                || Math.abs(after.p[i + 2] - before.p[j + 2]) > 40;

    if (jumped) {
      pellets.push(after.p[i + 1], after.p[i + 2], after.p[i + 3]);
    } else {
      pellets.push(
        before.p[j + 1] + (after.p[i + 1] - before.p[j + 1]) * t,
        before.p[j + 2] + (after.p[i + 2] - before.p[j + 2]) * t,
        after.p[i + 3]);
    }
  }

  return { blobs, pellets };
}

/**
 * What a blob has grown into, near enough to pick a tier with.
 *
 * Only ever used for the fire and the aura, which is all it is good for: it
 * runs on the rounded radius that travels in a frame, and past the collapse
 * that is a third of a per cent out. The numbers anybody reads - their own
 * score and the leaderboard - both come from the server.
 *
 * Score rather than radius, because score is the number the player is
 * watching go up and radius is the square root of it - tiers picked on radius
 * land at sizes that mean nothing to anybody reading the board.
 */
/*
 * The same sum the server does, including the part where it stops being area.
 *
 * Past the knee the score climbs far faster than the radius does, or ten
 * billion would be a blob fifty six thousand units across in an arena nine
 * thousand wide. The knee and the collapse are two different numbers - one
 * is an area and one is a score - and the two halves have to agree about
 * both or the fire comes on at the wrong size.
 */
const HOLE = 5e6;          // the plain area a blob collapses at
const COLLAPSE = 1e7;      // and the score it has when it does
const LIFT = COLLAPSE / HOLE;
const STEEPEN = 156.90;

function SCORE(r) {
  const plain = Math.PI * (r * r - 14 * 14);

  return plain <= HOLE ? plain * LIFT : COLLAPSE * Math.pow(plain / HOLE, STEEPEN);
}

/** The grades a hole comes in, matching World.Grades9. */
/*
 * The grades a hole comes in, matching World.Grades9 - radii, not scores.
 *
 * Two hundred orders of magnitude between the collapse and the ceiling means
 * any score you pick for a middle grade is an absurd number sitting between
 * two other absurd numbers. The radii are what a player can actually see.
 */
const GRADES = [1261.63, 1900, 2700, 3700, 4700];

/**
 * A score, short enough to read while something is chasing you.
 *
 * Nobody can take a nineteen digit number off a screen in the half second
 * they have to look at it, and the leaderboard is eight of them stacked. Full
 * digits up to a million, where they are still worth reading exactly, and
 * three significant figures with a suffix above it.
 */
const SUFFIX = [[1e33, 'Dc'], [1e30, 'No'], [1e27, 'Oc'], [1e24, 'Sp'], [1e21, 'Sx'],
                [1e18, 'Qi'], [1e15, 'Qa'], [1e12, 'T'], [1e9, 'B'], [1e6, 'M']];

function tally(n) {
  if (!(n > 0)) return '0';

  // past a decillion the names stop being names anybody knows, and the
  // ceiling is a hundred and ninety three orders of magnitude further up
  if (n >= 1e36) return n.toExponential(2).replace('e+', 'e');

  for (const [at, tag] of SUFFIX) {
    if (n >= at) {
      const v = n / at;

      return `${v >= 100 ? v.toFixed(0) : v >= 10 ? v.toFixed(1) : v.toFixed(2)}${tag}`;
    }
  }

  return Math.round(n).toLocaleString();
}

function holeLevel(r) {
  let level = 0;

  for (const step of GRADES) if (r >= step) level++;

  return level;
}

function stageOf(r) {
  const s = SCORE(r);

  if (s >= COLLAPSE) return 4;  // collapsed
  if (s >= 6000000) return 3;   // black fire
  if (s >= 2500000) return 2;   // fire
  if (s >= 400000) return 1;    // embers
  return 0;
}

/**
 * The fire around something large, and the hole around something larger.
 *
 * Drawn as a handful of arcs at rotating offsets rather than as particles:
 * there can be a dozen of these on screen and a particle system for each is
 * more frame than the effect is worth. The arcs are seeded off the blob's own
 * position so two blobs of the same size do not flicker in step.
 */
function drawAura(x, y, r, stage, id, spin, level) {
  if (stage <= 0) return;

  const seed = (id.charCodeAt(0) + id.charCodeAt(1) * 7) % 100 / 100;

  if (stage === 4) {
    const grade = Math.max(1, level || 1);
    // the hole: a rim of light round a disc of nothing, and the light is all
    // of it - the middle is painted out so it reads as absence
    const spread = r * (2.1 + grade * 0.45);

    const halo = ctx.createRadialGradient(x, y, r * 0.86, x, y, spread);

    halo.addColorStop(0, 'rgba(255,238,200,0.85)');
    halo.addColorStop(0.12, 'rgba(255,150,60,0.55)');
    halo.addColorStop(0.4, 'rgba(120,40,160,0.30)');
    halo.addColorStop(1, 'rgba(0,0,0,0)');

    ctx.fillStyle = halo;
    ctx.beginPath();
    ctx.arc(x, y, spread, 0, Math.PI * 2);
    ctx.fill();

    for (let i = 0; i < 2 + grade; i++) {
      ctx.strokeStyle = `rgba(255,${190 - i * 50},${120 - i * 40},${0.5 - i * 0.12})`;
      ctx.lineWidth = Math.max(1.5, r * (0.05 - i * 0.012));
      ctx.beginPath();
      ctx.ellipse(x, y, r * (1.15 + i * 0.2), r * (0.3 + i * 0.09),
                  spin * (0.6 + i * 0.25) + seed * 6, 0, Math.PI * 2);
      ctx.stroke();
    }

    ctx.fillStyle = '#05060a';
    ctx.beginPath();
    ctx.arc(x, y, r * 0.94, 0, Math.PI * 2);
    ctx.fill();

    return;
  }

  const tongues = stage === 1 ? 5 : stage === 2 ? 8 : 10;
  const reach = stage === 1 ? 0.22 : stage === 2 ? 0.38 : 0.5;

  ctx.save();
  ctx.globalCompositeOperation = 'lighter';

  for (let i = 0; i < tongues; i++) {
    const a = (i / tongues) * Math.PI * 2 + spin * (0.5 + (i % 3) * 0.2) + seed * 6;
    const wob = 0.6 + 0.4 * Math.sin(spin * 3 + i * 1.7 + seed * 10);
    const len = r * reach * wob;

    const fx = x + Math.cos(a) * (r + len * 0.5);
    const fy = y + Math.sin(a) * (r + len * 0.5);

    const flame = ctx.createRadialGradient(fx, fy, 0, fx, fy, Math.max(2, len));

    if (stage === 3) {
      // black fire: violet at the root going to nothing, so it eats the
      // starfield behind it rather than lighting it
      flame.addColorStop(0, `rgba(196,132,255,${0.42 * wob})`);
      flame.addColorStop(0.45, `rgba(92,30,140,${0.30 * wob})`);
      flame.addColorStop(1, 'rgba(10,0,20,0)');
    } else if (stage === 2) {
      flame.addColorStop(0, `rgba(255,236,170,${0.5 * wob})`);
      flame.addColorStop(0.4, `rgba(255,132,40,${0.34 * wob})`);
      flame.addColorStop(1, 'rgba(120,20,0,0)');
    } else {
      flame.addColorStop(0, `rgba(255,190,120,${0.3 * wob})`);
      flame.addColorStop(1, 'rgba(120,50,0,0)');
    }

    ctx.fillStyle = flame;
    ctx.beginPath();
    ctx.arc(fx, fy, Math.max(2, len), 0, Math.PI * 2);
    ctx.fill();
  }

  ctx.restore();
}

/**
 * A teleport that has not landed yet.
 *
 * The mark is drawn at the size of the thing arriving, which is the whole
 * reason anybody else is told about it: a small one landing next to you is
 * lunch and a large one is the end of your round, and a reticle that was the
 * same either way would be noise. It fills as the crossing runs, so the
 * question it answers is not only what is coming but how long there is.
 *
 * Flat, not put: the marker has to be exactly where the blob will be, and
 * the floor under it may be bent by half a screen if a hole is nearby.
 */
/**
 * A number in, the same number out every time, between nought and one.
 *
 * Lightning has to be jagged and it has to hold still between redraws, or at
 * sixty frames a second it is a strobe rather than an arc. Seeding it on the
 * blob, the bolt and a slow counter gives both: a shape nobody can predict
 * and one that survives long enough to be seen.
 */
function jolt(n) {
  const x = Math.sin(n * 127.1) * 43758.5453;

  return x - Math.floor(x);
}

/**
 * The charge a kill leaves on whoever took it.
 *
 * Drawn on everybody, not only on the player: it is a warning, and a warning
 * only their own screen can see is not one. The count is legible from the
 * arcs - more of them, reaching further, brighter at the core - so that
 * deciding whether to go near somebody does not require reading a number.
 */
function drawSpark(x, y, r, sparks, seed, now) {
  const heat = Math.min(1, sparks / 5);
  const bolts = 3 + sparks * 2;

  // a new shape a dozen times a second. Every frame is a strobe; once a
  // second is a diagram
  const step = Math.floor(now / 75);

  ctx.save();
  ctx.lineCap = 'round';

  // the air around it going, which is what carries at a distance where the
  // arcs themselves are a pixel wide
  ctx.globalAlpha = 0.12 + heat * 0.22;
  ctx.strokeStyle = '#8adcff';
  ctx.lineWidth = 2 + heat * 4;
  ctx.beginPath();
  ctx.arc(x, y, r + 5 + heat * 5, 0, Math.PI * 2);
  ctx.stroke();

  for (let b = 0; b < bolts; b++) {
    const seeded = seed + b * 31 + step * 7;
    const a = (b / bolts) * Math.PI * 2 + jolt(seeded) * 0.7;
    const tip = r + 8 + (14 + 16 * heat) * (0.4 + jolt(seeded + 3) * 0.6);

    ctx.beginPath();
    ctx.moveTo(x + Math.cos(a) * r * 0.9, y + Math.sin(a) * r * 0.9);

    // three segments, each kicked sideways: a straight line from the rim is
    // a spoke, and a spoke reads as a wheel rather than as electricity
    for (let s = 1; s <= 3; s++) {
      const along = r * 0.9 + (tip - r * 0.9) * (s / 3);
      const off = (jolt(seeded + s * 17) - 0.5) * (0.42 + 0.3 * heat);

      ctx.lineTo(x + Math.cos(a + off) * along, y + Math.sin(a + off) * along);
    }

    ctx.globalAlpha = 0.55 + heat * 0.45;
    ctx.strokeStyle = '#8adcff';
    ctx.lineWidth = 3 + heat * 2;
    ctx.stroke();

    // a white core inside the blue one, which is what makes it read as hot
    ctx.globalAlpha = 0.5 + heat * 0.5;
    ctx.strokeStyle = '#ffffff';
    ctx.lineWidth = 1 + heat;
    ctx.stroke();
  }

  ctx.restore();
}

/**
 * A whirl, which is the only thing a hole can do to somebody at a distance.
 *
 * Drawn under everything, on the floor, because that is what it is: a patch
 * of the arena that has stopped being safe to stand on. The edge is drawn as
 * a hard line rather than a fade, since the edge is the escape and somebody
 * deciding whether to run needs to know where it is rather than roughly
 * where it is.
 */
/**
 * The whole arena, small, in the corner.
 *
 * The one view in the game that is not about what you can see. A blob three
 * screens away is a thing you can do nothing about and very much need to
 * know the direction of, and before this the only way to find out was to
 * grow until it came into view - which is exactly the wrong order.
 *
 * Holes get their own symbols, one per grade, rather than a bigger dot. The
 * difference between a grade one and a grade five is the difference between
 * something to be careful near and something that owns a third of the map,
 * and a size ramp at this resolution is four pixels against six.
 */
function drawChart(world, self) {
  const side = mapped.width;

  mapping.clearRect(0, 0, side, side);

  if (!chart.length || !world) return;

  const span = side / world;

  // the floor of the arena, so the edges read as edges
  mapping.fillStyle = 'rgba(138, 180, 248, 0.05)';
  mapping.fillRect(0, 0, side, side);

  const spin = performance.now() / 1000;

  for (let i = 0; i < chart.length; i += 3) {
    const x = chart[i] * span;
    const y = chart[i + 1] * span;
    const mark = chart[i + 2];

    if (mark < 4) {
      // an ordinary blob: a dot that grows a little with the size band
      mapping.globalAlpha = 0.5 + mark * 0.15;
      mapping.fillStyle = '#9aa0a6';
      mapping.beginPath();
      mapping.arc(x, y, 3 + mark * 1.6, 0, Math.PI * 2);
      mapping.fill();

      continue;
    }

    grim(x, y, mark - 3, spin);
  }

  // and you, last and in white, because the one thing this has to answer
  // instantly is where you are on it
  if (self) {
    mapping.globalAlpha = 1;
    mapping.strokeStyle = '#fff';
    mapping.lineWidth = 3;
    mapping.beginPath();
    mapping.arc(self.x * span, self.y * span, 7, 0, Math.PI * 2);
    mapping.stroke();

    mapping.fillStyle = '#fff';
    mapping.beginPath();
    mapping.arc(self.x * span, self.y * span, 2.5, 0, Math.PI * 2);
    mapping.fill();
  }

  mapping.globalAlpha = 1;
}

/**
 * A hole on the minimap: a different mark for each of the five grades.
 *
 * Meant to be told apart at a glance and at eight pixels, so they differ in
 * shape before they differ in size - a ring, a ring with an eye, a
 * crosshair, a star, and a black sun. Size and brightness climb with them as
 * well, but the shape is what carries at this scale.
 */
function grim(x, y, grade, spin) {
  const r = 5 + grade * 1.9;

  mapping.save();
  mapping.translate(x, y);
  mapping.globalAlpha = 1;

  // every grade sits in a dark well, which is what says "hole" before the
  // shape says which one
  mapping.fillStyle = `rgba(6, 2, 14, ${0.45 + grade * 0.1})`;
  mapping.beginPath();
  mapping.arc(0, 0, r + 3, 0, Math.PI * 2);
  mapping.fill();

  mapping.strokeStyle = grade >= 4 ? '#ffd66b' : '#c884ff';
  mapping.fillStyle = mapping.strokeStyle;
  mapping.lineWidth = 2;
  mapping.lineCap = 'round';

  mapping.beginPath();
  mapping.arc(0, 0, r, 0, Math.PI * 2);
  mapping.stroke();

  if (grade >= 2) {
    // an eye
    mapping.beginPath();
    mapping.arc(0, 0, grade >= 5 ? r * 0.5 : 2, 0, Math.PI * 2);
    mapping.fill();
  }

  if (grade === 3 || grade === 4) {
    // a crosshair at three, and the same turned into a star at four
    const arms = grade === 3 ? 4 : 6;
    const turn = grade === 4 ? spin * 0.6 : 0;

    mapping.beginPath();

    for (let a = 0; a < arms; a++) {
      const t = turn + (a / arms) * Math.PI * 2;

      mapping.moveTo(Math.cos(t) * r * 0.45, Math.sin(t) * r * 0.45);
      mapping.lineTo(Math.cos(t) * (r + 4), Math.sin(t) * (r + 4));
    }

    mapping.stroke();
  }

  if (grade >= 5) {
    // a corona, turning, which nothing else on the map has
    mapping.beginPath();

    for (let a = 0; a < 10; a++) {
      const t = spin + (a / 10) * Math.PI * 2;

      mapping.moveTo(Math.cos(t) * (r + 1), Math.sin(t) * (r + 1));
      mapping.lineTo(Math.cos(t) * (r + 6), Math.sin(t) * (r + 6));
    }

    mapping.lineWidth = 1.5;
    mapping.stroke();
  }

  mapping.restore();
}

function drawWhirls(flat, scale, now) {
  if (!whirls.length) return;

  const turn = now / 620;

  for (const eddy of whirls) {
    const at = flat(eddy.x, eddy.y);
    const r = eddy.r * scale;

    // nothing to draw if the whole of it is off the side of the screen
    if (at.x + r < 0 || at.x - r > innerWidth || at.y + r < 0 || at.y - r > innerHeight) continue;

    const bite = Math.min(1, (eddy.t ?? 0) * 4) * (1 - Math.max(0, (eddy.t ?? 0) - 0.85) / 0.15);

    ctx.save();

    // the dark of it, painted over the square it covers rather than the
    // whole canvas - the same economy the wells are drawn with
    const left = Math.max(0, at.x - r), top = Math.max(0, at.y - r);
    const right = Math.min(innerWidth, at.x + r), bottom = Math.min(innerHeight, at.y + r);

    if (right > left && bottom > top) {
      const shade = ctx.createRadialGradient(at.x, at.y, 0, at.x, at.y, r);

      shade.addColorStop(0, `rgba(0,0,0,${0.82 * bite})`);
      shade.addColorStop(0.3, `rgba(10,4,24,${0.5 * bite})`);
      shade.addColorStop(0.75, `rgba(14,8,30,${0.16 * bite})`);
      shade.addColorStop(1, 'rgba(14,8,30,0)');

      ctx.fillStyle = shade;
      ctx.fillRect(left, top, right - left, bottom - top);
    }

    // arms, wound in. One path for all of them: this is drawn every frame
    ctx.globalAlpha = 0.5 * bite;
    ctx.strokeStyle = eddy.c;
    ctx.lineWidth = 2;
    ctx.beginPath();

    for (let arm = 0; arm < 6; arm++) {
      const lead = turn + (arm / 6) * Math.PI * 2;

      for (let s = 0; s <= 22; s++) {
        const along = s / 22;
        const rad = r * (0.06 + along * 0.94);
        const a = lead + along * 3.4;
        const x = at.x + Math.cos(a) * rad, y = at.y + Math.sin(a) * rad;

        if (s === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
      }
    }

    ctx.stroke();

    // and the edge, which is the line worth being on the far side of
    ctx.globalAlpha = 0.75 * bite;
    ctx.lineWidth = 3;
    ctx.setLineDash([16, 12]);
    ctx.lineDashOffset = -turn * 40;
    ctx.beginPath();
    ctx.arc(at.x, at.y, r, 0, Math.PI * 2);
    ctx.stroke();

    ctx.restore();
  }
}

function drawMarks(flat, scale) {
  if (!marks.length) return;

  const turn = performance.now() / 700;

  for (const mark of marks) {
    const at = flat(mark.x, mark.y);
    const r = Math.max(7, mark.r * scale);
    const t = Math.min(1, Math.max(0, mark.t ?? 0));

    ctx.save();
    ctx.translate(at.x, at.y);

    // the ground it is taking, filling as it comes
    ctx.globalAlpha = 0.1 + t * 0.24;
    ctx.fillStyle = mark.c;
    ctx.beginPath();
    ctx.arc(0, 0, r * t, 0, Math.PI * 2);
    ctx.fill();

    // the edge of it, turning, so a mark is never read as a blob
    ctx.globalAlpha = 0.45 + t * 0.5;
    ctx.strokeStyle = mark.c;
    ctx.lineWidth = 2;
    ctx.setLineDash([r * 0.34, r * 0.26]);
    ctx.lineDashOffset = -turn * r;
    ctx.beginPath();
    ctx.arc(0, 0, r, 0, Math.PI * 2);
    ctx.stroke();
    ctx.setLineDash([]);

    // four spokes closing on it, which is the part that reads as a
    // countdown rather than as decoration
    const ring = r + 8 + 30 * (1 - t);
    const arm = Math.max(5, r * 0.3);

    ctx.lineWidth = 3;
    ctx.globalAlpha = 0.25 + t * 0.65;
    ctx.beginPath();

    for (let q = 0; q < 4; q++) {
      const a = q * Math.PI / 2 + Math.PI / 4;
      const cx = Math.cos(a), cy = Math.sin(a);

      ctx.moveTo(cx * (ring + arm), cy * (ring + arm));
      ctx.lineTo(cx * ring, cy * ring);
    }

    ctx.stroke();
    ctx.restore();
  }
}

/**
 * Blobs coming apart where they were eaten.
 *
 * This has to be told to us rather than noticed: by the time a browser could
 * work out that somebody has stopped appearing in the frames, it has already
 * drawn one they are simply absent from, and a blob that vanishes between
 * two frames reads as a dropped connection rather than as a death.
 *
 * Drawn over the blobs rather than under them, including over whatever just
 * ate this one - the swallowing is the thing being shown.
 */
function drawDeaths(flat, scale) {
  if (!deaths.length) return;

  const now = performance.now();

  ctx.save();

  // backwards, because finished ones are dropped on the way through
  for (let i = deaths.length - 1; i >= 0; i--) {
    const gone = deaths[i];
    const age = (now - gone.at) / DEATH;

    if (age >= 1) {
      deaths.splice(i, 1);

      continue;
    }

    const at = flat(gone.x, gone.y);
    const r = Math.max(5, gone.r * scale);
    const swell = 1 - Math.pow(1 - age, 3);
    const left = 1 - age;

    // the shell of it, thrown outwards and thinning as it goes
    ctx.globalAlpha = left * 0.85;
    ctx.strokeStyle = gone.c;
    ctx.lineWidth = Math.max(1, 6 * left);
    ctx.beginPath();
    ctx.arc(at.x, at.y, r * (1 + swell * 1.5), 0, Math.PI * 2);
    ctx.stroke();

    // and what it was made of, going with it. One path for all ten, because
    // a crowded arena can have several of these at once
    ctx.lineWidth = Math.max(1, 4 * left);
    ctx.beginPath();

    for (let s = 0; s < 10; s++) {
      const a = gone.turn + s * Math.PI / 5;
      const near = r * (0.7 + swell * 1.5);
      const far = near + r * 0.55 * left + 6;

      ctx.moveTo(at.x + Math.cos(a) * near, at.y + Math.sin(a) * near);
      ctx.lineTo(at.x + Math.cos(a) * far, at.y + Math.sin(a) * far);
    }

    ctx.stroke();

    // the flash, which is over almost before it starts
    if (age < 0.3) {
      ctx.globalAlpha = (1 - age / 0.3) * 0.7;
      ctx.fillStyle = '#fff';
      ctx.beginPath();
      ctx.arc(at.x, at.y, r * (1 + age), 0, Math.PI * 2);
      ctx.fill();
    }
  }

  ctx.restore();
}

function draw() {
  requestAnimationFrame(draw);

  ctx.setTransform(devicePixelRatio, 0, 0, devicePixelRatio, 0, 0);
  ctx.fillStyle = '#0b0d11';
  ctx.fillRect(0, 0, innerWidth, innerHeight);

  const now = moment();

  if (!now) return;

  const mine = now.blobs.find((b) => b.i === me);

  if (!mine) return;

  // the camera sits on the same position the blob is drawn at, so the
  // player is exactly in the middle of the screen every frame rather than
  // swinging about it
  camera = mine;

  const scale = scaleFor(mine.r);
  // how far the screen reaches into the world, which is what decides
  // whether a dent is worth carrying through the rest of the frame
  const spread = Math.max(innerWidth, innerHeight) / scale;

  const wells = wellsOf(now.blobs, mine, spread);

  /*
   * How deep the floor is under the player, which everything else is drawn
   * relative to. Without it, standing in a well would slide the whole world
   * off the screen instead of the floor rising around the player - which is
   * what standing in a hole actually looks like. Their own dent is left out,
   * so that growing does not drag the world out from under them.
   */
  const under = sink(mine.x, mine.y, wells, mine.i).z;

  const ox = innerWidth / 2 - mine.x * scale;
  const oy = innerHeight / 2 - mine.y * scale;

  /**
   * Where a point of the floor lands on the screen.
   *
   * For the floor only. Nothing that can be eaten goes through here - see
   * `flat` below, and the note on it, which is the whole reason this is two
   * functions rather than one.
   */
  const put = (x, y) => {
    const s = sink(x, y, wells);

    return {
      x: ox + s.x * scale,
      y: oy + s.y * scale + (s.z - under) * scale * TILT,
      z: s.z,
    };
  };

  /*
   * Where a thing standing on the floor lands on the screen: exactly where
   * it is, with none of the bending applied to it.
   *
   * This is not a shortcut, it is the point. The server decides who eats
   * whom from where things actually are, so if the drawing moved them the
   * two would disagree - and they disagree by a lot, because the gathering
   * is strongest exactly where the eating happens. Next to a blob of radius
   * ninety six, something eighty units too far away to be caught is drawn
   * comfortably inside it: you would aim at what you could see, miss, and
   * conclude the hit box was broken. It was; this is it fixed.
   *
   * The floor keeps bending underneath them, which is where the whole
   * effect lives anyway. A blob covers the deepest part of its own dent, so
   * what shows is the ring around it - and that reads as sitting in it.
   */
  const flat = (x, y) => ({ x: ox + x * scale, y: oy + y * scale });

  /*
   * The dark of each well, under the grid so the lines read as lying on it.
   *
   * Painted over the square the well actually covers rather than over the
   * whole canvas. A radial gradient across twelve hundred by eight hundred
   * is a megapixel of work, and there is one of these per well: at one or
   * two that was merely wasteful, but a hold widens its own reach and pulls
   * more blobs into view, and a screen with eight of them was repainting
   * eight million pixels of gradient every frame for a dent that covers a
   * fraction of it.
   */
  for (const w of wells) {
    const at = put(w.x, w.y);
    const span = Math.sqrt(w.span) * scale;
    const sag = Math.min(w.deep, FLOOR) * scale * TILT;

    const left = Math.max(0, at.x - span);
    const top = Math.max(0, at.y - span - sag);
    const right = Math.min(innerWidth, at.x + span);
    const bottom = Math.min(innerHeight, at.y + span);

    if (right <= left || bottom <= top) continue;

    // the two focuses are offset, which tilts the cone for free and lights
    // it from the top of the screen without a second pass
    const shade = ctx.createRadialGradient(at.x, at.y, 0, at.x, at.y - sag * 0.45, span);

    shade.addColorStop(0, 'rgba(0,0,0,0.72)');
    shade.addColorStop(0.35, 'rgba(4,6,16,0.46)');
    shade.addColorStop(0.7, 'rgba(8,12,26,0.18)');
    shade.addColorStop(1, 'rgba(8,12,26,0)');

    ctx.fillStyle = shade;
    ctx.fillRect(left, top, right - left, bottom - top);
  }

  drawSpace(put, scale,
            Math.max(0, mine.x - innerWidth / 2 / scale),
            Math.min(world, mine.x + innerWidth / 2 / scale),
            Math.max(0, mine.y - innerHeight / 2 / scale - FLOOR * TILT),
            Math.min(world, mine.y + innerHeight / 2 / scale + FLOOR * TILT));

  // over the stars and under everything that stands on the floor
  drawWhirls(flat, scale, performance.now());

  // pellets arrive flat - x, y, colour, x, y, colour - so they are walked
  // three at a time rather than unpacked into objects that live one frame
  ctx.globalAlpha = 0.75;

  const pellets = now.pellets;

  /*
   * Gathered by colour and filled once each.
   *
   * There are fifteen hundred of these and the loop used to set fillStyle,
   * open a path and fill it for every one - three thousand state changes a
   * frame, which is most of what the canvas was doing. A path holds as many
   * arcs as it likes and a fill costs the same whether it is one circle or
   * four hundred, so the whole floor is a dozen fills now.
   */
  const heaps = new Map();

  for (let i = 0; i < pellets.length; i += 3) {
    const at = flat(pellets[i], pellets[i + 1]);

    // colour in the low four bits, how big it is above them
    const tag = pellets[i + 2];
    const key = tag & 15;

    let heap = heaps.get(key);

    if (!heap) heaps.set(key, heap = new Path2D());

    const r = Math.max(2, (girths[tag >> 4] ?? girths[0]) * scale);

    heap.moveTo(at.x + r, at.y);
    heap.arc(at.x, at.y, r, 0, Math.PI * 2);
  }

  for (const [key, heap] of heaps) {
    ctx.fillStyle = palette[key] ?? '#9aa0a6';
    ctx.fill(heap);
  }

  ctx.globalAlpha = 1;

  /*
   * The line as it is being drawn.
   *
   * Drawn through the same warp as everything else, so a line dragged past
   * something enormous bends with the space around it - which is honest,
   * because that is the path the blob will actually be moved along.
   */
  if (drawing && drawing.length > 0) {
    ctx.save();
    ctx.strokeStyle = 'rgba(138,220,255,0.85)';
    ctx.lineWidth = 3;
    ctx.setLineDash([10, 8]);
    ctx.lineJoin = 'round';
    ctx.beginPath();

    const head = put(drawing[0].x, drawing[0].y);

    ctx.moveTo(head.x, head.y);

    for (const q of drawing.slice(1)) {
      const at = put(q.x, q.y);
      ctx.lineTo(at.x, at.y);
    }

    const live = aim();

    if (live) {
      const at = put(live.x, live.y);
      ctx.lineTo(at.x, at.y);
    }

    ctx.stroke();
    ctx.restore();
  }

  // under the blobs, because a mark is ground and something may be standing
  // on it already - which is exactly what the person teleporting wants to see
  drawMarks(flat, scale);

  /*
   * Ghosts behind anybody mid-teleport.
   *
   * A blob crossing a thousand units in four tenths of a second is drawn at
   * six or seven places on the way and at no two of them does it overlap
   * itself, so without this it reads as a series of jumps rather than as one
   * fast movement. Sampling where it actually was, out of the frames already
   * kept for interpolating, rather than guessing from a heading: the path is
   * a straight line but the speed along it is the server's.
   */
  const crossing = marks.length ? new Set(marks.map((m) => m.i)) : null;

  if (crossing) {
    for (let back = 3; back >= 1; back--) {
      const was = history[history.length - 1 - back];

      if (!was) continue;

      const fade = (4 - back) / 9;

      for (const b of was.b) {
        if (!crossing.has(b.i)) continue;

        const at = flat(b.x, b.y);

        ctx.globalAlpha = fade;
        ctx.fillStyle = b.c;
        ctx.beginPath();
        ctx.arc(at.x, at.y, b.r * scale * (0.5 + fade), 0, Math.PI * 2);
        ctx.fill();
      }
    }

    ctx.globalAlpha = 1;
  }

  // smallest first, so being swallowed looks like being swallowed
  for (const b of [...now.blobs].sort((a, z) => a.r - z.r)) {
    const at = flat(b.x, b.y);

    const y = at.y;
    const r = b.r * scale;

    // a shadow on the sheet under it, which is what says it is resting on
    // something rather than floating above it
    ctx.fillStyle = 'rgba(0,0,0,0.32)';
    ctx.beginPath();
    ctx.ellipse(at.x, y + r * 0.2, r * 1.06, r * 0.36, 0, 0, Math.PI * 2);
    ctx.fill();

    const stage = stageOf(b.r);

    drawAura(at.x, y, r, stage, b.i, performance.now() / 1000, holeLevel(b.r));

    // the body stays a circle whatever the sheet is doing, because how big
    // somebody is has to be readable at a glance and a squashed one is not
    ctx.fillStyle = stage === 4 ? '#0a0b12' : b.c;
    ctx.beginPath();
    ctx.arc(at.x, y, r, 0, Math.PI * 2);
    ctx.fill();

    ctx.strokeStyle = b.i === me ? '#ffffff' : '#00000055';
    ctx.lineWidth = b.i === me ? 3 : 2;
    ctx.stroke();

    // whoever has just eaten somebody, and is about to try it again
    if (b.s > 0) drawSpark(at.x, y, r, b.s, b.i.charCodeAt(0) * 13 + b.i.charCodeAt(1), performance.now());

    // a ring that tightens as the grip comes up, so it is obvious at a
    // glance who is rooted and how far along they are
    if ((b.g ?? 0) > 1) {
      const wound = Math.min(1, ((b.g ?? 0) - 1) / 2);

      ctx.strokeStyle = `rgba(255,214,99,${0.25 + wound * 0.55})`;
      ctx.lineWidth = 2 + wound * 3;
      ctx.beginPath();
      ctx.arc(at.x, y, r + 14 * scale + (1 - wound) * 26 * scale, 0, Math.PI * 2);
      ctx.stroke();
    }

    if (r > 14) {
      ctx.fillStyle = '#0b0d11';
      ctx.font = `${Math.min(r * 0.55, 22)}px -apple-system, system-ui, sans-serif`;
      ctx.textAlign = 'center';
      ctx.textBaseline = 'middle';
      ctx.fillText(b.n, at.x, y);
    }
  }

  drawDeaths(flat, scale);

  drawChart(world, mine);
}

/**
 * Deep space, and what gravity does to it.
 *
 * There was a grid here. A grid is the easy way to show a surface bending -
 * every line is visibly not straight any more - and it also made the arena
 * look like graph paper, which is the wrong thing for a game about something
 * enormous eating everything near it.
 *
 * The stars do the same job. They are a fixed field in world coordinates, so
 * they hold still while the camera moves over them, and each one is put
 * through the same warp the blobs are - so near a well they slide inwards and
 * smear, which reads as lensing rather than as a dented sheet. The field is
 * generated from a fixed seed, so it is the same field every frame and the
 * same field for everybody.
 */
const STARS = (() => {
  // a small deterministic generator, so the sky does not shimmer between
  // frames and two players describing it are describing the same thing
  let seed = 0x9e3779b9;

  const next = () => {
    seed ^= seed << 13; seed >>>= 0;
    seed ^= seed >> 17;
    seed ^= seed << 5; seed >>>= 0;
    return seed / 0xffffffff;
  };

  return Array.from({ length: 1400 }, () => ({
    x: next(), y: next(),
    size: 0.4 + next() * 1.6,
    lit: 0.25 + next() * 0.75,
    warm: next()
  }));
})();

function drawSpace(put, scale, left, right, top, bottom) {
  const w = world || 1;

  // a few soft clouds, drawn flat: they are far enough away that bending
  // them would say they were close
  for (let i = 0; i < 5; i++) {
    const cx = ((i * 0.37 + 0.11) % 1) * w;
    const cy = ((i * 0.61 + 0.29) % 1) * w;

    if (cx < left - 900 || cx > right + 900 || cy < top - 900 || cy > bottom + 900) continue;

    const at = put(cx, cy);
    const span = (700 + i * 240) * scale;

    const cloud = ctx.createRadialGradient(at.x, at.y, 0, at.x, at.y, span);

    cloud.addColorStop(0, i % 2 ? 'rgba(60,32,86,0.30)' : 'rgba(22,48,88,0.30)');
    cloud.addColorStop(0.55, i % 2 ? 'rgba(42,22,64,0.13)' : 'rgba(16,34,66,0.13)');
    cloud.addColorStop(1, 'rgba(0,0,0,0)');

    ctx.fillStyle = cloud;
    ctx.fillRect(at.x - span, at.y - span, span * 2, span * 2);
  }

  /*
   * Three fills for the whole sky.
   *
   * A star is one or two pixels across, so it is drawn as a square: nobody
   * can tell, and a rectangle costs a fraction of an arc. They are sorted
   * into three brightnesses and each brightness is one path, so the sky is
   * three fills rather than several hundred.
   */
  const sky = [new Path2D(), new Path2D(), new Path2D()];

  for (const s of STARS) {
    const x = s.x * w;
    const y = s.y * w;

    if (x < left || x > right || y < top || y > bottom) continue;

    const at = put(x, y);

    const r = Math.max(0.6, s.size * Math.min(1.5, scale * 1.2));

    const band = s.warm > 0.82 ? 0 : s.warm > 0.62 ? 1 : 2;

    sky[band].rect(at.x - r, at.y - r, r * 2, r * 2);
  }

  const shades = ['rgba(255,214,170,0.78)', 'rgba(196,214,255,0.7)', 'rgba(226,232,246,0.55)'];

  for (let i = 0; i < 3; i++) {
    ctx.fillStyle = shades[i];
    ctx.fill(sky[i]);
  }
}

requestAnimationFrame(draw);

/** Whether this tab means to be in the arena, and how hard reconnecting is. */
let playing = false, tries = 0;

/*
 * The name this player chose, kept here rather than read back out of the box
 * every time. Coming back from being eaten went through the input, and an
 * input that has been emptied - by a reload, by the browser, by anything -
 * sent nothing and the server quite correctly called them "a blob".
 */
let chosen = localStorage.getItem('arena.name') || '';

function connect(name) {
  playing = true;

  chosen = (name ?? '').trim() || chosen;

  if (chosen) {
    try { localStorage.setItem('arena.name', chosen); } catch { /* private mode */ }
  }

  name = chosen;
  const address = new URL('play', location.href.replace(/[^/]*$/, ''));
  address.protocol = address.protocol === 'https:' ? 'wss:' : 'ws:';

  socket = new WebSocket(address);
  socket.onopen = () => socket.send(JSON.stringify({ kind: 'join', name }));

  socket.onmessage = (event) => {
    const m = JSON.parse(event.data);

    if (m.kind === 'welcome') {
      me = m.id; world = m.world; palette = m.palette; wellSize = m.well ?? 40; girths = m.girths ?? [5];
      history.length = 0; camera = null;
      out = false;
      tries = 0;
      $('gate').hidden = true; $('over').hidden = true;
      $('hud').hidden = false; $('board').hidden = false; $('crowd').hidden = false;
      $('map').hidden = false; $('chat').hidden = false;
      return;
    }

    if (m.kind === 'won') {
      $('champion').textContent = `${m.who.replace(/[<&]/g, '')} took the arena`;
      $('tally').textContent = `${tally(m.score)} points.`;
      $('over').hidden = true;
      $('won').hidden = false;
      history.length = 0;
      marks = [];
      whirls = [];
      chart = [];
      deaths.length = 0;
      feed.replaceChildren();
      return;
    }

    if (m.kind === 'said') {
      heard(m.who, m.text, m.who === chosen);

      return;
    }

    if (m.kind === 'eaten') {
      out = true;
      $('verdict').textContent = `Eaten by ${m.by}`;
      $('final').textContent = `You got to ${tally(m.score)}.`;
      $('over').hidden = false;
      camera = null;
      marks = [];
      whirls = [];
      deaths.length = 0;
      feed.replaceChildren();
      return;
    }

    world = m.w ?? world;

    history.push({ at: performance.now(), b: m.b, p: m.p, k: m.k, s: m.s });

    marks = m.v ?? [];
    whirls = m.o ?? [];

    // every third beat, and kept in between
    if (m.m) chart = m.m;

    for (const gone of m.e ?? []) {
      deaths.push({
        x: gone.x, y: gone.y, r: gone.r, c: gone.c,
        at: performance.now(),
        // so that two at once are not the same picture twice
        turn: Math.random() * Math.PI,
      });
    }

    // a tab left in the background wakes to a pile of these
    while (deaths.length > 40) deaths.shift();

    for (const name of m.a ?? []) say(`You ate ${name}`);

    pen = m.draw ?? 0;
    cool = m.cool ?? 0;
    collapsed = !!m.hole;

    // a line begun before the wait came back is not a line
    if (pen > 0) drawing = null;

    // a couple of seconds is far more than DELAY ever reaches back for, and
    // keeps this from growing for as long as somebody leaves the tab open
    while (history.length > 40) history.shift();

    const self = m.b.find((b) => b.i === me);

    // the same sum as the board, which is the server's - the plain area it
    // used to be stops matching the moment anything collapses
    // the server's figure, not one worked out again from a rounded radius -
    // the two disagree by a fraction of a per cent, which is invisible on a
    // blob and glaring when the same number is printed twice on one screen
    $('score').textContent = tally(m.mine ?? 0);
    $('here').textContent = `${m.alive} in the arena`;
    $('hold').hidden = !(self && ((self.g ?? 0) > 1 || collapsed));

    const perk = $('perk');

    perk.hidden = collapsed;
    perk.className = 'streak';
    perk.textContent = pen > 0
      ? `line in ${pen.toFixed(1)}s`
      : touching
        ? 'line ready: press draw, then trace where to go'
        : 'line ready: hold the button and draw a line, let go and run it';

    const grade = self ? holeLevel(self.r) : 0;

    /*
     * One line for whatever this player's button does next, which is not the
     * same button at both ends of a round: a blob teleports and a hole opens
     * whirls, and neither of them can do the other.
     */
    $('warp').textContent = collapsed
      ? cool > 0
        ? `whirl in ${cool.toFixed(1)}s`
        : `whirl ready · ${touching ? 'tap' : 'click'} anywhere to open one`
      : cool > 0
        ? `teleport in ${cool.toFixed(1)}s`
        : `teleport ready${touching ? '' : ' · E'}`;
    $('warp').className = collapsed ? 'collapsed' : cool > 0 ? 'cooling' : 'ready';

    $('hold').textContent = collapsed
      ? `collapsed, grade ${grade}. You cannot move, and everything near you is coming to you.`
      : 'holding on. Everything nearby is coming to you, and it costs you nothing';

    // a hole holds on by being one, so the pull button has nothing to do; the
    // teleport button has nothing to do either, since the floor is the control
    $('pull').hidden = collapsed;
    $('hop').hidden = collapsed;
    $('spend').hidden = collapsed;
    $('spend').textContent = pen > 0 ? `${pen.toFixed(0)}s` : sketching ? 'drawing' : 'draw';

    if (pen > 0 && sketching) {
      sketching = false;
      $('spend').classList.remove('on');
    }

    // a hole cannot do any of the three things the hint offers, and the two
    // lines under it already say what it can do instead
    $('hint').hidden = collapsed;
    $('hint').innerHTML = touching
      ? 'drag to move &middot; eat anything smaller &middot; the buttons do the rest'
      : 'move the mouse &middot; eat anything smaller &middot; hold <kbd>space</kbd> to pull &middot; <kbd>E</kbd> to teleport &middot; drag to draw a line &middot; <kbd>enter</kbd> to talk';

    if (collapsed && aiming) {
      aiming = false;
      $('hop').classList.remove('on');
    }

    const lit = self ? (self.s ?? 0) : 0;

    $('spark').hidden = lit === 0;
    $('spark').textContent = lit === 0 ? '' :
      `charged ${'\u26a1'.repeat(lit)} ${Math.round(lit * 14)}% faster. Eat again before it goes out`;
    $('list').innerHTML = m.l.map((s) =>
      `<li class="${self && s.n === self.n ? 'me' : ''}"><span>${s.n.replace(/[<&]/g, '')}</span><span>${tally(s.s)}</span></li>`).join('');
  };

  /*
   * A dropped socket is picked back up rather than being the end of it.
   *
   * The server faults a connection now and then - two writes landing on one
   * socket at once, which is not something this side can prevent - and the
   * arena carries on without you while you are looking at the word
   * "disconnected". Coming straight back is a couple of seconds of being
   * small again, which is a far better answer than reloading the page.
   *
   * Only while playing, and not after a deliberate close, or somebody who has
   * closed the tab is reconnected to a game they have left.
   */
  socket.onclose = () => {
    if (!playing) {
      $('here').textContent = 'disconnected';
      return;
    }

    $('here').textContent = 'reconnecting…';

    // a little longer each time, so a server that is genuinely gone is not
    // hammered by every tab that was watching it
    tries = Math.min(tries + 1, 5);

    setTimeout(() => { if (playing) connect(name); }, 250 * tries);
  };
}

$('go').onclick = () => connect($('name').value);
$('again').onclick = () => connect($('name').value || chosen);

// so the box is not empty for somebody who has played before
if (chosen) $('name').value = chosen;

// a win does not put anybody out - the next round is already running
/*
 * Company, for an arena with nobody in it.
 *
 * Asked for rather than always there. An arena that fills itself is one that
 * can never be had to yourself, and the first thing anybody does on an empty
 * server is leave - so it is a button, and it is capped server side.
 */
$('crowd').onclick = () => {
  if (socket?.readyState !== 1) return;

  socket.send(JSON.stringify({ kind: 'fill', count: 10 }));
};

$('carry').onclick = () => {
  $('won').hidden = true;

  // somebody who was already out goes back to the screen that has the way
  // back in on it, rather than to an arena they are not in
  if (out) $('over').hidden = false;
};
