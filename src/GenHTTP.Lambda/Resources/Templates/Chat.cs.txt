// A chat room with accounts: register, sign in, and talk to whoever else is here.
//
//   GET  /lambda/{{KEY}}/              the page
//   POST /lambda/{{KEY}}/api/register  make an account
//   POST /lambda/{{KEY}}/api/login     get a token
//   GET  /lambda/{{KEY}}/api/history   what was said before you arrived
//   WS   /lambda/{{KEY}}/room       the room itself, token as the first frame
//
// Accounts and the last fifty lines live in Workspace, the private folder this
// lambda gets, so they outlive a redeployment. Who is signed in does not - that
// is a dictionary in memory, and a restart signs everybody out.

// the platform answers the REST routes in camel case, so the frames this sends
// down the socket have to match or the page would need two shapes for one thing
var json = new JsonSerializerOptions
{
    PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
    PropertyNameCaseInsensitive = true
};

var users = new ConcurrentDictionary<string, Account>(StringComparer.OrdinalIgnoreCase);
var sessions = new ConcurrentDictionary<string, string>(StringComparer.Ordinal);
var room = new ConcurrentDictionary<IReactiveConnection, string>();
var history = new List<Line>();
var writing = new object();

// one broadcast at a time. Two people arriving together means two handlers
// writing to the same sockets at once, and a socket is one stream of frames -
// without this they interleave, the write fails, and the listener is dropped
// for a fault that was ours.
var sending = new SemaphoreSlim(1, 1);

if (Workspace.Exists("users.json"))
{
    foreach (var account in JsonSerializer.Deserialize<List<Account>>(Workspace.ReadText("users.json"), json) ?? [])
    {
        users[account.Name] = account;
    }
}

if (Workspace.Exists("messages.json"))
{
    history.AddRange(JsonSerializer.Deserialize<List<Line>>(Workspace.ReadText("messages.json"), json) ?? []);
}

// PBKDF2 rather than a bare hash. Being an example is not a reason to show
// somebody the wrong way to keep a password.
string Hash(string password, string salt)
    => Convert.ToHexString(System.Security.Cryptography.Rfc2898DeriveBytes.Pbkdf2(
           password, Convert.FromHexString(salt), 60_000,
           System.Security.Cryptography.HashAlgorithmName.SHA256, 32));

string Random(int bytes)
    => Convert.ToHexString(System.Security.Cryptography.RandomNumberGenerator.GetBytes(bytes));

void Remember(Line line)
{
    lock (writing)
    {
        history.Add(line);

        while (history.Count > 50)
        {
            history.RemoveAt(0);
        }

        Workspace.WriteText("messages.json", JsonSerializer.Serialize(history, json));
    }
}

Answer Register(Credentials request)
{
    var name = (request?.Name ?? "").Trim();

    if (name.Length is < 2 or > 20)
    {
        return new Answer(false, "A name is between two and twenty characters.", null, null);
    }

    if ((request?.Password ?? "").Length < 6)
    {
        return new Answer(false, "A password needs six characters at least.", null, null);
    }

    var salt = Random(16);

    if (!users.TryAdd(name, new Account(name, salt, Hash(request!.Password, salt))))
    {
        return new Answer(false, "That name is taken.", null, null);
    }

    lock (writing)
    {
        Workspace.WriteText("users.json", JsonSerializer.Serialize(users.Values.ToList(), json));
    }

    var token = Random(24);

    sessions[token] = name;

    return new Answer(true, null, token, name);
}

Answer Login(Credentials request)
{
    var name = (request?.Name ?? "").Trim();

    // one answer for an unknown name and for a wrong password, so this does
    // not become a way of asking who has an account here
    if (!users.TryGetValue(name, out var account)
        || !System.Security.Cryptography.CryptographicOperations.FixedTimeEquals(
               Encoding.UTF8.GetBytes(Hash(request?.Password ?? "", account.Salt)),
               Encoding.UTF8.GetBytes(account.Hash)))
    {
        return new Answer(false, "That name and password do not go together.", null, null);
    }

    var token = Random(24);

    sessions[token] = account.Name;

    return new Answer(true, null, token, account.Name);
}

async ValueTask Broadcast(Line line)
{
    Remember(line);

    var payload = JsonSerializer.Serialize(line, json);

    await sending.WaitAsync();

    try
    {
        foreach (var listener in room.Keys.ToArray())
        {
            try
            {
                await listener.WritePayloadAsync(payload);
            }
            catch (Exception)
            {
                // this one really is gone - a connection that dies between the
                // snapshot and the write must not stop the other listeners
                room.TryRemove(listener, out _);
            }
        }
    }
    finally
    {
        sending.Release();
    }
}

var api = Inline.Create()
                .Post("register", (Credentials body) => Register(body))
                .Post("login", (Credentials body) => Login(body))
                .Get("history", () => history.ToList())
                .Get("who", () => new Population(room.Count, users.Count));

var socket = Websocket.Functional()
                      .OnConnected(connection => connection.WritePayloadAsync(JsonSerializer.Serialize(
                          new Line("room", "Send your token as the first message.", DateTime.UtcNow), json)))
                      .OnMessage(async (connection, message) =>
                      {
                          var text = (await message.ReadPayloadAsync<string>() ?? "").Trim();

                          /*
                           * The first frame is the token and every frame after
                           * it is talk. The token cannot travel in the query or
                           * a header: once a socket is upgraded the request is
                           * gone, and reading it here throws. A browser cannot
                           * set a header on a websocket handshake anyway.
                           */
                          if (!room.TryGetValue(connection, out var speaker))
                          {
                              if (!sessions.TryGetValue(text, out var name))
                              {
                                  await connection.WritePayloadAsync(JsonSerializer.Serialize(
                                      new Line("room", "That token is not one of mine. Sign in again.", DateTime.UtcNow), json));

                                  return;
                              }

                              room[connection] = name;

                              await Broadcast(new Line("room", $"{name} came in.", DateTime.UtcNow));

                              return;
                          }

                          if (text.Length is 0 or > 500)
                          {
                              return;
                          }

                          await Broadcast(new Line(speaker, text, DateTime.UtcNow));
                      })
                      .OnClose(async (connection, _) =>
                      {
                          if (room.TryRemove(connection, out var name))
                          {
                              await Broadcast(new Line("room", $"{name} left.", DateTime.UtcNow));
                          }
                      });

return Layout.Create()
             .Add("api", api)
             .Add("room", socket)
             .Add(Content.From(Resource.FromString(Page()).Type(new ContentType("text/html; charset=utf-8"))));

record Credentials(string Name, string Password);
record Account(string Name, string Salt, string Hash);
record Answer(bool Ok, string? Problem, string? Token, string? Name);
record Line(string From, string Text, DateTime At);
record Population(int Here, int Registered);

// The page. One file, no build step, no framework - it is here so the example
// is one thing you can read top to bottom rather than a server plus a project.

  .gate button { width: 100%; margin-bottom: 8px; }
  .warn { font-size: 12px; color: var(--dim); margin-top: 14px; }
</style>

<header>
  <h1>Chat</h1>
  <span class="dim" id="status">not signed in</span>
</header>

<div class="gate" id="gate">
  <input id="name" placeholder="Name" autocomplete="username">
  <input id="password" type="password" placeholder="Password" autocomplete="current-password">
  <button id="login">Sign in</button>
  <button id="register" style="background:#5f6368">Make an account</button>
  <div class="problem" id="problem"></div>
  <p class="warn">
    A demo on a shared machine. Do not use a password you use anywhere else - and
    anyone can read what is said here.
  </p>
</div>

<main id="log" hidden></main>

<form id="say" hidden>
  <input id="text" placeholder="Say something" autocomplete="off" maxlength="500">
  <button>Send</button>
</form>

<script type="module">
const $ = (id) => document.getElementById(id);
const base = location.pathname.replace(/\/$/, '');
let socket = null;

const escape = (s) => s.replace(/[&<>"]/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]));

function add(line) {
  const at = new Date(line.at + 'Z');
  const room = line.from === 'room';
  $('log').insertAdjacentHTML('beforeend',
    `<div class="line ${room ? 'room' : ''}">` +
    (room ? '' : `<span class="who">${escape(line.from)}</span> `) +
    escape(line.text) +
    `<span class="when">${at.toLocaleTimeString()}</span></div>`);
  $('log').scrollTop = $('log').scrollHeight;
}

async function send(path, body) {
  const response = await fetch(`${base}/api/${path}`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(body),
  });
  return response.json();
}

async function enter(path) {
  $('problem').textContent = '';
  const answer = await send(path, { name: $('name').value, password: $('password').value });
  if (!answer.ok) { $('problem').textContent = answer.problem; return; }

  $('gate').hidden = true;
  $('log').hidden = false;
  $('say').hidden = false;
  $('status').textContent = `signed in as ${answer.name}`;

  for (const line of await (await fetch(`${base}/api/history`)).json()) add(line);

  const address = new URL(`${base}/room`, location.href);
  address.protocol = address.protocol === 'https:' ? 'wss:' : 'ws:';
  socket = new WebSocket(address);
  // the first frame is the token; the request is gone by the time the socket
  // is open, so there is nowhere else to put it
  socket.onopen = () => socket.send(answer.token);
  socket.onmessage = (event) => add(JSON.parse(event.data));
  socket.onclose = () => { $('status').textContent = 'disconnected'; };
  $('text').focus();
}

$('login').onclick = () => enter('login');
$('register').onclick = () => enter('register');
$('say').onsubmit = (event) => {
  event.preventDefault();
  const text = $('text').value.trim();
  if (text && socket?.readyState === 1) { socket.send(text); $('text').value = ''; }
};
</script>
</html>
""";
