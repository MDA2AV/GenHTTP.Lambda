// The page it is served as.

static class Site
{
    public const string Page =
"""
<!doctype html>
<html lang="en">
<meta charset="utf-8">
<meta name="viewport" content="width=device-width,initial-scale=1">
<title>Guestbook</title>
<style>
  :root { color-scheme: light dark; --line: #d5d7da; --dim: #5f6368; }
  * { box-sizing: border-box; }
  body { margin: 0 auto; max-width: 38rem; padding: 3rem 1.25rem;
         font: 15px/1.6 -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, sans-serif; }
  h1 { font-size: 20px; margin: 0 0 .25rem; }
  p.dim { color: var(--dim); margin-top: 0; }
  form { margin: 1.5rem 0; display: grid; gap: .5rem; }
  input, textarea { padding: .6rem .7rem; border: 1px solid var(--line); background: transparent;
                    color: inherit; font: inherit; width: 100%; }
  textarea { resize: vertical; min-height: 4.5rem; }
  button { padding: .6rem 1.1rem; border: 0; background: #1a73e8; color: #fff; font: inherit;
           cursor: pointer; justify-self: start; }
  .entry { border-top: 1px solid var(--line); padding: .9rem 0; }
  .who { font-weight: 600; }
  .when { color: var(--dim); font-size: 12px; margin-left: .4rem; }
  .what { margin: .2rem 0 0; white-space: pre-wrap; word-break: break-word; }
  .problem { color: #c5221f; font-size: 13px; }
  .count { color: var(--dim); font-size: 13px; }
</style>

<h1>Guestbook</h1>
<p class="dim">Leave a line. Everyone can read it, and there is no way to take it back.</p>

<form id="sign">
  <input id="name" maxlength="24" placeholder="Your name" required>
  <textarea id="message" maxlength="280" placeholder="Say something" required></textarea>
  <button>Sign it</button>
</form>

<p class="problem" id="problem"></p>
<p class="count" id="count"></p>

<div id="entries"></div>

<script type="module">
const base = location.pathname.replace(/\/$/, '');
const $ = (id) => document.getElementById(id);
const escape = (s) => s.replace(/[&<>"]/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]));

async function show() {
  const entries = await (await fetch(`${base}/api/entries`)).json();

  $('count').textContent = entries.length === 0
    ? 'Nobody has signed it yet.'
    : `${entries.length} ${entries.length === 1 ? 'line' : 'lines'}, newest first.`;

  $('entries').innerHTML = entries.map((e) => `
    <div class="entry">
      <span class="who">${escape(e.name)}</span>
      <span class="when">${new Date(e.at + 'Z').toLocaleString()}</span>
      <p class="what">${escape(e.message)}</p>
    </div>`).join('');
}

$('sign').onsubmit = async (event) => {
  event.preventDefault();
  $('problem').textContent = '';

  const answer = await (await fetch(`${base}/api/entries`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ name: $('name').value, message: $('message').value }),
  })).json();

  if (!answer.ok) { $('problem').textContent = answer.problem; return; }

  $('message').value = '';
  show();
};

show();
</script>
</html>
""";
}
