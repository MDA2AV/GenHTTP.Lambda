// What the two halves say to each other.
//
// Every frame on the socket is one of these, tagged by its `kind`. The maze is
// sent once, when you arrive; everything that moves is sent on every tick.

/// <summary>Sent by a browser as its first frame.</summary>
public record Join(string Kind, string Name);

/// <summary>
/// What a browser is doing with the controls, sent many times a second.
/// </summary>
/// <remarks>
/// Intent rather than position. The client says "forward and left, turret at
/// this angle, firing"; where that puts the tank is the server's business, so
/// an altered client can ask to drive through a wall and simply not.
/// </remarks>
public record Input(string Kind, double Throttle, double Steer, double Aim, bool Fire);

/// <summary>One tank, as everybody sees it.</summary>
public record Tank(string Id, string Name, string Colour, double X, double Z,
                   double Heading, double Aim, int Health, int Score, bool Alive, double Boost);

/// <summary>A shell in flight.</summary>
public record Shell(string Id, double X, double Z, double Heading);

/// <summary>Something on the floor worth driving over.</summary>
public record Pickup(string Id, string Sort, double X, double Z);

/// <summary>Who you are and what the room looks like, sent once.</summary>
public record Welcome(string Kind, string Id, string Colour, int Width, int Height, double Cell, string Walls);

/// <summary>Where everything is, sent on every tick to everybody.</summary>
public record Snapshot(string Kind, List<Tank> Tanks, List<Shell> Shells, List<Pickup> Pickups);

/// <summary>Something worth a line in the corner of the screen.</summary>
public record Event(string Kind, string Text);

/// <summary>The best rounds the room remembers.</summary>
public record Standing(string Name, int Score, DateTime At);
