// The browser half: draws the room, sends where this player would like to be,
// and believes whatever the server says about everyone including itself.

static class Site
{
    public const string Page =
"""
<!doctype html>
<html lang="en">
<meta charset="utf-8">
<meta name="viewport" content="width=device-width,initial-scale=1">
<title>Arena</title>
<style>
  :root { color-scheme: dark; }
  * { box-sizing: border-box; }
  body { margin: 0; background: #0d0f13; color: #e8eaed; overflow: hidden;
         font: 15px/1.5 -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, sans-serif; }
  canvas { display: block; }
  #hud { position: fixed; top: 14px; left: 16px; }
  #hud b { font-size: 24px; font-variant-numeric: tabular-nums; }
  .dim { color: #9aa0a6; font-size: 13px; }
  #here { position: fixed; top: 14px; right: 16px; width: 200px; background: #171a1fdd; padding: 10px 12px; }
  #here h2 { font-size: 11px; margin: 0 0 6px; color: #9aa0a6; text-transform: uppercase; letter-spacing: .06em; }
  #here li { display: flex; justify-content: space-between; gap: 8px; list-style: none; align-items: center; }
  #here ul { margin: 0; padding: 0; }
  .dot { width: 9px; height: 9px; display: inline-block; margin-right: 6px; }
  #gate { position: fixed; inset: 0; display: grid; place-items: center; background: #0d0f13; }
  /* a display of its own beats the hidden attribute, so say so explicitly */
  [hidden] { display: none !important; }
  #gate div { text-align: center; }
  input { padding: 9px 11px; border: 1px solid #3c4043; background: #171a1f; color: inherit; font: inherit; }
  button { padding: 9px 18px; border: 0; background: #1a73e8; color: #fff; font: inherit; cursor: pointer; }
</style>

<div id="gate"><div>
  <h1>Arena</h1>
  <p class="dim">Everybody who opens this page is in the same room.<br>Arrow keys or drag to move. Run into a cube.</p>
  <p><input id="name" maxlength="14" placeholder="Your name"> <button id="go">Go in</button></p>
</div></div>

<div id="hud" hidden><b id="score">0</b><div class="dim">arrow keys, or drag</div></div>
<div id="here" hidden><h2>In the room</h2><ul id="list"></ul></div>

<script type="module">
import * as THREE from 'https://cdn.jsdelivr.net/npm/three@0.170.0/build/three.module.js';

const base = location.pathname.replace(/\/$/, '');
const $ = (id) => document.getElementById(id);

let socket = null, me = null, half = 14, want = { x: 0, z: 0 }, known = new Map(), gems = new Map();

const scene = new THREE.Scene();
scene.fog = new THREE.Fog(0x0d0f13, 26, 62);

const camera = new THREE.PerspectiveCamera(55, innerWidth / innerHeight, 0.1, 200);
const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
renderer.setSize(innerWidth, innerHeight);
document.body.appendChild(renderer.domElement);

scene.add(new THREE.HemisphereLight(0x9fc0ff, 0x202124, 1.2));
const sun = new THREE.DirectionalLight(0xffffff, 1.2);
sun.position.set(6, 14, 8);
scene.add(sun);

const floor = new THREE.Mesh(new THREE.BoxGeometry(1, 0.5, 1),
                             new THREE.MeshStandardMaterial({ color: 0x1b1f26 }));
scene.add(floor);

const grid = new THREE.GridHelper(1, 14, 0x3c4043, 0x2a2e35);
scene.add(grid);

const bodyGeometry = new THREE.BoxGeometry(1, 1, 1);
const gemGeometry = new THREE.OctahedronGeometry(0.45);
const gemMaterial = new THREE.MeshStandardMaterial({ color: 0xfdd663, emissive: 0x6b5200 });

function sized() {
  floor.scale.set(half * 2 + 2, 1, half * 2 + 2);
  floor.position.y = -0.75;
  grid.scale.set(half * 2, 1, half * 2);
  grid.position.y = -0.49;
}

addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
});

const held = new Set();
addEventListener('keydown', (e) => held.add(e.key));
addEventListener('keyup', (e) => held.delete(e.key));

let dragging = null;
addEventListener('pointerdown', (e) => { dragging = { x: e.clientX, y: e.clientY }; });
addEventListener('pointermove', (e) => {
  if (!dragging || !me) return;
  want.x += (e.clientX - dragging.x) * 0.02;
  want.z += (e.clientY - dragging.y) * 0.02;
  dragging = { x: e.clientX, y: e.clientY };
});
addEventListener('pointerup', () => { dragging = null; });

function paint(state) {
  const seen = new Set();

  for (const player of state.players) {
    seen.add(player.id);

    let mesh = known.get(player.id);

    if (!mesh) {
      mesh = new THREE.Mesh(bodyGeometry, new THREE.MeshStandardMaterial({ color: player.colour, roughness: 0.35 }));
      scene.add(mesh);
      known.set(player.id, mesh);
    }

    // the server said so, so this is where they are - the only smoothing is
    // the walk between the last position and this one
    mesh.userData.target = { x: player.x, z: player.z };

    if (player.id === me) {
      $('score').textContent = player.score;
      // somebody else may have moved us; keep asking from where we are
      if (!dragging && held.size === 0) { want.x = player.x; want.z = player.z; }
    }
  }

  for (const [id, mesh] of known) {
    if (!seen.has(id)) { scene.remove(mesh); known.delete(id); }
  }

  const standing = new Set();

  for (const prize of state.prizes) {
    standing.add(prize.id);

    if (!gems.has(prize.id)) {
      const gem = new THREE.Mesh(gemGeometry, gemMaterial);
      gem.position.set(prize.x, 0.3, prize.z);
      scene.add(gem);
      gems.set(prize.id, gem);
    }
  }

  for (const [id, gem] of gems) {
    if (!standing.has(id)) { scene.remove(gem); gems.delete(id); }
  }

  $('list').innerHTML = state.players.map((p) =>
    `<li><span><span class="dot" style="background:${p.colour}"></span>${p.name.replace(/[<&]/g, '')}</span><span>${p.score}</span></li>`
  ).join('');
}

function frame(now) {
  requestAnimationFrame(frame);

  if (me) {
    const speed = 0.22;
    if (held.has('ArrowLeft')) want.x -= speed;
    if (held.has('ArrowRight')) want.x += speed;
    if (held.has('ArrowUp')) want.z -= speed;
    if (held.has('ArrowDown')) want.z += speed;

    want.x = Math.max(-half, Math.min(half, want.x));
    want.z = Math.max(-half, Math.min(half, want.z));
  }

  for (const mesh of known.values()) {
    const target = mesh.userData.target;
    if (target) {
      mesh.position.x += (target.x - mesh.position.x) * 0.25;
      mesh.position.z += (target.z - mesh.position.z) * 0.25;
      mesh.position.y = 0.5;
    }
  }

  for (const gem of gems.values()) {
    gem.rotation.y = now / 600;
    gem.position.y = 0.35 + Math.sin(now / 400) * 0.12;
  }

  const mine = known.get(me);
  const look = mine ? mine.position : new THREE.Vector3(0, 0, 0);
  camera.position.set(look.x, 17, look.z + 19);
  camera.lookAt(look.x, 0, look.z);

  renderer.render(scene, camera);
}

requestAnimationFrame(frame);

// the server is told where we would like to be, at a rate that is about the
// refresh of a screen rather than every frame of one
setInterval(() => {
  if (socket?.readyState === 1 && me) {
    socket.send(JSON.stringify({ kind: 'move', x: want.x, z: want.z }));
  }
}, 60);

$('go').onclick = () => {
  const address = new URL(`${base}/join`, location.href);
  address.protocol = address.protocol === 'https:' ? 'wss:' : 'ws:';

  socket = new WebSocket(address);

  socket.onopen = () => socket.send(JSON.stringify({ kind: 'join', name: $('name').value }));

  socket.onmessage = (event) => {
    const message = JSON.parse(event.data);

    if (message.kind === 'welcome') {
      me = message.id;
      half = message.half;
      sized();
      $('gate').hidden = true;
      $('hud').hidden = false;
      $('here').hidden = false;
    } else if (message.kind === 'state') {
      paint(message);
    }
  };

  socket.onclose = () => { $('hud').innerHTML = '<b>disconnected</b>'; };
};
</script>
</html>
""";
}
