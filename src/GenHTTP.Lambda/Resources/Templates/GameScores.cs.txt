// The scoreboard, and what it will accept onto itself.
//
// Nothing here trusts the browser. A score arrives as a number somebody could
// have typed themselves, so it is bounded, and the board is capped so the file
// behind it cannot grow without end. That is about as far as this can go
// without running the game on the server too, which would be a different
// example.

/// <summary>What a browser claims it scored.</summary>
public record Attempt(string Name, int Points);

/// <summary>A run that made it onto the board.</summary>
public record Score(string Name, int Points, DateTime At);

/// <summary>The answer to a submission, with the board as it now stands.</summary>
public record Posted(bool Ok, string Problem, List<Score> Board);

static class Scoreboard
{
    /// <summary>The highest a run can claim before it stops being believable.</summary>
    public const int Ceiling = 100_000;

    /// <summary>How many runs are kept at all.</summary>
    public const int Keep = 200;

    /// <summary>How many are shown.</summary>
    public const int Shown = 10;

    /// <summary>The best runs, best first, oldest winning a tie.</summary>
    public static List<Score> Best(List<Score> board)
        => board.OrderByDescending(s => s.Points).ThenBy(s => s.At).Take(Shown).ToList();

    /// <summary>
    /// Whether a submission may go on the board, and why not when it may not.
    /// </summary>
    public static string Refuse(Attempt attempt)
    {
        var name = (attempt?.Name ?? "").Trim();

        if (name.Length is < 1 or > 16)
        {
            return "A name is between one and sixteen characters.";
        }

        if (attempt.Points is < 0 or > Ceiling)
        {
            return "That is not a score this game can produce.";
        }

        return null;
    }

    /// <summary>Trims the board back to what is kept.</summary>
    public static List<Score> Trim(List<Score> board)
        => board.OrderByDescending(s => s.Points).ThenBy(s => s.At).Take(Keep).ToList();
}
