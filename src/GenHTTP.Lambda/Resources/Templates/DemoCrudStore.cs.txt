// The tasks, kept in the lambda's database: a table the migration in
// migrations/ makes, read and written through Entity Framework Core.
//
//   - TaskContext maps the table onto TaskItem. It makes no schema: the table
//     is the migration's, and the context only says how to read it.
//   - Every call makes a context on a connection of its own -
//     new TaskContext(Database.GetConnection()) - and disposes of it, which
//     disposes of the connection too. Connections are pooled, so that costs
//     nothing, and a context is never shared between two requests at once.
//   - Every call is synchronous - ToList, Find, SaveChanges, not their Async
//     forms. The database is a file on the same machine, and the server this
//     runs on has an asynchronous model of its own that a blocking call does
//     not disturb.
//   - Queries are LINQ, so what somebody types goes in as a parameter and
//     cannot change what the statement does.
//   - At most `limit` tasks are kept, the oldest going first. Anybody can
//     write to a demo, so it must not grow without bound. For your own, the
//     real limit is the room the database may take, which read_database says.

using System.Globalization;

/// <summary>A stored task, as the table holds it and the API sends it.</summary>
public sealed class TaskItem
{
    public string Id { get; set; }

    public string Title { get; set; }

    public string Notes { get; set; } = "";

    public bool Done { get; set; }

    public DateTime Created { get; set; }

    public DateTime Updated { get; set; }
}

/// <summary>The lambda's database, as Entity Framework Core sees it.</summary>
public sealed class TaskContext(SqliteConnection connection) : DbContext
{
    public DbSet<TaskItem> Tasks => Set<TaskItem>();

    // the context closes the connection it was handed when it is disposed of;
    // inside a DbContext, Database is the context's own, which is why the
    // connection comes in from outside
    protected override void OnConfiguring(DbContextOptionsBuilder options)
        => options.UseSqlite(connection, contextOwnsConnection: true);

    protected override void OnModelCreating(ModelBuilder model)
    {
        // SQLite does not mind the case of a name, so Title is the column
        // title without saying so - only the table needs naming
        var task = model.Entity<TaskItem>().ToTable("tasks");

        // times as ISO 8601 text in UTC: they sort as they should, read well
        // in the editor's view of the database, and come back as the time
        // they were - Entity Framework's own format would lose the UTC
        task.Property(t => t.Created).HasConversion(t => Write(t), t => Read(t));
        task.Property(t => t.Updated).HasConversion(t => Write(t), t => Read(t));
    }

    private static string Write(DateTime value) => value.ToUniversalTime().ToString("O", CultureInfo.InvariantCulture);

    private static DateTime Read(string value) => DateTime.Parse(value, CultureInfo.InvariantCulture, DateTimeStyles.RoundtripKind);
}

public sealed class TaskStore(int limit)
{
    public int Count()
    {
        using var db = new TaskContext(Database.GetConnection());

        return db.Tasks.Count();
    }

    /// <summary>Every task, open ones first and the newest of each first - or those matching a search.</summary>
    public List<TaskItem> List(string search)
    {
        using var db = new TaskContext(Database.GetConnection());

        // read only, so the context need not keep track of what it handed out
        var tasks = db.Tasks.AsNoTracking();

        if (!string.IsNullOrWhiteSpace(search))
        {
            // a search for "50%" looks for "50%", not for anything starting with 50
            var term = "%" + search.Trim().Replace("\\", "\\\\").Replace("%", "\\%").Replace("_", "\\_") + "%";

            // LIKE rather than Contains, which SQLite would compare case by case
            tasks = tasks.Where(t => EF.Functions.Like(t.Title, term, "\\") || EF.Functions.Like(t.Notes, term, "\\"));
        }

        return tasks.OrderBy(t => t.Done).ThenByDescending(t => t.Created).ToList();
    }

    public TaskItem Find(string id)
    {
        using var db = new TaskContext(Database.GetConnection());

        return db.Tasks.AsNoTracking().FirstOrDefault(t => t.Id == id);
    }

    public void Add(TaskItem task)
    {
        using var db = new TaskContext(Database.GetConnection());

        // both or neither: the task is not added if making room fails
        using var transaction = db.Database.BeginTransaction();

        db.Tasks.Add(task);
        db.SaveChanges();

        // everything past the newest `limit` goes, in one statement
        db.Tasks.OrderByDescending(t => t.Created).Skip(limit).ExecuteDelete();

        transaction.Commit();
    }

    /// <summary>Changes a task, and hands it back - or nothing, where there is none.</summary>
    public TaskItem Change(string id, string title, string notes, bool done)
    {
        using var db = new TaskContext(Database.GetConnection());

        // tracked, so SaveChanges writes what changed on it and nothing else
        var task = db.Tasks.Find(id);

        if (task == null)
        {
            return null;
        }

        task.Title = title;
        task.Notes = notes;
        task.Done = done;
        task.Updated = DateTime.UtcNow;

        db.SaveChanges();

        return task;
    }

    /// <summary>Removes a task, and says whether there was one.</summary>
    public bool Remove(string id)
    {
        using var db = new TaskContext(Database.GetConnection());

        // a DELETE straight away, without reading the task first
        return db.Tasks.Where(t => t.Id == id).ExecuteDelete() > 0;
    }
}
