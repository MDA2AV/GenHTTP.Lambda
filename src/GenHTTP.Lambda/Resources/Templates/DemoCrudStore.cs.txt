// The tasks, kept in the lambda's database: a table the migration in
// migrations/ makes, read and written with plain SQL.
//
//   - Every call opens a connection with Database.GetConnection() and disposes
//     of it when it is done. Connections are pooled, so that costs nothing,
//     and a connection is never shared between two requests at once.
//   - Every call is synchronous - ExecuteReader, not ExecuteReaderAsync. The
//     database is a file on the same machine, and the server this runs on has
//     an asynchronous model of its own that a blocking call does not disturb.
//   - Values go in as parameters ($title), never pasted into the SQL, so what
//     somebody types cannot change what the statement does.
//   - At most `limit` tasks are kept, the oldest going first. Anybody can
//     write to a demo, so it must not grow without bound. For your own, the
//     real limit is the room the database may take, which read_database says.

using System.Globalization;

public sealed class TaskStore(int limit)
{
    private const string Columns = "id, title, notes, done, created, updated";

    public long Count()
    {
        using var db = Database.GetConnection();
        using var command = db.CreateCommand();

        command.CommandText = "SELECT count(*) FROM tasks";

        return (long)command.ExecuteScalar();
    }

    /// <summary>Every task, open ones first and the newest of each first - or those matching a search.</summary>
    public List<TaskItem> List(string search)
    {
        using var db = Database.GetConnection();
        using var command = db.CreateCommand();

        command.CommandText = $"SELECT {Columns} FROM tasks WHERE $term IS NULL OR title LIKE $term ESCAPE '\\' OR notes LIKE $term ESCAPE '\\' ORDER BY done, created DESC";

        // a search for "50%" looks for "50%", not for anything starting with 50
        var term = string.IsNullOrWhiteSpace(search)
            ? null
            : "%" + search.Trim().Replace("\\", "\\\\").Replace("%", "\\%").Replace("_", "\\_") + "%";

        command.Parameters.AddWithValue("$term", (object)term ?? DBNull.Value);

        return Read(command);
    }

    public TaskItem Find(string id)
    {
        using var db = Database.GetConnection();
        using var command = db.CreateCommand();

        command.CommandText = $"SELECT {Columns} FROM tasks WHERE id = $id";
        command.Parameters.AddWithValue("$id", id);

        return Read(command).FirstOrDefault();
    }

    public void Add(TaskItem task)
    {
        using var db = Database.GetConnection();

        // both statements or neither: the task is not added if making room fails
        using var transaction = db.BeginTransaction();

        using (var insert = db.CreateCommand())
        {
            insert.CommandText = $"INSERT INTO tasks ({Columns}) VALUES ($id, $title, $notes, $done, $created, $updated)";

            Bind(insert, task);

            insert.ExecuteNonQuery();
        }

        using (var trim = db.CreateCommand())
        {
            trim.CommandText = "DELETE FROM tasks WHERE id NOT IN (SELECT id FROM tasks ORDER BY created DESC LIMIT $limit)";
            trim.Parameters.AddWithValue("$limit", limit);

            trim.ExecuteNonQuery();
        }

        transaction.Commit();
    }

    /// <summary>Replaces a task, and says whether there was one to replace.</summary>
    public bool Replace(TaskItem task)
    {
        using var db = Database.GetConnection();
        using var command = db.CreateCommand();

        command.CommandText = "UPDATE tasks SET title = $title, notes = $notes, done = $done, created = $created, updated = $updated WHERE id = $id";

        Bind(command, task);

        return command.ExecuteNonQuery() > 0;
    }

    /// <summary>Removes a task, and says whether there was one.</summary>
    public bool Remove(string id)
    {
        using var db = Database.GetConnection();
        using var command = db.CreateCommand();

        command.CommandText = "DELETE FROM tasks WHERE id = $id";
        command.Parameters.AddWithValue("$id", id);

        return command.ExecuteNonQuery() > 0;
    }

    private static void Bind(SqliteCommand command, TaskItem task)
    {
        command.Parameters.AddWithValue("$id", task.Id);
        command.Parameters.AddWithValue("$title", task.Title);
        command.Parameters.AddWithValue("$notes", task.Notes);
        command.Parameters.AddWithValue("$done", task.Done);
        command.Parameters.AddWithValue("$created", Time(task.Created));
        command.Parameters.AddWithValue("$updated", Time(task.Updated));
    }

    private static List<TaskItem> Read(SqliteCommand command)
    {
        using var reader = command.ExecuteReader();

        var tasks = new List<TaskItem>();

        while (reader.Read())
        {
            tasks.Add(new TaskItem(reader.GetString(0), reader.GetString(1), reader.GetString(2), reader.GetBoolean(3),
                                   Time(reader.GetString(4)), Time(reader.GetString(5))));
        }

        return tasks;
    }

    // times as ISO 8601 text in UTC: they sort as they should, read well in
    // the editor's view of the database, and come back as the time they were
    private static string Time(DateTime value) => value.ToUniversalTime().ToString("O", CultureInfo.InvariantCulture);

    private static DateTime Time(string value) => DateTime.Parse(value, CultureInfo.InvariantCulture, DateTimeStyles.RoundtripKind);
}
