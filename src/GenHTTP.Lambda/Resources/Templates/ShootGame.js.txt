// The browser half of Headwall.
//
// three.js comes from a CDN rather than being vendored: a lambda is code you
// can read in one sitting, and a megabyte of somebody else's minified library
// pasted into it would not be.
//
// Three things are going on at once in here and they are worth separating
// before reading it.
//
// The server owns the game. It runs twenty ticks a second and its word is
// final, so everything below either draws what it said or predicts what it is
// about to say.
//
// Other people are drawn a tenth of a second in the past, interpolated
// between the two frames either side of that moment. Drawing them at the
// newest frame would mean drawing them wrong between frames and snapping them
// every time one arrived.
//
// You are drawn in the present. Waiting for the server to agree that you
// pressed W would put your own movement a round trip behind your hands, which
// is the difference between a shooter that feels good and one that does not,
// so the movement below is a line-for-line port of the server's and runs here
// as well. When a frame arrives it says which of your inputs it had seen; the
// rest are replayed on top of the position it sent, and any disagreement is
// smoothed away over a few frames rather than snapped.

import * as THREE from 'https://cdn.jsdelivr.net/npm/three@0.170.0/build/three.module.js';

/* ------------------------------------------------------------------ rules */

// Everything in this block is the server's, copied. If a constant changes in
// World.cs it has to change here too, or the prediction drifts and the
// crosshair starts lying about the spread.
const R = {
  pace: 420, quicken: 3500, drag: 4000, brake: 8000,
  fall: 1800, leap: 550, float: 0.1, fold: 0.1, stoop: 0.5,
  girth: 30, tall: 180, squat: 120, eyeTall: 165, eyeSquat: 105, skull: 30,
  cadence: 0.09, magazine: 20, reload: 1.8, settle: 0.35,
  tick: 0.05, behind: 0.1
};

/*
 * Both halves of the game agree that a yaw of zero looks down negative Z and
 * that ninety looks along positive X, which is World.Facing:
 *
 *     (sin yaw, sin pitch, -cos yaw)
 *
 * three.js does not. An object rotated by theta about Y points along
 *
 *     (-sin theta, 0, -cos theta)
 *
 * which is the same direction only when theta is a multiple of a half turn.
 * Anywhere else the two are mirrored through the Z axis, so a camera handed
 * the yaw raw looks one way while the shot the server judges goes the other -
 * and at the two angles anybody naturally tests at, straight up the map and
 * straight down it, they agree exactly and nothing looks wrong.
 *
 * So every three.js rotation about Y in here takes the negated yaw, and this
 * is the only place that says so.
 */
const turned = yaw => -yaw * Math.PI / 180;

const spread = (speed, airborne, crouched, burst) => {
  let cone = 0.05 + 1.2 * Math.min(1, speed / R.pace) + (airborne ? 3 : 0);
  cone += Math.min(0.9, 0.15 * burst);
  return crouched ? cone * 0.6 : cone;
};

const World = {
  facing(yaw, pitch) {
    const y = yaw * Math.PI / 180, p = pitch * Math.PI / 180;
    const flat = Math.cos(p);
    return { x: Math.sin(y) * flat, y: Math.sin(p), z: -Math.cos(y) * flat };
  }
};

const climb = burst => ({
  pitch: Math.min(3.6, 0.6 * burst),
  yaw: 0.35 * Math.sin(0.9 * burst)
});

const BTN = { fwd: 1, back: 2, left: 4, right: 8, jump: 16, crouch: 32, fire: 64, reload: 128 };
const FLAG = { crouched: 1, airborne: 2, sheltered: 4, firing: 8, dead: 16 };

const CM = 0.01;                       // the world is in centimetres, three.js in metres

// filled in from the welcome, so that the two sides are one colour rather
// than one colour here and another one in the server's idea of them
const TEAM = [0x4a9eff, 0xffa22b];

const $ = id => document.getElementById(id);
const clamp = (v, lo, hi) => v < lo ? lo : v > hi ? hi : v;

renderer.setPixelRatio(Math.min(devicePixelRatio, 2));

/*
 * Shooters quote a horizontal field of view and three.js takes a vertical
 * one, so the ninety everybody means is not the ninety this constructor
 * wants. It is converted on every resize, because the conversion depends on
 * the shape of the window.
 */
const FOV = 90;

const camera = new THREE.PerspectiveCamera(59, 1, 0.04, 140);

// the gun is drawn by a second camera on top of the first, so that it cannot
// clip through a wall you are standing against and cannot be fogged
const overlay = new THREE.Scene();
const handCam = new THREE.PerspectiveCamera(60, 1, 0.005, 3);

function resize() {
  const w = innerWidth, h = innerHeight;

  renderer.setSize(w, h, false);

  camera.aspect = handCam.aspect = w / h;

  // horizontal to vertical, for whatever shape the window happens to be
  camera.fov = 2 * Math.atan(Math.tan(FOV * Math.PI / 360) / camera.aspect) * 180 / Math.PI;

  camera.updateProjectionMatrix();
  handCam.updateProjectionMatrix();
}

addEventListener('resize', resize);

/* --------------------------------------------------------------- textures */

// Everything drawn here is made in a canvas rather than downloaded. A floor
// with nothing on it gives the eye no way to judge speed, and speed is what
// the whole game is about, so the floor gets a grid whether or not there is
// an image to put on it.

function paint(size, draw) {
  const c = document.createElement('canvas');
  c.width = c.height = size;
  draw(c.getContext('2d'), size);
  const t = new THREE.CanvasTexture(c);
  t.colorSpace = THREE.SRGBColorSpace;
  t.anisotropy = 8;
  return t;
}

const floorTexture = paint(512, (g, s) => {
  g.fillStyle = '#10161e';
  g.fillRect(0, 0, s, s);

  // a fine speckle, so that a flat colour does not band across the screen
  const grain = g.createImageData(s, s);
  for (let i = 0; i < grain.data.length; i += 4) {
    const n = 118 + ((Math.random() * 16) | 0);
    grain.data[i] = grain.data[i + 1] = grain.data[i + 2] = n;
    grain.data[i + 3] = 12;
  }
  g.putImageData(grain, 0, 0);

  g.strokeStyle = 'rgba(120,160,200,.14)';
  g.lineWidth = 2;
  g.strokeRect(1, 1, s - 2, s - 2);

  g.strokeStyle = 'rgba(120,160,200,.05)';
  g.lineWidth = 1;
  for (let i = 1; i < 4; i++) {
    const at = (s / 4) * i;
    g.beginPath(); g.moveTo(at, 0); g.lineTo(at, s); g.stroke();
    g.beginPath(); g.moveTo(0, at); g.lineTo(s, at); g.stroke();
  }
});

floorTexture.wrapS = floorTexture.wrapT = THREE.RepeatWrapping;
floorTexture.repeat.set(26, 39);

const concreteTexture = paint(256, (g, s) => {
  g.fillStyle = '#9aa1a9';
  g.fillRect(0, 0, s, s);
  for (let i = 0; i < 2600; i++) {
    const v = 150 + ((Math.random() * 60) | 0);
    g.fillStyle = `rgba(${v},${v + 3},${v + 8},.35)`;
    g.fillRect(Math.random() * s, Math.random() * s, 2, 2);
  }
  g.strokeStyle = 'rgba(60,66,74,.25)';
  g.lineWidth = 3;
  g.strokeRect(1.5, 1.5, s - 3, s - 3);
});

concreteTexture.wrapS = concreteTexture.wrapT = THREE.RepeatWrapping;

const sparkTexture = paint(64, (g, s) => {
  const r = g.createRadialGradient(s / 2, s / 2, 0, s / 2, s / 2, s / 2);
  r.addColorStop(0, 'rgba(255,255,255,1)');
  r.addColorStop(.35, 'rgba(255,214,140,.75)');
  r.addColorStop(1, 'rgba(255,150,40,0)');
  g.fillStyle = r;
  g.fillRect(0, 0, s, s);
});

const holeTexture = paint(64, (g, s) => {
  g.clearRect(0, 0, s, s);
  const r = g.createRadialGradient(s / 2, s / 2, 1, s / 2, s / 2, s / 2);
  r.addColorStop(0, 'rgba(8,9,12,.95)');
  r.addColorStop(.45, 'rgba(20,22,26,.72)');
  r.addColorStop(.75, 'rgba(120,124,130,.28)');
  r.addColorStop(1, 'rgba(120,124,130,0)');
  g.fillStyle = r;
  g.beginPath(); g.arc(s / 2, s / 2, s / 2, 0, 7); g.fill();
});

const flashTexture = paint(128, (g, s) => {
  const r = g.createRadialGradient(s / 2, s / 2, 0, s / 2, s / 2, s / 2);
  r.addColorStop(0, 'rgba(255,255,245,1)');
  r.addColorStop(.2, 'rgba(255,226,150,.9)');
  r.addColorStop(.55, 'rgba(255,150,40,.35)');
  r.addColorStop(1, 'rgba(255,120,0,0)');
  g.fillStyle = r;
  g.fillRect(0, 0, s, s);
  // four spikes, so it is a muzzle flash and not a dot
  g.strokeStyle = 'rgba(255,240,200,.85)';
  g.lineWidth = 5;
  for (let i = 0; i < 4; i++) {
    const a = i * Math.PI / 2 + Math.PI / 4;
    g.beginPath();
    g.moveTo(s / 2, s / 2);
    g.lineTo(s / 2 + Math.cos(a) * s * .48, s / 2 + Math.sin(a) * s * .48);
    g.stroke();
  }
});

/* ------------------------------------------------------------------ light */

// Indoors, lit from a ceiling that is drawn: the strips overhead are where
// the light looks like it comes from, and one directional light does the
// actual shadowing because a dozen real lights would cost more than the rest
// of the frame put together.

/*
 * Metal without something to reflect is black. A rough material gets away
 * with punctual lights alone; a polished one has nothing to be polished
 * against, and the rifle in the corner of the screen came out as a
 * silhouette until this existed.
 *
 * So: a room made of six coloured planes, prefiltered into an environment
 * map once at startup and then thrown away. It costs one render at load and
 * it is the difference between surfaces that look like materials and
 * surfaces that look like flat colours.
 */
function studio() {
  const room = new THREE.Scene();
  room.background = new THREE.Color(0x10151d);

  const lit = (colour, w, h, d, x, y, z) => {
    const m = new THREE.Mesh(
      new THREE.BoxGeometry(w, h, d),
      new THREE.MeshBasicMaterial({ color: colour, side: THREE.BackSide })
    );
    m.position.set(x, y, z);
    room.add(m);
    return m;
  };

  lit(0x6a7381, 30, 30, 30, 0, 0, 0);            // the room itself, a neutral grey
  lit(0xffffff, 14, .5, 14, 0, 9, 0);            // the ceiling it is lit from
  lit(0x7d93b4, 20, 9, .5, 0, 2, -10);           // a cool wall
  lit(0xb0977a, 20, 9, .5, 0, 2, 10);           // a warm one, for the other cheek
  lit(0x4a5059, 30, .5, 30, 0, -9, 0);           // and a floor that bounces

  const pmrem = new THREE.PMREMGenerator(renderer);
  pmrem.compileEquirectangularShader();

  const map = pmrem.fromScene(room, .06).texture;

  pmrem.dispose();
  room.traverse(o => { o.geometry?.dispose(); o.material?.dispose(); });

  return map;
}

/* ------------------------------------------------------------------ world */

const world = new THREE.Group();
scene.add(world);

const solidMaterial = new THREE.MeshStandardMaterial({
  map: concreteTexture, roughness: .92, metalness: .02, color: 0xffffff
});

const CEILING = 4.6;

const bases = new THREE.MeshStandardMaterial({ color: 0x59616d, roughness: .96, metalness: .02 });

// The trim along the top of every box says how tall it is without making you
// walk up to it: red on the wall you fight over, amber on the cover you can
// crouch behind, cold grey on what stops a bullet outright. In an aim map the
// only thing you need to read across the room is which boxes hide your head.
const TRIM = { duel: 0xff4d5e, peek: 0xffc233, blind: 0x44526a };

/*
 * The three heights the map is built from, said in paint:
 *
 *   amber  crouch behind it and you are gone; stand and your chest is out
 *   red    only your head clears it, which is the whole point of the map
 *   grey   nothing goes over this
 *
 * Read across a room it is the only thing you need to know about a box.
 */
function trimFor(height) {
  if (height <= 135) return TRIM.peek;
  if (height <= 175) return TRIM.duel;
  return TRIM.blind;
}

const boxGeometry = new THREE.BoxGeometry(1, 1, 1);

function buildMap(solids, width, length) {
  const w = width * 2 * CM, l = length * 2 * CM;

  const floor = new THREE.Mesh(
    new THREE.PlaneGeometry(w, l),
    new THREE.MeshStandardMaterial({ map: floorTexture, roughness: .78, metalness: .06 })
  );
  floor.rotation.x = -Math.PI / 2;
  floor.receiveShadow = true;
  world.add(floor);

  // the halfway line, which is the only thing painted on the floor
  const line = new THREE.Mesh(
    new THREE.PlaneGeometry(w, .06),
    new THREE.MeshBasicMaterial({ color: 0x55637a, transparent: true, opacity: .35 })
  );
  line.rotation.x = -Math.PI / 2;
  line.position.y = .012;
  world.add(line);

  // the ceiling overhangs the walls: flush with them leaves a hand's width of
  // nothing at the join, and nothing is the colour of the sky in a room
  const ceiling = new THREE.Mesh(
    new THREE.PlaneGeometry(w + 2, l + 2),
    // Lit only from above, the underside of this comes out at two in every
    // channel, which is not a dark ceiling but a hole in the roof. The
    // emissive is the light the strips would be throwing back at it.
    new THREE.MeshStandardMaterial({
      color: 0x39424e, roughness: 1, side: THREE.DoubleSide,
      emissive: 0x222a36, emissiveIntensity: 1
    })
  );
  ceiling.rotation.x = Math.PI / 2;
  ceiling.position.y = CEILING;
  world.add(ceiling);

  // the strips the light is pretending to come from
  const stripMaterial = new THREE.MeshBasicMaterial({ color: 0xe9f2ff, fog: false });
  for (let i = -5; i <= 5; i++) {
    for (const side of [-1, 1]) {
      const strip = new THREE.Mesh(new THREE.PlaneGeometry(w * .3, .28), stripMaterial);
      strip.rotation.x = Math.PI / 2;
      strip.position.set(side * w * .22, CEILING - .04, i * (l / 11.5));
      world.add(strip);
    }
  }

  /*
   * Which end of the map is yours, said in colour. The two sides are far
   * enough apart that there is no other way to know which way you are
   * facing, and a player who has just respawned turned around is a player
   * who walks into their own spawn screen.
   */
  for (const side of [0, 1]) {
    const band = new THREE.Mesh(
      new THREE.BoxGeometry(w * .999, .12, .06),
      new THREE.MeshBasicMaterial({ color: TEAM[side], toneMapped: false, fog: false })
    );
    band.position.set(0, 2.4, (side ? 1 : -1) * (length * CM - .04));
    world.add(band);
  }

  const trims = new Map();

  for (const [x0, y0, z0, x1, y1, z1] of solids) {
    const height = y1 - y0;
    const tall = height > 280;                      // one of the four outer walls

    // A wall that stops where the server stops you leaves a strip of nothing
    // between it and the ceiling, and nothing renders as the background. The
    // collision is the server's; the drawing goes all the way up.
    const top = tall ? CEILING / CM : y1;

    const sx = (x1 - x0) * CM, sy = (top - y0) * CM, sz = (z1 - z0) * CM;
    const cx = (x0 + x1) / 2 * CM, cy = (y0 + top) / 2 * CM, cz = (z0 + z1) / 2 * CM;

    const material = tall
      ? bases
      : solidMaterial.clone();

    if (!tall) {
      material.map = concreteTexture.clone();
      material.map.needsUpdate = true;
      material.map.repeat.set(Math.max(1, sx * 1.4), Math.max(1, sy * 1.4));
      // a little tonal variation, so a row of identical boxes is not identical
      const t = .88 + ((Math.abs(x0 * 31 + z0 * 17) % 100) / 100) * .2;
      material.color.setScalar(t);
    }

    const mesh = new THREE.Mesh(boxGeometry, material);
    mesh.position.set(cx, cy, cz);
    mesh.scale.set(sx, sy, sz);
    mesh.castShadow = !tall;
    mesh.receiveShadow = true;
    world.add(mesh);

    if (tall) continue;

    // a thin cap on top rather than an outline: an outline is a wireframe and
    // looks like a wireframe, a cap looks like paint
    const colour = trimFor(height);
    if (!trims.has(colour)) {
      trims.set(colour, new THREE.MeshBasicMaterial({ color: colour, toneMapped: false }));
    }

    const cap = new THREE.Mesh(boxGeometry, trims.get(colour));
    cap.position.set(cx, sy + .012, cz);
    cap.scale.set(sx + .012, .024, sz + .012);
    world.add(cap);
  }
}

/* ---------------------------------------------------------------- bodies */

// Everybody is a box with a head on it, and the head is exactly the slab the
// server counts as a head. Drawing a rounder one would look better and lie.

const bodyGeometry = new THREE.BoxGeometry(R.girth * 2 * CM, 1, R.girth * 2 * CM);
const headGeometry = new THREE.BoxGeometry(R.girth * 1.5 * CM, R.skull * CM, R.girth * 1.5 * CM);

function nameplate(text, colour) {
  const c = document.createElement('canvas');
  c.width = 256; c.height = 64;
  const g = c.getContext('2d');
  g.font = '600 34px system-ui, sans-serif';
  g.textAlign = 'center';
  g.textBaseline = 'middle';
  g.lineWidth = 6;
  g.strokeStyle = 'rgba(0,0,0,.75)';
  g.strokeText(text, 128, 34);
  g.fillStyle = colour;
  g.fillText(text, 128, 34);
  const t = new THREE.CanvasTexture(c);
  t.colorSpace = THREE.SRGBColorSpace;
  const s = new THREE.Sprite(new THREE.SpriteMaterial({ map: t, depthTest: false, transparent: true }));
  s.scale.set(1.5, .375, 1);
  return s;
}

class Avatar {
  constructor(side, name) {
    this.side = side;
    this.group = new THREE.Group();

    const colour = TEAM[side];

    this.torso = new THREE.Mesh(bodyGeometry, new THREE.MeshStandardMaterial({
      color: colour, roughness: .55, metalness: .12
    }));
    this.torso.castShadow = true;
    this.group.add(this.torso);

    this.head = new THREE.Mesh(headGeometry, new THREE.MeshStandardMaterial({
      color: 0xf2f4f7, roughness: .5, emissive: new THREE.Color(colour), emissiveIntensity: .12
    }));
    this.head.castShadow = true;
    this.group.add(this.head);

    // a stub pointing the way they are looking, so a body across the room
    // tells you whether it has seen you
    this.snout = new THREE.Mesh(
      new THREE.BoxGeometry(.06, .06, .34),
      new THREE.MeshStandardMaterial({ color: 0x20262e, roughness: .6 })
    );
    this.group.add(this.snout);

    /*
     * Only your own side gets a name over it. Drawing everybody's would mean
     * drawing the other side's through the walls they are hiding behind,
     * which is the one thing an aim map cannot have, and testing the depth
     * instead would put half a name behind a box and look broken.
     */
    if (side === mySide) {
      this.plate = nameplate(name, '#' + colour.toString(16).padStart(6, '0'));
      this.group.add(this.plate);
    }

    scene.add(this.group);
  }

  /// Angles here are degrees in the server's convention, turned where three.js
  /// needs them and used as they are where the position is being worked out.
  place(x, y, z, yaw, pitch, crouched, sheltered, dead) {
    const height = (crouched ? R.squat : R.tall) * CM;

    this.group.visible = !dead;
    if (dead) return;

    this.torso.scale.y = height - R.skull * CM;
    this.torso.position.set(x, y + (height - R.skull * CM) / 2, z);

    this.head.position.set(x, y + height - R.skull * CM / 2, z);
    this.head.rotation.y = turned(yaw);

    const facing = World.facing(yaw, pitch);

    const eye = (crouched ? R.eyeSquat : R.eyeTall) * CM;

    this.snout.position.set(x + facing.x * .26, y + eye + facing.y * .26, z + facing.z * .26);
    this.snout.rotation.set(pitch * Math.PI / 180, turned(yaw), 0, 'YXZ');

    if (this.plate) this.plate.position.set(x, y + height + .34, z);

    const ghost = sheltered ? .45 : 1;
    this.torso.material.opacity = this.head.material.opacity = ghost;
    this.torso.material.transparent = this.head.material.transparent = ghost < 1;
  }

  remove() {
    scene.remove(this.group);

    if (this.plate) {
      this.plate.material.map.dispose();
      this.plate.material.dispose();
    }
  }
}

/* ---------------------------------------------------------------- effects */

// Pools, all of them. A shot allocating three objects at ten rounds a second
// across eight players is a garbage collection pause in the middle of a duel,
// which is the one place a pause is unforgivable.

const tracerMaterial = new THREE.MeshBasicMaterial({
  color: 0xffd9a0, transparent: true, blending: THREE.AdditiveBlending,
  depthWrite: false, toneMapped: false
});

const tracers = [];

for (let i = 0; i < 40; i++) {
  const m = new THREE.Mesh(new THREE.CylinderGeometry(.012, .004, 1, 5, 1, true), tracerMaterial.clone());
  m.visible = false;
  m.matrixAutoUpdate = false;
  scene.add(m);
  tracers.push({ mesh: m, life: 0 });
}

const tmpA = new THREE.Vector3(), tmpB = new THREE.Vector3(), tmpC = new THREE.Vector3();
const tmpQ = new THREE.Quaternion();
const UP = new THREE.Vector3(0, 1, 0);

function tracer(from, to) {
  const slot = tracers.find(t => t.life <= 0) || tracers[0];

  tmpA.copy(from); tmpB.copy(to);
  tmpC.subVectors(tmpB, tmpA);

  const len = tmpC.length();
  if (len < .01) return;

  slot.mesh.position.copy(tmpA).addScaledVector(tmpC, .5);
  slot.mesh.quaternion.setFromUnitVectors(UP, tmpC.normalize());
  slot.mesh.scale.set(1, len, 1);
  slot.mesh.updateMatrix();
  slot.mesh.visible = true;
  slot.mesh.material.opacity = 1;
  slot.life = .09;
}

const holes = [];
const holeGeometry = new THREE.PlaneGeometry(.17, .17);

for (let i = 0; i < 70; i++) {
  const m = new THREE.Mesh(holeGeometry, new THREE.MeshBasicMaterial({
    map: holeTexture, transparent: true, depthWrite: false,
    polygonOffset: true, polygonOffsetFactor: -4, opacity: 0
  }));
  m.visible = false;
  scene.add(m);
  holes.push({ mesh: m, life: 0 });
}

let holeAt = 0;

function hole(point, normal) {
  const slot = holes[holeAt = (holeAt + 1) % holes.length];
  slot.mesh.position.copy(point).addScaledVector(normal, .012);
  slot.mesh.lookAt(tmpA.copy(point).add(normal));
  slot.mesh.rotation.z = Math.random() * 6.28;
  slot.mesh.visible = true;
  slot.mesh.material.opacity = .9;
  slot.life = 14;
}

// sparks: one Points cloud reused, because a burst is eight dots for a third

const SPARKS = 220;

let sparkAt = 0;

function sparkle(point, normal, red) {
  for (let i = 0; i < 9; i++) {
    const s = sparkAt = (sparkAt + 1) % SPARKS;
    sparkPos[s * 3] = point.x;
    sparkPos[s * 3 + 1] = point.y;
    sparkPos[s * 3 + 2] = point.z;

    const spray = 2.2 + Math.random() * 2.8;
    sparkVel[s * 3] = (normal.x + (Math.random() - .5) * 1.5) * spray;
    sparkVel[s * 3 + 1] = (normal.y + (Math.random() - .5) * 1.5 + .5) * spray;
    sparkVel[s * 3 + 2] = (normal.z + (Math.random() - .5) * 1.5) * spray;

    sparkLife[s] = .18 + Math.random() * .22;

    if (red) {
      sparkColour[s * 3] = 1; sparkColour[s * 3 + 1] = .12; sparkColour[s * 3 + 2] = .2;
    } else {
      sparkColour[s * 3] = 1; sparkColour[s * 3 + 1] = .78; sparkColour[s * 3 + 2] = .42;
    }
  }
}

function stepSparks(dt) {
  let alive = false;
  for (let s = 0; s < SPARKS; s++) {
    if (sparkLife[s] <= 0) {
      if (sparkPos[s * 3 + 1] > -900) { sparkPos[s * 3 + 1] = -999; }
      continue;
    }
    alive = true;
    sparkLife[s] -= dt;
    sparkVel[s * 3 + 1] -= 16 * dt;
    sparkPos[s * 3] += sparkVel[s * 3] * dt;
    sparkPos[s * 3 + 1] += sparkVel[s * 3 + 1] * dt;
    sparkPos[s * 3 + 2] += sparkVel[s * 3 + 2] * dt;
  }
  sparks.visible = alive;
  sparkGeometry.attributes.position.needsUpdate = true;
  sparkGeometry.attributes.color.needsUpdate = true;
}

/*
 * Where the rifle sits when nothing is happening to it. Far enough from the
 * lens that it reads as a rifle rather than as a slab: at a hand's width it
 * fills a third of the screen and all you can see is the top of the
 * handguard.
 */
const REST = new THREE.Vector3(.086, -.082, -.34);

// held a few degrees off square, the way a rifle actually sits in a shoulder
const HELD = { x: -.015, y: .10, z: -.06 };

const kick = { back: 0, rise: 0, roll: 0 };
const swayAt = { x: 0, y: 0 };
let bob = 0, flashFor = 0, reloadFor = 0;

function stepHand(dt, speed, grounded, reloading) {
  // the bob is driven by distance covered rather than by time, so standing
  // still is still and walking backwards bobs the same as walking forwards
  bob += speed * dt * .06;

  const walk = Math.min(1, speed / R.pace);

  kick.back += (0 - kick.back) * Math.min(1, dt * 13);
  kick.rise += (0 - kick.rise) * Math.min(1, dt * 11);
  kick.roll += (0 - kick.roll) * Math.min(1, dt * 10);

  swayAt.x += (look.yawDelta - swayAt.x) * Math.min(1, dt * 9);
  swayAt.y += (look.pitchDelta - swayAt.y) * Math.min(1, dt * 9);

  // reloading: drop the magazine out, tip the rifle over, put one back
  let dip = 0, tip = 0, magDrop = 0;

  if (reloading > 0) {
    const through = clamp(1 - reloading / R.reload, 0, 1);
    dip = Math.sin(through * Math.PI) * .07;
    tip = Math.sin(through * Math.PI) * .55;
    magDrop = through < .45
      ? (through / .45) * .13
      : Math.max(0, (1 - (through - .45) / .3)) * .13;
  }

  magazine.position.y = -.075 - magDrop;

  hand.position.set(
    REST.x - clamp(swayAt.x, -.035, .035) - Math.sin(bob) * .008 * walk - kick.roll * .01,
    REST.y - clamp(swayAt.y, -.03, .03) - Math.abs(Math.cos(bob)) * .006 * walk - dip
      + (grounded ? 0 : -.016),
    REST.z + kick.back
  );

  hand.rotation.set(
    HELD.x + kick.rise * .9 + clamp(swayAt.y, -.05, .05) * .6,
    HELD.y + clamp(swayAt.x, -.06, .06) * .7 + tip * .35,
    HELD.z + kick.roll + tip
  );

  if (flashFor > 0) {
    flashFor -= dt;
    handFlash.visible = true;
    handFlash.material.rotation += dt * 26;
    handFlash.scale.setScalar(.20 + flashFor * 1.4);
  } else {
    handFlash.visible = false;
  }
}

function recoil() {
  kick.back = .052;
  kick.rise = .07;
  kick.roll = (Math.random() - .5) * .05;
  flashFor = .045;
}

// Synthesised rather than downloaded, for the same reason as the textures.
// Hearing which side of you a shot came from is a real part of playing this,
// so other people's rifles go through a panner and yours does not.

let audio = null, master = null;

function startAudio() {
  if (audio) return;
  audio = new (window.AudioContext || window.webkitAudioContext)();
  master = audio.createGain();
  master.gain.value = .55;
  master.connect(audio.destination);
}

function noise(seconds) {
  const frames = Math.ceil(audio.sampleRate * seconds);
  const buffer = audio.createBuffer(1, frames, audio.sampleRate);
  const data = buffer.getChannelData(0);
  for (let i = 0; i < frames; i++) data[i] = Math.random() * 2 - 1;
  const source = audio.createBufferSource();
  source.buffer = buffer;
  return source;
}

function sink(at) {
  if (!at) return master;
  const panner = audio.createPanner();
  panner.panningModel = 'HRTF';
  panner.distanceModel = 'inverse';
  panner.refDistance = 4;
  panner.rolloffFactor = .85;
  panner.maxDistance = 60;
  panner.positionX.value = at.x;
  panner.positionY.value = at.y;
  panner.positionZ.value = at.z;
  panner.connect(master);
  return panner;
}

function bang(at, mine) {
  if (!audio) return;
  const now = audio.currentTime;
  const out = sink(at);

  // the crack: filtered noise, gone in a tenth of a second
  const crack = noise(.16);
  const band = audio.createBiquadFilter();
  band.type = 'bandpass';
  band.frequency.value = mine ? 2100 : 1750;
  band.Q.value = .7;

  const g1 = audio.createGain();
  g1.gain.setValueAtTime(mine ? .85 : .5, now);
  g1.gain.exponentialRampToValueAtTime(.001, now + .15);

  crack.connect(band).connect(g1).connect(out);
  crack.start(now);

  // the thump underneath it, which is what makes it sound like a rifle and
  // not like static
  const thump = audio.createOscillator();
  thump.type = 'sine';
  thump.frequency.setValueAtTime(190, now);
  thump.frequency.exponentialRampToValueAtTime(48, now + .12);

  const g2 = audio.createGain();
  g2.gain.setValueAtTime(mine ? .7 : .4, now);
  g2.gain.exponentialRampToValueAtTime(.001, now + .13);

  thump.connect(g2).connect(out);
  thump.start(now);
  thump.stop(now + .14);
}

function blip(hz, seconds, volume, type) {
  if (!audio) return;
  const now = audio.currentTime;
  const o = audio.createOscillator();
  o.type = type || 'square';
  o.frequency.value = hz;
  const g = audio.createGain();
  g.gain.setValueAtTime(volume, now);
  g.gain.exponentialRampToValueAtTime(.001, now + seconds);
  o.connect(g).connect(master);
  o.start(now);
  o.stop(now + seconds + .01);
}

const sound = {
  shot: (at, mine) => bang(at, mine),
  hit: () => blip(1550, .07, .3),
  head: () => { blip(2100, .06, .34); setTimeout(() => blip(2700, .07, .3), 45); },
  kill: () => { blip(880, .09, .28, 'triangle'); setTimeout(() => blip(1320, .13, .26, 'triangle'), 80); },
  hurt: () => { if (!audio) return; const n = noise(.12); const g = audio.createGain();
                g.gain.setValueAtTime(.4, audio.currentTime);
                g.gain.exponentialRampToValueAtTime(.001, audio.currentTime + .11);
                const f = audio.createBiquadFilter(); f.type = 'lowpass'; f.frequency.value = 700;
                n.connect(f).connect(g).connect(master); n.start(); },
  die: () => blip(90, .5, .4, 'sawtooth'),
  clack: () => blip(320, .04, .18),
  empty: () => blip(180, .05, .16),
  step: () => { if (!audio) return; const n = noise(.05); const g = audio.createGain();
                g.gain.setValueAtTime(.09, audio.currentTime);
                g.gain.exponentialRampToValueAtTime(.001, audio.currentTime + .05);
                const f = audio.createBiquadFilter(); f.type = 'bandpass'; f.frequency.value = 420;
                n.connect(f).connect(g).connect(master); n.start(); }
};

/* ------------------------------------------------------- what you predict */

// A line-for-line port of Step() in lambda.cs and Clear() in World.cs. It has
// to stay a port: the moment the two disagree, every frame from the server
// corrects you and the movement gets a shiver in it that nothing else
// explains.

let SOLIDS = [];

function clearAt(x, y, z, height) {
  const x0 = x - R.girth, x1 = x + R.girth;
  const z0 = z - R.girth, z1 = z + R.girth;
  const y1 = y + height;

  for (let i = 0; i < SOLIDS.length; i += 6) {
    if (x1 > SOLIDS[i] && x0 < SOLIDS[i + 3]
     && y1 > SOLIDS[i + 1] && y > SOLIDS[i + 1] - 0.001
     && y < SOLIDS[i + 4]
     && z1 > SOLIDS[i + 2] && z0 < SOLIDS[i + 5]) return false;
  }

  return true;
}

const me = {
  x: 0, y: 0, z: 0, vx: 0, vy: 0, vz: 0,
  folded: 0, grounded: true, alive: true
};

function step(s, seconds, b, yaw) {
  let wantCrouch = (b & BTN.crouch) !== 0;

  if (!wantCrouch && s.folded > 0 && !clearAt(s.x, s.y, s.z, R.tall)) wantCrouch = true;

  s.folded = clamp(s.folded + (wantCrouch ? 1 : -1) * seconds / R.fold, 0, 1);

  const rad = yaw * Math.PI / 180;
  const fx = Math.sin(rad), fz = -Math.cos(rad);
  const sx = -fz, sz = fx;

  let wishX = 0, wishZ = 0;

  if (b & BTN.fwd)   { wishX += fx; wishZ += fz; }
  if (b & BTN.back)  { wishX -= fx; wishZ -= fz; }
  if (b & BTN.right) { wishX += sx; wishZ += sz; }
  if (b & BTN.left)  { wishX -= sx; wishZ -= sz; }

  const wish = Math.hypot(wishX, wishZ);
  const top = R.pace * (1 - (1 - R.stoop) * s.folded);

  if (wish > 0.001) {
    wishX /= wish; wishZ /= wish;
    const going = s.vx * wishX + s.vz * wishZ;
    const push = (going < -1 ? R.brake : R.quicken) * (s.grounded ? 1 : R.float);
    s.vx += wishX * push * seconds;
    s.vz += wishZ * push * seconds;
  } else if (s.grounded) {
    const speed = Math.hypot(s.vx, s.vz);
    const slowed = Math.max(0, speed - R.drag * seconds);
    if (speed > 0.001) { s.vx *= slowed / speed; s.vz *= slowed / speed; }
  }

  const carried = Math.hypot(s.vx, s.vz);
  if (carried > top) { s.vx *= top / carried; s.vz *= top / carried; }

  if ((b & BTN.jump) && s.grounded) { s.vy = R.leap; s.grounded = false; }

  s.vy -= R.fall * seconds;

  const height = s.folded > 0.5 ? R.squat : R.tall;

  const toX = s.x + s.vx * seconds;
  if (clearAt(toX, s.y, s.z, height)) s.x = toX; else s.vx = 0;

  const toZ = s.z + s.vz * seconds;
  if (clearAt(s.x, s.y, toZ, height)) s.z = toZ; else s.vz = 0;

  const toY = s.y + s.vy * seconds;

  if (clearAt(s.x, toY, s.z, height)) {
    s.y = toY;
    s.grounded = false;
  } else {
    if (s.vy <= 0) {
      let rest = 0;
      for (let i = 0; i < SOLIDS.length; i += 6) {
        if (s.x + R.girth > SOLIDS[i] && s.x - R.girth < SOLIDS[i + 3]
         && s.z + R.girth > SOLIDS[i + 2] && s.z - R.girth < SOLIDS[i + 5]
         && SOLIDS[i + 4] <= s.y + 1 && SOLIDS[i + 4] > rest) rest = SOLIDS[i + 4];
      }
      s.y = rest;
      s.grounded = true;
    }
    s.vy = 0;
  }

  s.x = clamp(s.x, -half.width + R.girth, half.width - R.girth);
  s.z = clamp(s.z, -half.length + R.girth, half.length - R.girth);

  if (s.y <= 0) { s.y = 0; s.vy = 0; s.grounded = true; }
}

const half = { width: 1300, length: 1950 };

let socket = null, joined = false, myId = null, mySide = 0;

let seq = 0;
const pending = [];                    // inputs the server has not confirmed yet

const frames = [];                     // recent frames, for drawing other people late
const avatars = new Map();

let serverClock = 0, clockAt = 0, started = 0;
let mine = { health: 100, ammo: R.magazine, reloading: 0, returns: 0, kills: 0, deaths: 0 };

// the error between where we thought we were and where the server says we
// were, bled off over a few frames instead of snapped
const drift = { x: 0, y: 0, z: 0 };

let burst = 0, lastFired = -9, nextShot = 0;

function connect(name) {
  const address = new URL('play', location.href.replace(/[^/]*$/, ''));
  address.protocol = address.protocol === 'https:' ? 'wss:' : 'ws:';

  socket = new WebSocket(address);

  socket.onopen = () => socket.send(JSON.stringify({ kind: 'join', name }));

  socket.onmessage = e => {
    const m = JSON.parse(e.data);
    if (m.kind === 'welcome') welcome(m);
    else if (m.kind === 'f') frame(m);
  };

  socket.onclose = () => { joined = false; $('gone').classList.add('on'); document.exitPointerLock(); };
  socket.onerror = () => socket.close();
}

function welcome(m) {
  myId = m.id;
  mySide = m.side;

  half.width = m.width;
  half.length = m.length;

  SOLIDS = new Float64Array(m.solids.length * 6);
  m.solids.forEach((s, i) => SOLIDS.set(s.slice(0, 6), i * 6));

  m.teams?.forEach((colour, side) => { TEAM[side] = parseInt(colour.replace('#', ''), 16); });

  buildMap(m.solids, m.width, m.length);

  gunAccent.color.setHex(TEAM[mySide]);
  gunAccent.emissive.setHex(TEAM[mySide]).multiplyScalar(.35);

  $('tally' + (mySide === 0 ? 'A' : 'B')).classList.add('mine');

  joined = true;
  started = performance.now();

  $('gate').classList.add('off');
  $('hud').classList.add('on');

  if (!locked) canvas.requestPointerLock();
}

function frame(m) {
  serverClock = m.t;
  clockAt = performance.now();

  frames.push(m);
  while (frames.length > 24) frames.shift();

  const was = mine;
  mine = m.m;

  // reconcile: take the server's word for where we were at the input it last
  // saw, then replay everything it has not seen yet
  const authority = m.b.find(b => b.i === myId);

  if (authority && !(authority.f & FLAG.dead)) {
    const beforeX = me.x, beforeY = me.y, beforeZ = me.z;

    me.x = authority.x;
    me.y = authority.y;
    me.z = authority.z;

    while (pending.length && pending[0].seq <= m.i) pending.shift();

    for (const input of pending) step(me, R.tick, input.b, input.yaw);

    drift.x = clamp(drift.x + beforeX - me.x, -220, 220);
    drift.y = clamp(drift.y + beforeY - me.y, -220, 220);
    drift.z = clamp(drift.z + beforeZ - me.z, -220, 220);
  }

  /*
   * The aim belongs here and is sent there, except for one tick. The server
   * turns somebody who has just arrived to face the way out of their spawn,
   * and says so in this field exactly once. Reading it off the body instead
   * would not work: the browser is sending an aim twenty times a second and
   * one of those lands between the server turning them and the frame that
   * would have said so.
   */
  if (mine.t != null) {
    look.yaw = mine.t;
    look.pitch = 0;
    look.yawDelta = look.pitchDelta = 0;
  }

  me.alive = !authority || !(authority.f & FLAG.dead);

  // everything that happened since the last frame
  for (const s of m.s) shotSeen(s);
  for (const k of m.k) killSeen(k);

  if (m.d) hurtSeen(m.d[0], m.d[1]);

  if (was.ammo > mine.ammo) { recoil(); burst++; lastFired = m.t; }
  if (was.reloading <= 0 && mine.reloading > 0) { sound.clack(); setTimeout(sound.clack, 260); }
  if (was.reloading > 0 && mine.reloading <= 0) sound.clack();
  if (was.health > mine.health && mine.health <= 0) sound.die();

  if (m.t - lastFired > R.settle) burst = 0;
}

function shotSeen(s) {
  const from = new THREE.Vector3(s.x0 * CM, s.y0 * CM, s.z0 * CM);
  const to = new THREE.Vector3(s.x1 * CM, s.y1 * CM, s.z1 * CM);

  const isMine = s.i === myId;

  // Your own tracer starts near the barrel you can see rather than at the
  // middle of your own head, which is where the server quite correctly put
  // it. It cannot start exactly at the drawn muzzle: the gun is rendered by
  // its own camera with its own field of view, so the same point is in two
  // places on screen. Close and in front is what it needs to be.
  if (isMine) {
    from.copy(camera.position)
        .addScaledVector(rightOf(camera, tmpA), .1)
        .addScaledVector(camera.up, -.07)
        .addScaledVector(aheadOf(camera, tmpB), .45);
  } else {
    flash(from);
  }

  tracer(from, to);
  sound.shot(isMine ? null : to, isMine);

  if (s.k) {
    // somebody was hit: red sparks back along the shot, and no hole punched
    // in a wall that was not hit
    sparkle(to, tmpB.subVectors(from, to).normalize(), true);

    if (isMine) {
      markHit(s.k === 2);
      s.k === 2 ? sound.head() : sound.hit();
    }
  } else {
    const normal = normalAt(to);
    hole(to, normal);
    sparkle(to, normal, false);
  }
}

// Which face a shot landed on, worked out from the boxes rather than sent:
// the server has no reason to say and the browser already knows the map.
const NORMALS = [
  new THREE.Vector3(-1, 0, 0), new THREE.Vector3(1, 0, 0),
  new THREE.Vector3(0, -1, 0), new THREE.Vector3(0, 1, 0),
  new THREE.Vector3(0, 0, -1), new THREE.Vector3(0, 0, 1)
];

function normalAt(point) {
  const x = point.x / CM, y = point.y / CM, z = point.z / CM;
  const pad = 2;

  let best = null, closest = 6;

  for (let i = 0; i < SOLIDS.length; i += 6) {
    const x0 = SOLIDS[i], y0 = SOLIDS[i + 1], z0 = SOLIDS[i + 2];
    const x1 = SOLIDS[i + 3], y1 = SOLIDS[i + 4], z1 = SOLIDS[i + 5];

    // a face of this box is only the answer if the point is on this box, so
    // the other two axes have to be inside it first
    if (x < x0 - pad || x > x1 + pad || y < y0 - pad || y > y1 + pad
     || z < z0 - pad || z > z1 + pad) continue;

    const faces = [x - x0, x1 - x, y - y0, y1 - y, z - z0, z1 - z];

    for (let f = 0; f < 6; f++) {
      if (Math.abs(faces[f]) < closest) { closest = Math.abs(faces[f]); best = NORMALS[f]; }
    }
  }

  return best || NORMALS[3];
}

const feed = $('feed');

function killSeen(k) {
  const row = document.createElement('div');
  row.className = 'kill';

  const by = frames.at(-1)?.b.find(b => b.n === k.by);
  const who = frames.at(-1)?.b.find(b => b.n === k.who);

  row.innerHTML = `<span class="${by?.t ? 'b' : 'a'}">${escaped(k.by)}</span>`
                + `<em>${k.head ? '<span class="head">head</span>' : 'shot'}</em>`
                + `<span class="${who?.t ? 'b' : 'a'}">${escaped(k.who)}</span>`;

  feed.append(row);
  while (feed.children.length > 5) feed.firstChild.remove();
  setTimeout(() => row.remove(), 6000);

  if (k.by === myName) { sound.kill(); markHit(k.head, true); }

  if (k.who === myName) {
    $('byline').innerHTML = `${escaped(k.by)} ${k.head ? 'took your head off' : 'shot you'}`;
  }
}

function aheadOf(of, into) { return into.set(0, 0, -1).applyQuaternion(of.quaternion); }

function drawHud(dt) {

  $('again').classList.toggle('on', dead);
  if (dead) $('counter').textContent = mine.returns.toFixed(1);

  // the crosshair opens by exactly the cone the server would spread by
  const speed = Math.hypot(me.vx, me.vz);
  const cone = spread(speed, !me.grounded, me.folded > .5, burst);
  const pixels = Math.tan(cone * Math.PI / 180) * (innerHeight / 2) / Math.tan(camera.fov * Math.PI / 360);
  const gap = clamp(4 + pixels, 4, 90);

  $('ct').style.top = (-gap - 9) + 'px';
  $('cb').style.top = gap + 'px';
  $('cl').style.left = (-gap - 9) + 'px';
  $('cr').style.left = gap + 'px';

  drawTally();

  if (showBoard) drawBoard();
}

function drawBoard() {
  const latest = frames.at(-1);
  if (!latest) return;

  const rows = [...latest.b].sort((p, q) => q.k - p.k || p.d - q.d);

  $('rows').innerHTML = rows.map(b =>
    `<tr class="${b.t ? 'b' : 'a'}${b.i === myId ? ' me' : ''}">`
    + `<td>${escaped(b.n)}</td>`
    + `<td class="num">${b.k}</td>`
    + `<td class="num">${b.d}</td>`
    + `<td class="num">${(b.k / Math.max(1, b.d)).toFixed(2)}</td></tr>`
  ).join('');
}

function drawTally() {
  const latest = frames.at(-1);
  if (!latest) return;

  let a = 0, b = 0;
  for (const body of latest.b) (body.t ? (b += body.k) : (a += body.k));

  $('tallyA').textContent = a;
  $('tallyB').textContent = b;
}

/* ------------------------------------------------------- everybody else */

// Drawn a tenth of a second late, between the two frames either side of that
// moment. The cost is that you shoot at where somebody was rather than where
// they are; the server knows this and rewinds by the same amount when it
// judges the shot, which is why the browser tells it which moment it drew.

function drawOthers(at) {
  let before = null, after = null;

  for (const f of frames) {
    if (f.t <= at && (!before || f.t > before.t)) before = f;
    if (f.t >= at && (!after || f.t < after.t)) after = f;
  }

  if (!before && !after) return;

  before ||= after;
  after ||= before;

  const span = after.t - before.t;
  const share = span > .0001 ? clamp((at - before.t) / span, 0, 1) : 0;

  const seen = new Set();

  for (const body of after.b) {
    if (body.i === myId) continue;

    seen.add(body.i);

    let avatar = avatars.get(body.i);

    if (!avatar) {
      avatar = new Avatar(body.t, body.n);
      avatars.set(body.i, avatar);
    }

    const old = before.b.find(b => b.i === body.i) || body;

    const x = (old.x + (body.x - old.x) * share) * CM;
    const y = (old.y + (body.y - old.y) * share) * CM;
    const z = (old.z + (body.z - old.z) * share) * CM;

    // angles wrap, so they are mixed the short way round or somebody spins
    // through three hundred degrees every time they cross north
    const yaw = mixAngle(old.a / 100, body.a / 100, share);
    const pitch = (old.p + (body.p - old.p) * share) / 100;

    avatar.place(x, y, z, yaw, pitch,
                 !!(body.f & FLAG.crouched),
                 !!(body.f & FLAG.sheltered),
                 !!(body.f & FLAG.dead));
  }

  for (const [id, avatar] of avatars) {
    if (!seen.has(id)) { avatar.remove(); avatars.delete(id); }
  }
}

function mixAngle(a, b, t) {
  let d = ((b - a) % 360 + 540) % 360 - 180;
  return a + d * t;
}

/* ------------------------------------------------------------- the clock */

// One input every fifty milliseconds, whatever the frame rate: the server
// takes one tick per input, so sending two hundred a second would run the
// game four times too fast for whoever had the best monitor.

let owed = 0, walked = 0, sinceStep = 0, pumped = performance.now();

/*
 * This runs off its own clock rather than off the render loop's, because the
 * two are not the same thing. The server steps twenty times a second whatever
 * happens; a browser drawing at fifteen frames a second that sent fifteen
 * inputs a second would be predicting three quarters of the movement the
 * server is about to apply, and would spend the whole game being corrected.
 */
function pump() {
  if (!joined || socket.readyState !== 1) return;

  const now = performance.now();

  owed += Math.min(.5, (now - pumped) / 1000);
  pumped = now;

  let sent = 0;

  while (owed >= R.tick && sent < 10) {
    owed -= R.tick;
    sent++;

    const b = me.alive ? buttons() : 0;

    if (me.alive) step(me, R.tick, b, look.yaw);
    else { me.vx = me.vy = me.vz = 0; }

    pending.push({ seq: ++seq, b, yaw: look.yaw });
    // four seconds of unacknowledged input, which is far more than any
    // connection worth playing on. Dropping one that has not been answered
    // yet would leave it out of the replay and put a step of movement back
    while (pending.length > 80) pending.shift();

    socket.send(JSON.stringify({
      kind: 'i', i: seq, b,
      y: +look.yaw.toFixed(2),
      p: +look.pitch.toFixed(2),
      s: +(serverNow() - R.behind).toFixed(3)
    }));

    // footsteps, timed to how far you have actually gone rather than to a
    // metronome, so crouch-walking is quieter and slower
    if (me.grounded && me.alive) {
      walked += Math.hypot(me.vx, me.vz) * R.tick;
      if (walked > 175) { walked = 0; sound.step(); }
    } else {
      walked = 0;
    }
  }

  // whatever is left in the accumulator is exactly how long it has been
  // since the last step, which is what the camera extrapolates by
  sinceStep = owed;
}

const serverNow = () => serverClock + (performance.now() - clockAt) / 1000;

/* ------------------------------------------------------------- the frame */

const eye = new THREE.Vector3();

let last = performance.now();

function render(now) {
  requestAnimationFrame(render);

  const dt = Math.min(.1, (now - last) / 1000);
  last = now;

  look.yawDelta *= Math.pow(.0016, dt);
  look.pitchDelta *= Math.pow(.0016, dt);

  if (joined) {
    pump();

    drift.x *= Math.pow(.0009, dt);
    drift.y *= Math.pow(.0009, dt);
    drift.z *= Math.pow(.0009, dt);

    // between two inputs the position is carried forward by the velocity it
    // already has. Drawing the last stepped position instead would step the
    // camera twenty times a second on a screen refreshing a hundred and forty
    const ahead = Math.min(sinceStep, R.tick);

    const height = me.folded > .5 ? R.eyeSquat : R.eyeTall;

    eye.set(
      (me.x + drift.x + me.vx * ahead) * CM,
      (me.y + drift.y + me.vy * ahead) * CM + height * CM,
      (me.z + drift.z + me.vz * ahead) * CM
    );

    camera.position.copy(eye);
    camera.rotation.set(look.pitch * Math.PI / 180, turned(look.yaw), 0, 'YXZ');

    // dying drops the camera, lets go of the aim, and puts the rifle away:
    // a crosshair over a corpse is a crosshair that says you can still shoot
    if (!me.alive) {
      camera.position.y = Math.max(.3, camera.position.y - .9);
      camera.rotation.z = .5;
    }

    hand.visible = me.alive;
    $('cross').style.opacity = me.alive ? '' : '0';

    drawOthers(serverNow() - R.behind);
    drawHud(dt);

    if (audio) {
      const l = audio.listener;
      if (l.positionX) {
        l.positionX.value = eye.x; l.positionY.value = eye.y; l.positionZ.value = eye.z;
        const f = new THREE.Vector3(0, 0, -1).applyQuaternion(camera.quaternion);
        l.forwardX.value = f.x; l.forwardY.value = f.y; l.forwardZ.value = f.z;
        l.upX.value = 0; l.upY.value = 1; l.upZ.value = 0;
      }
    }

    stepHand(dt, Math.hypot(me.vx, me.vz), me.grounded, mine.reloading);
  }

  // effects age whether or not you are playing
  for (const t of tracers) {
    if (t.life <= 0) continue;
    t.life -= dt;
    t.mesh.material.opacity = Math.max(0, t.life / .09);
    if (t.life <= 0) t.mesh.visible = false;
  }

  for (const h of holes) {
    if (h.life <= 0) continue;
    h.life -= dt;
    if (h.life < 3) h.mesh.material.opacity = Math.max(0, h.life / 3) * .9;
    if (h.life <= 0) h.mesh.visible = false;
  }

  for (const f of flashes) {
    if (f.life <= 0) continue;
    f.life -= dt;
    if (f.life <= 0) f.sprite.visible = false;
  }

  flashLight.intensity *= Math.pow(.00002, dt);

  stepSparks(dt);

  renderer.render(scene, camera);

  // the gun goes on top, in its own pass, so nothing in the world can be in
  // front of it and the fog cannot reach it
  renderer.autoClear = false;
  renderer.clearDepth();
  renderer.render(overlay, handCam);
  renderer.autoClear = true;
}
