// What a link is, and what this will accept as one.
//
// No storage here: lambda.cs keeps the file. This is the part that decides
// whether an address is one worth keeping, which is the part worth reading.

/// <summary>A stored link.</summary>
public record Link(string Key, string Address, DateTime Created, int Follows);

/// <summary>What is sent to shorten something.</summary>
public record Target(string Address);

/// <summary>The answer, carrying the short key when it worked.</summary>
public record Made(bool Ok, string Problem, string Key, string Path);

static class Links
{
    private const string Alphabet = "abcdefghijkmnopqrstuvwxyz23456789";

    /// <summary>
    /// A short key, from an alphabet without the characters people misread.
    /// </summary>
    /// <remarks>
    /// No l, o, 0 or 1: these get read aloud, typed from a screenshot and
    /// written down, and the pairs that look alike cost more than the few
    /// extra characters of alphabet are worth.
    /// </remarks>
    public static string NewKey()
    {
        var bytes = System.Security.Cryptography.RandomNumberGenerator.GetBytes(7);

        var key = new StringBuilder(7);

        foreach (var b in bytes)
        {
            key.Append(Alphabet[b % Alphabet.Length]);
        }

        return key.ToString();
    }

    /// <summary>
    /// Whether an address may be shortened, and why not when it may not.
    /// </summary>
    /// <remarks>
    /// Only http and https. A shortener that will hand out javascript: or
    /// data: addresses is a way of getting somebody else's page to run your
    /// script, and this one is handing links to strangers by design.
    /// </remarks>
    public static string Refuse(string address)
    {
        var trimmed = (address ?? "").Trim();

        if (trimmed.Length is < 8 or > 2000)
        {
            return "That does not look like an address.";
        }

        if (!Uri.TryCreate(trimmed, UriKind.Absolute, out var parsed))
        {
            return "That is not a complete address.";
        }

        if (parsed.Scheme != "http" && parsed.Scheme != "https")
        {
            return "Only http and https addresses can be shortened.";
        }

        return null;
    }
}
