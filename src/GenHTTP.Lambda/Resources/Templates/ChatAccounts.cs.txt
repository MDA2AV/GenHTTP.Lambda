// Who may talk, and how that is decided.
//
// No files are read or written here on purpose: this is the part worth reading
// on its own, and it is easier to be sure about when it only turns arguments
// into answers. lambda.cs owns the storage and hands the accounts in.

/// <summary>What is sent to register or to sign in.</summary>
public record Credentials(string Name, string Password);

/// <summary>A stored account. The password itself is not kept anywhere.</summary>
public record Account(string Name, string Salt, string Hash);

/// <summary>The answer to an attempt, carrying a token when it worked.</summary>
public record Answer(bool Ok, string Problem, string Token, string Name);

/// <summary>One line said in the room.</summary>
public record Line(string From, string Text, DateTime At);

/// <summary>How many are here, and how many have ever registered.</summary>
public record Population(int Here, int Registered);

static class Passwords
{
    // PBKDF2 over a random salt. Being an example is not a reason to show
    // somebody the wrong way to keep a password - a bare hash of one is
    // guessable at a rate that makes the hashing pointless.
    public static string Hash(string password, string salt)
        => Convert.ToHexString(System.Security.Cryptography.Rfc2898DeriveBytes.Pbkdf2(
               password, Convert.FromHexString(salt), 60_000,
               System.Security.Cryptography.HashAlgorithmName.SHA256, 32));

    public static string Random(int bytes)
        => Convert.ToHexString(System.Security.Cryptography.RandomNumberGenerator.GetBytes(bytes));

    /// <summary>
    /// Whether a password matches an account, compared in constant time.
    /// </summary>
    /// <remarks>
    /// The comparison takes the same time whether the first character differs
    /// or the last, so how long the answer took says nothing about how nearly
    /// right the guess was.
    /// </remarks>
    public static bool Matches(Account account, string password)
        => System.Security.Cryptography.CryptographicOperations.FixedTimeEquals(
               Encoding.UTF8.GetBytes(Hash(password ?? "", account.Salt)),
               Encoding.UTF8.GetBytes(account.Hash));
}

static class Accounts
{
    /// <summary>
    /// Makes an account, or says why it could not be made.
    /// </summary>
    public static Answer Register(ConcurrentDictionary<string, Account> users, Credentials request)
    {
        var name = (request?.Name ?? "").Trim();

        if (name.Length is < 2 or > 20)
        {
            return new Answer(false, "A name is between two and twenty characters.", null, null);
        }

        if ((request?.Password ?? "").Length < 6)
        {
            return new Answer(false, "A password needs six characters at least.", null, null);
        }

        var salt = Passwords.Random(16);

        if (!users.TryAdd(name, new Account(name, salt, Passwords.Hash(request.Password, salt))))
        {
            return new Answer(false, "That name is taken.", null, null);
        }

        return new Answer(true, null, Passwords.Random(24), name);
    }

    /// <summary>
    /// Checks a name and password.
    /// </summary>
    /// <remarks>
    /// An unknown name and a wrong password are answered identically, so this
    /// does not become a way of asking who has an account here.
    /// </remarks>
    public static Answer Login(ConcurrentDictionary<string, Account> users, Credentials request)
    {
        var name = (request?.Name ?? "").Trim();

        if (!users.TryGetValue(name, out var account) || !Passwords.Matches(account, request?.Password))
        {
            return new Answer(false, "That name and password do not go together.", null, null);
        }

        return new Answer(true, null, Passwords.Random(24), account.Name);
    }
}
