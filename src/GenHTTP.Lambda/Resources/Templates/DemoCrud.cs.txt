// A task list: records you can list, search, add, change and remove.
//
//   GET    tasks/           every task, newest first; tasks/?search=milk narrows it down
//   GET    tasks/{id}       one task                       404 if there is none
//   POST   tasks/           add one, JSON body             201, or 400 when it is not valid
//   PUT    tasks/{id}       change one, JSON body          404 if there is none
//   DELETE tasks/{id}      remove one                     204
//
//   openapi.json            the generated description of the API
//   scalar/                 a page to try the API in the browser
//   anything else           the page in web/, which uses the API
//
// The files:
//
//   lambda.cs     this one: the wiring, and bringing the database up to date
//   Tasks.cs      the API - a class whose methods are the routes
//   Store.cs      the tasks in the database, in plain SQL
//   migrations/   the SQL that makes the table, applied once by Evolve
//   web/          the front end, shipped with the code and served as it is
// [demo]
//
// This is a read-only demo. To build on it, create a lambda of your own from
// it - a copy is yours to change, and starts with a database of its own.
// [/demo]

// The database belongs to the lambda, not to a version: every version reads
// the same one, so each brings it up to the schema it expects before it
// serves anything. The migrations are SQL files shipped as assets, applied in
// order and once each - Evolve remembers which in a table of its own, so this
// runs on every start and only ever changes what is new. A migration that was
// applied is never edited: a change to the schema is the next file, V2__...
using (var connection = Database.GetConnection())
{
    var evolve = new Evolve(connection, message => Console.WriteLine(message))
    {
        Locations = [Assets.Root + "migrations"],
        IsEraseDisabled = true
    };

    evolve.Migrate();
}

var store = new TaskStore(limit: 200);

if (store.Count() == 0)
{
    Tasks.Seed(store);
}

return Layout.Create()
             .AddService("tasks", new Tasks(store))
             .AddOpenApi()
             .AddScalar()
             // last, so that it only answers what nothing above did
             .Add(Assets.App("web"));
