// Tic-tac-toe against whoever else opens the page, or against the house when
// nobody is around. The server holds every board and decides every move; the
// browser only draws what it is told and asks to play a cell.
//
//   GET  ./            the game, a single page application in resources/web/
//   WS   play          the socket the page opens; JSON messages both ways
//   GET  api/stats     games played and how they ended
//
// The files:
//
//   lambda.cs               this one: the wiring, and how the games ended, kept in the database
//   Lobby.cs                the websocket handler: pairs players, runs games, talks to sockets
//   Board.cs                the rules and the house's moves - no sockets, easy to reason about
//   Protocol.cs             every message that goes over the socket, in both directions
//   resources/migrations/   the SQL that makes the table of outcomes, applied once by Evolve
//   resources/web/          the page and its script
//   docs/, tests/           what it is for and why, and how it is tested
// [demo]
//
// This is a read-only demo. To build on it, create a lambda of your own from
// it - a copy is yours to change, and starts with a database of its own.
// [/demo]

// every version reads the same database, so each brings it up to the schema
// it expects first; Evolve applies what is new in resources/migrations/, once each
using (var connection = Database.GetConnection())
{
    var evolve = new Evolve(connection, message => Console.WriteLine(message))
    {
        Locations = [Resources.Root + "migrations"],
        IsEraseDisabled = true
    };

    evolve.Migrate();
}

// the lobby calls this whenever a game ends; the counts outlive a restart. One
// row per outcome, counted up where it is
void Record(string winner)
{
    var outcome = winner is "X" or "O" ? winner : "draw";

    using var db = new GameContext(Database.GetConnection());

    // the transaction takes the database's one writer's place before it reads,
    // so two games ending at once cannot both find no row and both add one
    using var transaction = db.Database.BeginTransaction();

    // one UPDATE ... SET games = games + 1, without reading the row first
    if (db.Outcomes.Where(o => o.Winner == outcome).ExecuteUpdate(s => s.SetProperty(o => o.Games, o => o.Games + 1)) == 0)
    {
        db.Outcomes.Add(new Outcome { Winner = outcome, Games = 1 });
        db.SaveChanges();
    }

    transaction.Commit();
}

// how the games so far have ended, added up from the outcomes
Stats Count()
{
    using var db = new GameContext(Database.GetConnection());

    var counts = db.Outcomes.ToDictionary(o => o.Winner, o => o.Games);

    return new Stats(counts.Values.Sum(), counts.GetValueOrDefault("X"), counts.GetValueOrDefault("O"), counts.GetValueOrDefault("draw"));
}

var lobby = new Lobby(Record);

var api = Inline.Create()
                .Get("stats", () => Count());

return Layout.Create()
             .Add("play", Websocket.Reactive().Handler(lobby))
             .Add("api", api)
             .Add(Resources.App("web"));

/// <summary>How many games ended one way: X, O or draw.</summary>
public sealed class Outcome
{
    public string Winner { get; set; }

    public int Games { get; set; }
}

/// <summary>The lambda's database, as Entity Framework Core sees it.</summary>
public sealed class GameContext(SqliteConnection connection) : DbContext
{
    public DbSet<Outcome> Outcomes => Set<Outcome>();

    protected override void OnConfiguring(DbContextOptionsBuilder options)
        => options.UseSqlite(connection, contextOwnsConnection: true);

    // the table is the migration's; this says how to read it
    protected override void OnModelCreating(ModelBuilder model)
        => model.Entity<Outcome>().ToTable("outcomes").HasKey(o => o.Winner);
}
