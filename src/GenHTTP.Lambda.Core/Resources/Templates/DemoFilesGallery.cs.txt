// What may be uploaded, where it is kept, and the list of everything that is.
//
// Only a fixed list of file types is accepted, and a file is stored under a
// name this class makes up, ending in the extension of its type. That is the
// important part: the content type a file is served with comes from that
// extension. Accept an .html or .svg from a stranger and serve it from here,
// and their script runs on this lambda's address, with its users' tokens.
//
// The files go into the workspace, through the few things this class is handed
// to do with it; the list of them - names, types, who may remove which - is a
// table in the database, read and written through Entity Framework Core. Every
// call makes a context on a connection of its own, uses it synchronously and
// lets it go.

using System.Globalization;
using System.Security.Cryptography;

/// <summary>What the gallery may do with the workspace.</summary>
public record Storage(Action<string, byte[]> WriteBytes, Action<string> Delete);

/// <summary>One upload as everybody sees it.</summary>
/// <param name="Url">Where it is served, relative to the lambda</param>
public record Upload(string Id, string Name, long Size, string Type, DateTime Uploaded, string Url);

/// <summary>What the uploader gets back: the entry, and the key to remove it with.</summary>
public record Uploaded(Upload Upload, string Key);

/// <summary>An upload as the table holds it: what everybody sees, and the hash of its key.</summary>
public sealed class UploadEntry
{
    public string Id { get; set; }

    public string Name { get; set; }

    public long Size { get; set; }

    public string Type { get; set; }

    public DateTime Uploaded { get; set; }

    public string Url { get; set; }

    public string KeyHash { get; set; }

    /// <summary>What everybody may see of it - never the hash.</summary>
    public Upload Public() => new(Id, Name, Size, Type, Uploaded, Url);
}

/// <summary>The lambda's database, as Entity Framework Core sees it.</summary>
public sealed class GalleryContext(SqliteConnection connection) : DbContext
{
    public DbSet<UploadEntry> Uploads => Set<UploadEntry>();

    protected override void OnConfiguring(DbContextOptionsBuilder options)
        => options.UseSqlite(connection, contextOwnsConnection: true);

    protected override void OnModelCreating(ModelBuilder model)
    {
        // the table and its columns are the migration's; this only maps them
        var upload = model.Entity<UploadEntry>().ToTable("uploads");

        upload.Property(u => u.KeyHash).HasColumnName("key_hash");

        // ISO 8601 text in UTC, which sorts as it should and reads well
        upload.Property(u => u.Uploaded).HasConversion(u => u.ToUniversalTime().ToString("O", CultureInfo.InvariantCulture),
                                                       u => DateTime.Parse(u, CultureInfo.InvariantCulture, DateTimeStyles.RoundtripKind));
    }
}

public sealed class Gallery(Storage storage)
{
    // the workspace limits only the room all files take together, so each
    // upload is held small enough that the gallery stays at 12 MB at most
    private const int MaxBytes = 512 * 1024;

    // anybody can upload to a demo, so it keeps the newest ones only
    private const int MaxUploads = 24;

    private static readonly Dictionary<string, string> Types = new(StringComparer.OrdinalIgnoreCase)
    {
        [".png"] = "image/png",
        [".jpg"] = "image/jpeg",
        [".jpeg"] = "image/jpeg",
        [".gif"] = "image/gif",
        [".webp"] = "image/webp",
        [".pdf"] = "application/pdf",
        [".txt"] = "text/plain"
    };

    public List<Upload> List()
    {
        using var db = new GalleryContext(Database.GetConnection());

        // a projection reads the columns it names - the hash stays in the table
        return db.Uploads.OrderByDescending(u => u.Uploaded)
                         .Select(u => new Upload(u.Id, u.Name, u.Size, u.Type, u.Uploaded, u.Url))
                         .ToList();
    }

    public async Task<Uploaded> UploadAsync(string name, Stream body)
    {
        // what the browser calls the file, without any folders in front of it
        name = (name ?? "").Trim().Split('/', '\\')[^1];

        var dot = name.LastIndexOf('.');

        var extension = dot < 0 ? "" : name[dot..].ToLowerInvariant();

        if (!Types.TryGetValue(extension, out var type))
        {
            throw new ProviderException(ResponseStatus.UnsupportedMediaType,
                $"Only {string.Join(", ", Types.Keys)} files can be uploaded here.");
        }

        // the body is awaited: it comes over the network, which the database does not
        var content = await ReadAsync(body);

        if (content.Length == 0)
        {
            throw new ProviderException(ResponseStatus.BadRequest, "The file is empty.");
        }

        var id = Convert.ToHexString(RandomNumberGenerator.GetBytes(8)).ToLowerInvariant();

        var key = Convert.ToHexString(RandomNumberGenerator.GetBytes(16));

        // the stored name is ours; the uploader's name is only ever shown as text
        var stored = id + (extension == ".jpeg" ? ".jpg" : extension);

        var entry = new UploadEntry
        {
            Id = id,
            Name = name.Length > 80 ? name[..80] : name,
            Size = content.Length,
            Type = type,
            Uploaded = DateTime.UtcNow,
            Url = $"files/{stored}",
            KeyHash = Hash(key)
        };

        // the file first: an entry without its file would be a broken picture
        storage.WriteBytes($"uploads/{stored}", content);

        using var db = new GalleryContext(Database.GetConnection());

        db.Uploads.Add(entry);

        // the oldest go, file and entry, once there are too many
        foreach (var old in db.Uploads.OrderByDescending(u => u.Uploaded).Skip(MaxUploads - 1).ToList())
        {
            Drop(db, old);
        }

        // the new entry and the old ones going are one SaveChanges, and so one transaction
        db.SaveChanges();

        return new Uploaded(entry.Public(), key);
    }

    public void Remove(string id, string key)
    {
        using var db = new GalleryContext(Database.GetConnection());

        var entry = db.Uploads.Find(id ?? "")
                 ?? throw new ProviderException(ResponseStatus.NotFound, "There is no such upload.");

        if (!CryptographicOperations.FixedTimeEquals(Encoding.UTF8.GetBytes(entry.KeyHash), Encoding.UTF8.GetBytes(Hash(key ?? ""))))
        {
            throw new ProviderException(ResponseStatus.Forbidden, "Only whoever uploaded this can remove it.");
        }

        Drop(db, entry);

        db.SaveChanges();
    }

    /// <summary>
    /// Reads the body, but never more than the limit - a request announcing a
    /// small file and sending a large one is cut off all the same.
    /// </summary>
    private static async Task<byte[]> ReadAsync(Stream body)
    {
        using var buffer = new MemoryStream();

        var chunk = new byte[16 * 1024];

        int read;

        while ((read = await body.ReadAsync(chunk)) > 0)
        {
            if (buffer.Length + read > MaxBytes)
            {
                throw new ProviderException(ResponseStatus.RequestEntityTooLarge,
                    $"A file can be {MaxBytes / 1024} KB at most.");
            }

            buffer.Write(chunk, 0, read);
        }

        return buffer.ToArray();
    }

    /// <summary>
    /// Removes an entry with the next SaveChanges, and its file straight away.
    /// </summary>
    private void Drop(GalleryContext db, UploadEntry entry)
    {
        db.Uploads.Remove(entry);

        storage.Delete(entry.Url.Replace("files/", "uploads/"));
    }

    private static string Hash(string key) => Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(key)));
}
