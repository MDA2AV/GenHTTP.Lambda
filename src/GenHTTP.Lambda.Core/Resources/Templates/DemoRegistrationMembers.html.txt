    <p style="margin-top:0">
      You only see this page with a valid session. The page itself is a plain file anybody could download -
      what is protected is the API it reads, which answers <code>401</code> without a session and <code>403</code>
      with one it does not know.
    </p>
    <h2>Newest members</h2>

<script type="module">
  import { call, escape } from './session.js';

  const back = () => location.replace('./');

  try {
    const [me, members] = await Promise.all([call('api/me'), call('api/members')]);

    document.getElementById('name').textContent = me.name;
    document.getElementById('joined').textContent =
      `Member since ${new Date(me.joined).toLocaleString()}, one of ${me.members}.`;

    document.getElementById('members').innerHTML = members
      .map((m) => `<li><span>${escape(m.name)}</span><span class="dim small">${new Date(m.joined).toLocaleDateString()}</span></li>`)
      .join('');

    document.body.classList.add('ready');
  } catch (e) {
    // no session, or one that has expired: back to the landing page
    if (e.signedOut) back();
    else document.body.textContent = e.message;
  }

  document.getElementById('logout').addEventListener('click', async () => {
    try { await call('logout', { method: 'POST' }); } finally { back(); }
  });
</script>
