// The question, the counts, and everybody who is listening.
//
// Every browser on the events stream gets a Listener: a small queue that the
// poll puts updates into and that stream takes them out of. A vote therefore
// never waits on a slow browser - it drops the update into every queue and is
// done, and each stream writes at its own pace. A queue that falls behind
// forgets its oldest updates, which is fine: only the newest counts matter.
//
// The counts are kept in the lambda's database, a row per option, read and
// written through Entity Framework Core and counted up there: the database
// takes one write at a time, so two votes at the same moment both count.

using System.Threading.Channels;

/// <summary>What the browser sends to vote.</summary>
public record Vote(string Option);

public record Option(string Id, string Label, int Votes);

public record Results(string Question, List<Option> Options, int Total);

/// <summary>
/// One event for the stream: what to send, and the event type - null for an
/// ordinary message.
/// </summary>
public record Update(string Event, object Data);

/// <summary>How many votes an option has, as the table holds it.</summary>
public sealed class Tally
{
    public string Option { get; set; }

    public int Votes { get; set; }
}

/// <summary>The lambda's database, as Entity Framework Core sees it.</summary>
public sealed class PollContext(SqliteConnection connection) : DbContext
{
    public DbSet<Tally> Tallies => Set<Tally>();

    protected override void OnConfiguring(DbContextOptionsBuilder options)
        => options.UseSqlite(connection, contextOwnsConnection: true);

    // the table is the migration's; this says how to read it
    protected override void OnModelCreating(ModelBuilder model)
        => model.Entity<Tally>().ToTable("votes").HasKey(t => t.Option);
}

public sealed class Poll
{
    private const string Question = "Tabs or spaces?";

    private static readonly (string Id, string Label)[] Choices =
    [
        ("tabs", "Tabs"),
        ("spaces", "Spaces"),
        ("formatter", "Whatever the formatter says")
    ];

    private readonly List<Listener> _listeners = [];

    private readonly object _lock = new();

    public Results Results()
    {
        Dictionary<string, int> votes;

        using (var db = new PollContext(Database.GetConnection()))
        {
            votes = db.Tallies.ToDictionary(t => t.Option, t => t.Votes);
        }

        var options = Choices.Select(c => new Option(c.Id, c.Label, votes.GetValueOrDefault(c.Id))).ToList();

        return new Results(Question, options, options.Sum(o => o.Votes));
    }

    public void Cast(Vote vote)
    {
        var option = vote?.Option;

        if (!Choices.Any(c => c.Id == option))
        {
            throw new ProviderException(ResponseStatus.BadRequest, $"Vote for one of: {string.Join(", ", Choices.Select(c => c.Id))}.");
        }

        using (var db = new PollContext(Database.GetConnection()))
        {
            // the transaction takes the database's one writer's place before it
            // reads, so two first votes cannot both find no row and both add one
            using var transaction = db.Database.BeginTransaction();

            // one UPDATE ... SET votes = votes + 1, without reading the row first
            if (db.Tallies.Where(t => t.Option == option).ExecuteUpdate(s => s.SetProperty(t => t.Votes, t => t.Votes + 1)) == 0)
            {
                db.Tallies.Add(new Tally { Option = option, Votes = 1 });
                db.SaveChanges();
            }

            transaction.Commit();
        }

        Tell(new Update(null, Results()));
    }

    /// <summary>
    /// Starts listening. Dispose the listener when the browser is gone, which
    /// the stream does with <c>using</c>.
    /// </summary>
    public Listener Listen()
    {
        var listener = new Listener(this);

        lock (_lock)
        {
            _listeners.Add(listener);
        }

        Watchers();

        return listener;
    }

    internal void Forget(Listener listener)
    {
        lock (_lock)
        {
            _listeners.Remove(listener);
        }

        Watchers();
    }

    private void Watchers()
    {
        int count;

        lock (_lock)
        {
            count = _listeners.Count;
        }

        // a number is sent as it is, under an event type of its own
        Tell(new Update("watchers", count));
    }

    private void Tell(Update update)
    {
        Listener[] listeners;

        lock (_lock)
        {
            listeners = [.. _listeners];
        }

        foreach (var listener in listeners)
        {
            listener.Post(update);
        }
    }
}

public sealed class Listener(Poll poll) : IDisposable
{
    // a few updates at most; when full, the oldest is dropped for the newest
    private readonly Channel<Update> _queue = Channel.CreateBounded<Update>(
        new BoundedChannelOptions(8) { FullMode = BoundedChannelFullMode.DropOldest });

    internal void Post(Update update) => _queue.Writer.TryWrite(update);

    /// <summary>
    /// The next update, or null when there was none for as long as given.
    /// </summary>
    public async Task<Update> NextAsync(TimeSpan patience)
    {
        using var timeout = new CancellationTokenSource(patience);

        try
        {
            return await _queue.Reader.ReadAsync(timeout.Token);
        }
        catch (OperationCanceledException)
        {
            return null;
        }
    }

    public void Dispose() => poll.Forget(this);
}
